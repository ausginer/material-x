  it.each(CASES)('%s', async (name, html) => {

    document.body.append(root);
    await nextFrame(); await nextFrame();
    const fmt = (rs: axe.Result[]): string => rs.flatMap((r) => r.nodes.map((n) => `${r.id}@${JSON.stringify(n.target)}`)).join(' ');
    const t = performance.now();
    const result = await axe.run(root);
    const a = performance.now() - t; total += a;
    const t2 = performance.now();
    const noContrast = await axe.run(root, { rules: { 'color-contrast': { enabled: false } } });
    const b = performance.now() - t2;
    const inner = [...result.passes, ...result.violations].filter((r) => r.nodes.some((n) => n.target.length > 0)).map((r) => r.id);
    out.push(`${pad(a)} ${pad(b)} ${name}\n      violations: ${fmt(result.violations) || '-'}\n      incomplete: ${fmt(result.incomplete) || '-'}\n      passes: ${inner.join(',')}`);
