## Scratch reporter: url-reporter.ts
// Scratch, not committed (RF-60 step 2): records every URL string that the
// worker's imports send to the project server, through each project's
// ModuleTransformService.fetch, with whether the answer says `invalidate`.
import { appendFileSync } from 'node:fs';

const log = process.env.RF60_URL_LOG!;

export default class UrlReporter {
  onInit(vitest: { projects: Array<{ name: string; _transformService: any }> }) {
    for (const project of vitest.projects) {
      const service = project._transformService;
      const fetch = service.fetch.bind(service);
      service.fetch = async (url: string, importer: string | undefined, ...rest: any[]) => {
        const start = performance.now();
        const result = await fetch(url, importer, ...rest);
        if (url.includes('/gen-') || importer?.includes('/gen-')) {
          const kind = 'externalize' in result ? 'externalize' : 'cache' in result ? 'cache' : 'code';
          appendFileSync(
            log,
            `${JSON.stringify({ project: project.name, url, importer: importer ?? null, cached: rest[2]?.cached ?? null, kind, invalidate: result.invalidate ?? null, ms: Math.round(performance.now() - start) })}\n`,
          );
        }
        return result;
      };
    }
  }
}

## Scratch test: packages/vite-traits-plugin/tests/rf60-repro.node.test.ts
// Scratch, not committed (RF-60 step 2): the brand row's setup and assertion,
// repeated in one worker with the imports still concurrent.
import { expect, it } from 'vitest';
import { flattenToDir } from './helpers.ts';

type BrandedTrait = (abstract new () => object) & { readonly brand: symbol };
type Ctor = new () => object;

const repeats = Number(process.env.RF60_REPEATS ?? 1000);

it('repeats the brand row', { timeout: 3_600_000 }, async () => {
  const failures: string[] = [];
  for (let i = 0; i < repeats; i += 1) {
    const gen = await flattenToDir('consumer.ts', {
      augment: ['checkable.ts', 'nameable.ts'],
      passthrough: ['base.ts'],
    });
    try {
      const [composedMod, flatMod, checkableMod, nameableMod] =
        (await Promise.all([
          import(gen.url('foo-composed.ts')),
          import(gen.url('foo-flat.ts')),
          import(gen.url('checkable.ts')),
          import(gen.url('nameable.ts')),
        ])) as [
          { default: Ctor },
          { default: Ctor },
          { Checkable: BrandedTrait },
          { Nameable: BrandedTrait },
        ];
      const { Checkable } = checkableMod;
      const { Nameable } = nameableMod;
      for (const [label, C] of [
        ['composed', composedMod.default],
        ['flattened', flatMod.default],
      ] as const) {
        const instance = new C();
        if (!(instance instanceof Checkable) || !(instance instanceof Nameable)) {
          failures.push(`repeat ${i}, ${label}\n${gen.brandReport(instance, { Checkable, Nameable })}`);
        }
      }
    } finally {
      await gen.dispose();
    }
  }
  console.log(`RF60 repeats ${repeats}, failures ${failures.length}`);
  expect(failures).toEqual([]);
});

// Scratch, not committed (RF-60 step 2): the smallest reproduction found. No
// flattener: a module that creates a symbol, and a module that re-exports it by
// a relative import, written to a fresh directory and imported concurrently
// by their file URLs.
import { mkdtemp, rm, writeFile } from 'node:fs/promises';
import { join } from 'node:path';
import { fileURLToPath, pathToFileURL } from 'node:url';
import { expect, it } from 'vitest';

const repeats = Number(process.env.RF60_REPEATS ?? 1000);
const root = fileURLToPath(new URL('./fixtures/', import.meta.url));

it('evaluates a module once per directory', { timeout: 3_600_000 }, async () => {
  let twice = 0;
  for (let i = 0; i < repeats; i += 1) {
    const dir = await mkdtemp(join(root, 'gen-'));
    try {
      await writeFile(join(dir, 'mod.ts'), "export const s: symbol = Symbol('s');\n");
      await writeFile(join(dir, 'user.ts'), "export { s as viaUser } from './mod.ts';\n");
      const [{ viaUser }, { s }] = (await Promise.all([
        import(pathToFileURL(join(dir, 'user.ts')).href),
        import(pathToFileURL(join(dir, 'mod.ts')).href),
      ])) as [{ viaUser: symbol }, { s: symbol }];
      if (viaUser !== s) twice += 1;
    } finally {
      await rm(dir, { recursive: true, force: true });
    }
  }
  console.log(`RF60 minimal repeats ${repeats}, second evaluations ${twice}`);
  expect(twice).toBe(0);
});
