// Runtime form: the trait layer holds a get/set pair; the component overrides the getter only.
class Base {}
const Layer = class extends Base { get x() { return 'trait'; } set x(v) { this._x = v; } };
class Composed extends Layer { get x() { return 'component'; } }
// Flattened form, as lowerModule emits it: trait accessors first in the component's own body.
class Flat extends Base { get x() { return 'trait'; } set x(v) { this._x = v; } get x() { return 'component'; } }
for (const C of [Composed, Flat]) {
  const o = new C();
  let r; try { o.x = 1; r = `assigned, _x=${o._x}`; } catch (e) { r = `${e.constructor.name}`; }
  console.log(C.name.padEnd(8), 'get:', o.x, '| set:', r);
}
// super.K from the component reaches the trait layer at runtime, the base after flattening.
class Composed2 extends Layer { get y() { return super.x; } }
class Flat2 extends Base { get x() { return 'trait'; } get y() { return super.x; } }
console.log('Composed2 super.x:', new Composed2().y, '| Flat2 super.x:', new Flat2().y);
