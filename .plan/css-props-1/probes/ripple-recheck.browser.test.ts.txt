async function release(how: string) {
  const host = document.createElement('div'); document.body.append(host);
  const e = make(); host.append(e); await wait(80); const ev: string[] = [];
  for (const type of ['pointerdown', 'mousedown', 'pointerup', 'mouseup', 'click', 'pointerleave', 'pointercancel']) e.addEventListener(type, (x) => ev.push(`${type}${type === 'click' ? ':' + (x as MouseEvent).detail + (x.isTrusted ? ':trusted' : '') : ''}`));
  document.addEventListener('click', (x) => ev.push('doc click on ' + (x.target as Element).localName), { once: true });
  const r = t(e).getBoundingClientRect(); await commands.pointerPress(r.left + r.width / 2, r.top + r.height / 2); await wait(100);
  const held = anims(e).length;
  if (how === 'put back') { e.remove(); host.append(e); }
  if (how === 'put back later') { e.remove(); await wait(50); host.append(e); }
  if (how === 'moved by insertBefore') host.prepend(e);
  if (how === 'moveBefore' && 'moveBefore' in host) (host as any).moveBefore(e, host.firstChild);
  await wait(50); const mid = anims(e).length; await commands.pointerRelease(); await wait(60);
  log(`release, ${how}: animations held/after the change/60 ms after the release, events`, [held, mid, anims(e).length, ev]); await wait(500);
  host.remove();
}
