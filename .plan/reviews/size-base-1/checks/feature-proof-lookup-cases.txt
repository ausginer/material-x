### extract.mjs
// Writes the `run` text of verify.yml's measurements steps, verbatim.
import { createRequire } from 'node:module';
import { readFileSync, writeFileSync } from 'node:fs';
const [tree, out] = process.argv.slice(2);
const YAML = createRequire(`${tree}/package.json`)('yaml');
const wf = YAML.parse(readFileSync(`${tree}/.github/workflows/verify.yml`, 'utf8'));
for (const step of wf.jobs.measurements.steps) {
  if (step.name === 'Sizes at the merge base') writeFileSync(`${out}/lookup.sh`, step.run);
  if (step.name === 'Report') writeFileSync(`${out}/report.sh`, step.run);
}
### mock.mjs
// A local stand-in for api.github.com on a unix socket. MODE picks the answer.
import { createServer } from 'node:http';
import { readFileSync, appendFileSync, rmSync } from 'node:fs';
const [socket, mode, zipPath, log] = process.argv.slice(2);
const json = (res, code, body) => { res.writeHead(code, { 'content-type': 'application/json' }); res.end(JSON.stringify(body)); };
const deny = (res, code) => json(res, code, { message: code === 403 ? 'Resource not accessible by integration' : `status ${code}` });
rmSync(socket, { force: true });
createServer((req, res) => {
  appendFileSync(log, `${req.method} ${req.url}\n`);
  const u = new URL(req.url, 'http://x');
  if (/\/actions\/workflows\/verify\.yml\/runs$/.test(u.pathname)) {
    if (mode === '403-runs') return deny(res, 403);
    if (mode === '429-runs') return deny(res, 429);
    if (mode === '500-runs') return deny(res, 500);
    if (mode === 'no-run') return json(res, 200, { total_count: 0, workflow_runs: [] });
    return json(res, 200, { total_count: 1, workflow_runs: [{ id: 777, head_sha: u.searchParams.get('head_sha') }] });
  }
  if (/\/actions\/runs\/777\/artifacts$/.test(u.pathname)) {
    if (mode === '403-artifacts') return deny(res, 403);
    if (mode === 'no-artifact') return json(res, 200, { total_count: 0, artifacts: [] });
    const expired = mode === 'expired';
    return json(res, 200, { total_count: 1, artifacts: [{ id: 1, name: 'size-reading', expired, size_in_bytes: 100,
      archive_download_url: 'https://api.github.com/repos/ausginer/material-x/actions/artifacts/1/zip' }] });
  }
  if (/\/actions\/artifacts\/1\/zip$/.test(u.pathname)) {
    if (mode === '403-download') return deny(res, 403);
    res.writeHead(200, { 'content-type': 'application/zip' });
    return res.end(readFileSync(zipPath));
  }
  deny(res, 404);
}).listen(socket);
### case.sh
#!/usr/bin/env bash
# case.sh <name> <mode|real> <event>: runs the two steps' text verbatim under bash -e.
set -u
B=$(cd "$(dirname "$0")" && pwd); S=$(dirname "$B"); name=$1; mode=$2; event=$3
C="${S:?}/cases/${name:?}"; rm -rf "${C:?}"; mkdir -p "$C/temp"
cp "$S/head-size.json" "$C/temp/head-size.json"; : > "$C/output"; : > "$C/summary"; : > "$C/requests"
export RUNNER_TEMP="$C/temp" GITHUB_OUTPUT="$C/output" GITHUB_STEP_SUMMARY="$C/summary" GITHUB_REPOSITORY=ausginer/material-x BASE_REF=main
if [ "$mode" != real ]; then
  mkdir -p "$C/gh"; printf 'http_unix_socket: %s\n' "${SOCK:-../cases/$name/api.sock}" > "$C/gh/config.yml"
  (cd "$C" && exec node "$B/mock.mjs" api.sock "$mode" "$B/base.zip" "$C/requests") & mock=$!
  for _ in $(seq 50); do [ -S "$C/api.sock" ]  && break; sleep 0.1; done
  export GH_CONFIG_DIR="$C/gh" GH_TOKEN=dummy-not-a-token
fi
echo "### $name (event $event, API: $mode)"
cd "$S/tree"
if [ "$event" = pull_request ]; then
  echo '$ bash -e <Sizes at the merge base>'
  PATH="${STUBPATH:-}${STUBPATH:+:}$PATH" bash -e "$B/lookup.sh" 2>&1 | sed "s#$S#<w>#g"
  echo "exit ${PIPESTATUS[0]}; GITHUB_OUTPUT: [$(tr '\n' ' ' < "$C/output")]; base-reading/head-size.json: $([ -f "$C/temp/base-reading/head-size.json" ] && echo present || echo absent)"
  [ -s "$C/requests" ] && sed 's/^/  mock: /' "$C/requests"
fi
[ "$mode" != real ] && kill "$mock" 2>/dev/null
missing=$(sed -n 's/^missing=//p' "$C/output")
echo "\$ bash -e <Report> (EVENT=$event, MISSING=${missing:-<empty>})"
EVENT=$event MISSING=$missing bash -e "$B/report.sh" > "$C/report.log" 2>&1; rc=$?
echo "exit $rc; the summary's sentence and first row:"
sed -n '4p' "$C/summary"; grep -m1 '^| box-quad' "$C/summary"
echo
