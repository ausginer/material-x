// @ts-nocheck
// The tab stop of a group against the contract's own words, read from the value, a log of the document's focusin and the children. No hook is the
// reference. From packages/core, with zz-roving-<name>.ts of the hooks in src/controllers: copy to tests/controllers/zz-model.browser.test.ts
import { it } from 'vitest';
import { commands, server } from 'vitest/browser';
import { ControlledElement } from '../../src/element.ts';
import type { Checkable } from '../../src/traits/checkable.ts';
import { compose, type Composed } from '../../src/traits/composition.ts';
import type { Disableable } from '../../src/traits/disableable.ts';
import type { Valuable } from '../../src/traits/valuable.ts';
import { defineCE, nameCE } from '../browser.ts';

const hooks = import.meta.glob('../../src/controllers/zz-roving-*.ts', { eager: true });
const HOOKS = Object.fromEntries(Object.entries(hooks).map(([p, m]) => [/zz-roving-(\w+)/.exec(p)[1], m.useRovingTabindex]));
const ItemComposition: Composed<typeof ControlledElement, [typeof Checkable, typeof Valuable, typeof Disableable]> = compose(ControlledElement);
// Focusable before the group manages it, as a switch button is.
class ItemBase extends ItemComposition { constructor() { super(); this.attachShadow({ mode: 'open', delegatesFocus: true }).append(document.createElement('button')); } }
const HostComposition: Composed<typeof ControlledElement, [typeof Valuable]> = compose(ControlledElement);
class HostBase extends HostComposition {}
const task = () => new Promise((r) => { const c = new MessageChannel(); c.port1.onmessage = () => r(); c.port2.postMessage(0); });
const hostTag = {};
for (const [n, hook] of Object.entries(HOOKS)) {
  hostTag[n] = nameCE();
  defineCE(hostTag[n], class extends HostBase { constructor() { super(); this.attachShadow({ mode: 'open' }).append(document.createElement('slot')); hook(this); } });
}
const ready = nameCE();
defineCE(ready, class extends ItemBase {});
const plain = nameCE();
defineCE(plain, class extends HTMLElement {});
const random = (seed) => () => { seed = (seed + 0x6d2b79f5) | 0; let t = Math.imul(seed ^ (seed >>> 15), 1 | seed); t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t; return ((t ^ (t >>> 14)) >>> 0) / 2 ** 32; };
// The kinds of step for each stratum. `steps` follow a first batch of children.
const STRATA = {
  'A: one batch; value, focus, blur': ['value', 'focus', 'blur'],
  'B: A, and ready or late items added': ['value', 'focus', 'blur', 'add'],
  'C: B, and items removed': ['value', 'focus', 'blur', 'add', 'remove'],
  'D: C, and items moved or put back': ['value', 'focus', 'blur', 'add', 'remove', 'move', 'back'],
  'E: A, and items disabled': ['value', 'focus', 'blur', 'disable'],
  'F: D, E, and host in and out': ['value', 'focus', 'blur', 'add', 'remove', 'move', 'back', 'disable', 'inout'],
  'G: B, several steps in a task': ['value', 'focus', 'blur', 'add', '+'],
  'L: B, with an item focused in the task that adds it': ['value', 'focus', 'blur', 'add', 'addfocus'],
  'M: L, with items removed, moved, put back, and the host in and out': ['value', 'focus', 'blur', 'add', 'addfocus', 'remove', 'move', 'back', 'inout'],
  'H: one batch, nothing else': [],
  'I: one batch, items added, no value': ['add'],
  'J: one batch with late elements that are no items, or never defined': ['other'],
  'K: one batch of one way; the focus moves, and a value': ['value', 'focus', 'focus'],
};
function make(kinds, r) {
  const pick = (a) => a[Math.floor(r() * a.length)];
  const VALUES = ['a', 'b', 'c', 'd', 'e', 'f'];
  let n = 0;
  const item = () => ({ id: n, kind: pick(kinds.includes('other') ? ['ready', 'A', 'B', 'N', 'X'] : ['ready', 'ready', 'clone', 'A', 'A', 'B']), value: VALUES[n++ % 6], disabled: kinds.includes('disable') && r() < 0.15 });
  const steps = [{ do: 'batch', items: Array.from({ length: 1 + Math.floor(r() * 4) }, item), value: kinds.includes('value') && r() < 0.5 ? pick(VALUES) : null, late: r() < 0.3 }];
  const list = kinds.filter((k) => k !== '+' && k !== 'other');
  const count = list.length ? 1 + Math.floor(r() * 5) : 0;
  const own = [];
  for (let i = 0; i < count; i += 1) {
    const k = pick(list);
    const s = { do: k, at: r(), sync: kinds.includes('+') && r() < 0.5 };
    if (k === 'add' || k === 'addfocus') { s.item = item(); s.where = r(); }
    if (k === 'value') s.value = pick([...VALUES, 'none', null]);
    if (k === 'move' || k === 'back') s.where = r();
    own.push(s);
  }
  for (const k of ['A', 'B', ...(kinds.includes('other') ? ['N'] : [])]) own.splice(Math.floor(r() * (own.length + 1)), 0, { do: 'define', k, sync: kinds.includes('+') && r() < 0.5 });
  return [...steps, ...own];
}
async function run(hook, steps, isLate) {
  document.body.replaceChildren();
  const tags = isLate ? { ready, clone: ready, A: nameCE(), B: nameCE() } : { ready, clone: ready, A: ready, B: ready };
  // N is defined as an element that is no item, and X never.
  tags.N = isLate ? nameCE() : plain; tags.X = nameCE();
  const defined = new Set(['ready']);
  const h = document.createElement(hostTag[hook]);
  let last; const log = (e) => { const t = e.target; if (t.parentNode === h && !t.hasAttribute('disabled')) last = t; };
  document.addEventListener('focusin', log, true);
  const items = new Map();
  let removed = [];
  const create = ({ id, kind, value, disabled }) => {
    let e;
    if (isLate && kind !== 'ready') {
      const t = document.createElement('template');
      t.innerHTML = `<${tags[kind]}></${tags[kind]}>`;
      e = t.content.cloneNode(true).firstChild;
    } else e = document.createElement(tags[kind]);
    e.setAttribute('value', value);
    if (disabled) e.setAttribute('disabled', '');
    e.kind = kind; e.id = `i${id}`;
    items.set(e.id, e);
    return e;
  };
  const place = (e, where) => { const rest = [...h.children].filter((c) => c !== e); h.insertBefore(e, rest[Math.floor(where * (rest.length + 1))] ?? null); };
  for (const s of steps) {
    const c = [...h.children];
    const at = c[Math.floor((s.at ?? 0) * c.length)];
    if (s.do === 'batch') {
      if (s.value) h.setAttribute('value', s.value);
      if (!s.late) document.body.append(h);
      h.append(...s.items.map(create));
      if (s.late) { await task(); await task(); document.body.append(h); }
    } else if (s.do === 'define') {
      if (isLate) defineCE(tags[s.k], class extends (s.k === 'N' ? HTMLElement : ItemBase) {});
      defined.add(s.k);
    } else if (s.do === 'value') s.value === null ? h.removeAttribute('value') : h.setAttribute('value', s.value);
    // Focus reaches an item that is one in both runs, and that the task's slot change has read.
    else if (s.do === 'focus') { if (at && (defined.has(at.kind) || at.kind === 'clone')) at.focus(); }
    else if (s.do === 'blur') document.activeElement?.blur?.();
    else if (s.do === 'add') place(create(s.item), s.where);
    else if (s.do === 'addfocus') { const e = create(s.item); place(e, s.where); if (defined.has(e.kind) || e.kind === 'clone') e.focus(); }
    else if (s.do === 'remove') { if (at) { at.remove(); removed.push(at); } }
    else if (s.do === 'move') { if (at) place(at, s.where); }
    else if (s.do === 'back') { const e = removed.shift(); if (e) place(e, s.where); }
    else if (s.do === 'disable') at?.toggleAttribute('disabled');
    else if (s.do === 'inout') h.isConnected ? h.remove() : document.body.append(h);
    if (!s.sync) { await task(); await task(); if (last && last.parentNode !== h) last = undefined; }
  }
  const wasOut = !h.isConnected;
  if (wasOut) document.body.append(h);
  await task(); await task();
  if (last && last.parentNode !== h) last = undefined;
  const state = () => {
    const all = [...h.children];
    const enabled = all.filter((e) => !e.hasAttribute('disabled'));
    const match = h.hasAttribute('value') ? enabled.find((e) => e.getAttribute('value') === h.getAttribute('value')) : undefined;
    const want = match ?? enabled.find((e) => e === last) ?? enabled[0];
    const stops = all.filter((e) => e.getAttribute('tabindex') === '0');
    const managed = all.every((e) => e.hasAttribute('tabindex'));
    if (!managed || stops.length !== (want ? 1 : 0)) return 'count';
    if (stops[0] === want) return 'contract';
    if (match && stops[0] === last) return 'focus moved it off the value';
    return 'other';
  };
  const show = () => [...h.children].map((e) => `${e.id}${e.hasAttribute('disabled') ? '!' : ''}:${e.getAttribute('tabindex')}`).join(' ') + ` value ${h.getAttribute('value')} last ${last?.id} active ${document.activeElement?.id}`;
  const end = state(); const endShow = show();
  // A slot change: a read of the same items.
  h.append(document.createTextNode('x'));
  await task(); await task();
  const read = state(); const readShow = show();
  document.removeEventListener('focusin', log, true);
  return { end, read, endShow, readShow };
}
it('model', async () => {
  const out = [`# ${server.browser}`];
  const N = Number(import.meta.env.VITE_N ?? 400);
  for (const [name, kinds] of Object.entries(STRATA)) {
    if (kinds.includes('other')) continue;
    out.push(`${name}: of ${N}, at the end | after one more slot change`);
    for (const hook of Object.keys(HOOKS)) for (const isLate of [true, false]) {
      const n = { end: {}, read: {} }; let shortest;
      for (let seed = 1; seed <= N; seed += 1) {
        const steps = make(kinds, random(seed * 7919));
        const got = await run(hook, steps, isLate);
        for (const k of ['end', 'read']) n[k][got[k]] = (n[k][got[k]] ?? 0) + 1;
        const isBad = got.read !== 'contract' || (got.end !== 'contract' && got.end !== 'focus moved it off the value');
        if (isBad) { const t = JSON.stringify(steps); if (!shortest || t.length < shortest[0].length) shortest = [t, got.end, got.endShow, got.read, got.readShow]; }
      }
      out.push(`  ${hook}, ${isLate ? 'late' : 'all ready'}: ${JSON.stringify(n.end)} | ${JSON.stringify(n.read)}`);
      if (shortest && hook === 'head') out.push(...shortest.map((x) => `      ${x}`));
    }
    await commands.writeFile(`./zz-model-${server.browser}.txt`, `${out.join('\n')}\n`);
  }
}, 3600000);
