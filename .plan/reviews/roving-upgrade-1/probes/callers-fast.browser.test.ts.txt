function look(e) {
  const s = [];
  const all = (n, d) => {
    const st = STATES.filter((x) => { try { return n.matches(`:state(${x})`); } catch { return false; } });
    const b = n.getBoundingClientRect();
    s.push(`${d}${n.localName}[${attrs(n)}]{${st}}<${refs(n)}>${Math.round(b.width)}x${Math.round(b.height)} ${n.localName === 'button' ? getComputedStyle(n).borderRadius : n.localName === 'slot' ? getComputedStyle(n).padding : ''}`);
    for (const c of n.children) all(c, d + ' ');
    if (n.shadowRoot) for (const c of n.shadowRoot.children) all(c, `${d}#`);
  };
  all(e, '');
  return s;
}

async function compare(out) {
  for (const [name, [html, changes]] of Object.entries(CASES)) {
    const read = {};
    for (const way of ['right', 'out', 'late', 'lateUndone']) {
      document.body.replaceChildren();
      const e = el(html);
      if (way === 'late') {
        // Content and changes before the first connect.
        await task();
      } else {
        document.body.append(e);
        await task();
        if (way !== 'right') { e.remove(); await task(); }
      }
      for (const c of changes) { c(e); await task(); }
      if (!e.isConnected) { document.body.append(e); await task(); }
      if (way === 'lateUndone') { e.remove(); document.body.append(e); await task(); e.remove(); await task(); document.body.append(e); await task(); }
      await new Promise((r) => setTimeout(r, 4));
      read[way] = look(e);
    }
    for (const way of ['out', 'late', 'lateUndone']) {
      const bad = read.right.map((x, i) => (x === read[way][i] ? null : `   right ${x}\n   got   ${read[way][i]}`)).filter(Boolean);
      out.push(`${name} | ${way}: ${bad.length || read.right.length !== read[way].length ? `DIFFERS\n${bad.join('\n')}` : 'same'}`);
    }
  }
}
// Only the comparison, in short tasks and many rounds. It reads whether a listing is of another element than the case's.
it('compare', async () => {
  const lines = [`# ${server.browser}`];
  let stale = 0;
  const names = {};
  for (let round = 1; round <= 12; round += 1) {
    const out = [];
    await compare(out);
    for (const o of out) {
      const [first, right, got] = o.split('\n');
      if (!right) continue;
      const tag = (l) => /^\s*(?:right|got)\s+#?([a-z0-9-]+)\[/.exec(l ?? '')?.[1];
      const want = /<(mx-[a-z-]+)/.exec(CASES[first.split(' | ')[0]][0])[1];
      // The first differing line of a stale listing is the element itself.
      if (/^\s+got\s{3}mx-/.test(got) && tag(got) !== want) { stale += 1; names[`round ${round}: ${tag(got)}`] = (names[`round ${round}: ${tag(got)}`] ?? 0) + 1; }
    }
  }
  lines.push(`stale listings: ${stale} of ${12 * Object.keys(CASES).length * 3}; ${JSON.stringify(names)}`);
  await commands.writeFile(`./out-${server.browser}.txt`, `${lines.join('\n')}\n`);
}, 1200000);
