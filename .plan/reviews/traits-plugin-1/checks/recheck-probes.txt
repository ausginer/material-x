// RF-44 item 2b recheck probe: language pairs that the check copies. Analyzes and lowers in memory against slot d's
// plugin source, then parses the lowered consumer with its own id.
import { analyzeModule, parseModule } from '/workspaces/worktrees/d/packages/vite-traits-plugin/src/analyze.ts';
import { lowerModule } from '/workspaces/worktrees/d/packages/vite-traits-plugin/src/lower.ts';
import { createRequire } from 'node:module';
const { parseSync } = createRequire('/workspaces/worktrees/d/packages/vite-traits-plugin/package.json')('oxc-parser');

const D = '/v/';
const origin = (member: string) => `import { trait } from '@ydinjs/core/traits/primitives.js';
const $t = Symbol('t');
export const T = trait((base) => class extends base {
  ${member}
}, $t);
`;
const consumer = (ext: string, originFile: string) => `import { impl } from '@ydinjs/core/traits/attributes.js';
import { T } from './${originFile}';
class Base extends HTMLElement {}
const FooBase = impl(Base, [T]);
export class Foo extends FooBase {}
`;
async function run(label: string, originFile: string, member: string, consumerExt: string) {
  const files: Record<string, string> = { [D + originFile]: origin(member) };
  const id = D + 'c.' + consumerExt;
  files[id] = consumer(consumerExt, originFile);
  const loader = {
    resolve: async (s: string) => (s.startsWith('./') ? D + s.slice(2) : null),
    load: async (i: string) => { const c = files[i]; if (c === undefined) throw new Error('no ' + i); return c; },
  };
  const { compositions, bailouts } = await analyzeModule(id, files[id]!, loader);
  const out = compositions.length ? lowerModule(files[id]!, compositions).toString() : '';
  const parsed = out ? parseSync(id, out) : undefined;
  console.log(`${label.padEnd(44)} ${originFile} -> c.${consumerExt.padEnd(3)} flattened=${compositions.length} bailouts=${JSON.stringify(bailouts.map((b) => b.reason))}` +
    (parsed ? ` lowered parse errors=${JSON.stringify(parsed.errors.map((e) => e.message))}` : ''));
  return { out, parsed };
}

const ASSERT = 'm(x: unknown): number { return <number>x; }';
const ARROW = 'm(): unknown { return <T>(x: T) => x; }';
for (const ext of ['ts', 'mts', 'cts', 'tsx']) await run('type assertion <number>x', 't.ts', ASSERT, ext);
for (const ext of ['ts', 'mts', 'cts']) await run('generic arrow <T>(x: T) => x', 't.ts', ARROW, ext);

const AMBIG = 'm(a, b, c) { return a < b > (c); }';
const js = parseSync('m.js', `(a, b, c) => a < b > (c)`).program.body[0].expression.body;
const ts = parseSync('m.ts', `(a, b, c) => a < b > (c)`).program.body[0].expression.body;
console.log(`\n"a < b > (c)" parses as ${js.type}${js.operator ? ' ' + js.operator : ''} in .js, and as ${ts.type} in .ts` +
  (ts.typeArguments ? ` with type arguments` : ''));
const { out } = await run('comparison a < b > (c)', 't.js', AMBIG, 'ts');
const line = out.split('\n').find((l) => l.includes('a < b'));
console.log('lowered member line:', line?.trim());
