// RF-44 item 2c recheck: analyzeModule over every material-x source that calls impl(, with the plugin's loader shape.
import { readFile } from 'node:fs/promises';
import { dirname, resolve } from 'node:path';
import { fileURLToPath } from 'node:url';
const { analyzeModule } = await import(process.argv[2] + '/analyze.ts');
const loader = {
  resolve: async (s: string, importer: string) => {
    if (s.startsWith('.')) return resolve(dirname(importer), s);
    if (s.startsWith('@ydinjs/')) { try { return fileURLToPath(import.meta.resolve(s)); } catch { return null; } }
    return null;
  },
  load: (i: string) => readFile(i, 'utf8'),
};
let total = 0, method = 0;
for (const id of process.argv.slice(3)) {
  const code = await readFile(id, 'utf8');
  const { compositions, bailouts } = await analyzeModule(id, code, loader);
  total += compositions.length;
  const m = compositions.filter((c: any) => JSON.stringify(c).includes('"kind":"method"')).length;
  method += m;
  console.log(`${id.replace(/.*\/src\//, '')}: flattened=${compositions.length} withMethodTrait=${m} bailouts=${JSON.stringify(bailouts.map((x: any) => x.reason))}`);
}
console.log(`total flattened=${total}, with a method trait=${method}`);
