function dump(root: Element): string[] {
  const out: string[] = [];
  const walk = (e: Element, path: string) => {
    for (const p of [null, '::before', '::after']) {
      const s = getComputedStyle(e, p);
      for (const n of PROPS) { out.push(`${path}${p ?? ''}|${n}|${s.getPropertyValue(n)}`); }
    }
    const kids = [...e.children, ...(e.shadowRoot ? [...e.shadowRoot.children] : [])];
    kids.forEach((k, i) => { if (!['STYLE'].includes(k.tagName)) walk(k, `${path}/${k.localName}${k.className ? '.' + k.className : ''}#${i}`); });
  };
  walk(root, root.localName); return out;
}
