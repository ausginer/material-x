// The sentences of JSDoc and card.mdx that a page can read.
//   node docs.mjs <engine>
import { playwright, code } from './lib.mjs';
const E = process.argv[2];
const browser = await playwright[E].launch();
console.log(`# ${E} ${browser.version()}`);
const open = async (html) => { const page = await browser.newPage({ viewport: { width: 400, height: 400 } }); await page.setContent(`<!doctype html><style>body{margin:40px}</style>${html}<script>${code('head')}</script>`); await page.evaluate(() => new Promise((r) => requestAnimationFrame(() => requestAnimationFrame(r)))); return page; };
for (const [n, inner] of [['no focusable content', '<p>text</p>'], ['a button inside', '<button id=b>b</button>'], ['a link inside', '<a id=b href="#">l</a>'], ['an mx-button inside', '<mx-button id=b>b</mx-button>'], ['a tabindex -1 inside', '<p tabindex=-1>b</p>']]) for (const tag of ['mx-card', 'div']) {
  const page = await open(`<input id=first><${tag} id=c style="display:block;inline-size:200px;max-block-size:100px;overflow:auto">${inner}<div style="block-size:400px"></div></${tag}><input id=last>`);
  await page.focus('#first'); const seq = []; for (let i = 0; i < 3; i++) { await page.keyboard.press(E === 'webkit' ? 'Alt+Tab' : 'Tab'); seq.push(await page.evaluate(() => document.activeElement.id || document.activeElement.localName)); }
  console.log(`scrolling ${tag}, ${n}: first > ${seq.join(' > ')}`); await page.close();
}
for (const css of ['', 'mx-card{padding:40px}', 'mx-card{padding:40px!important}', '*{padding:0!important}', 'mx-card{box-sizing:content-box!important;inline-size:200px}']) { const page = await open(`<style>${css}</style><mx-card id=c><div id=i style="block-size:10px"></div></mx-card>`); console.log(`[${css}] padding`, await page.evaluate(() => getComputedStyle(c).padding + ' ' + getComputedStyle(c).boxSizing + ' inset ' + (i.getBoundingClientRect().left - c.getBoundingClientRect().left))); await page.close(); }
{ const page = await open('<mx-card id=c style="padding:40px"><div id=i style="block-size:10px"></div></mx-card>'); console.log('inline style padding: 40px ->', await page.evaluate(() => getComputedStyle(c).padding)); await page.evaluate(() => { c.style.setProperty('padding', '40px', 'important'); }); console.log('inline !important ->', await page.evaluate(() => getComputedStyle(c).padding)); await page.evaluate(() => { c.style.removeProperty('padding'); c.style.setProperty('--md-card-padding', '8px 24px'); }); console.log('--md-card-padding 8px 24px ->', await page.evaluate(() => getComputedStyle(c).padding)); await page.evaluate(() => c.style.setProperty('--md-card-padding', 'garbage')); console.log('an invalid --md-card-padding ->', await page.evaluate(() => getComputedStyle(c).padding)); await page.close(); }
for (const [n, css] of [['page display: grid, two columns', 'display:grid;grid-template-columns:1fr 1fr'], ['page display: flex', 'display:flex;gap:8px'], ['columns', 'columns:2'], ['page display: none', 'display:none']]) { const page = await open(`<mx-card id=c style="inline-size:332px;${css}"><p class=i style="margin:0">a</p><p class=i style="margin:0">b</p></mx-card>`); console.log(n, await page.evaluate(() => [...document.querySelectorAll('.i')].map((e) => { const r = e.getBoundingClientRect(), o = c.getBoundingClientRect(); return `${r.left - o.left},${r.top - o.top} ${r.width}`; }).join(' | '))); await page.close(); }
for (const [n, css] of [['margin-top', 'margin-top:30px'], ['float', 'float:left;block-size:60px']]) { const page = await open(`<mx-card id=c style="--md-card-padding:0;inline-size:200px"><div style="${css}">x</div></mx-card>`); console.log('flow-root,', n, await page.evaluate(() => c.getBoundingClientRect().height)); await page.close(); }
await browser.close();
