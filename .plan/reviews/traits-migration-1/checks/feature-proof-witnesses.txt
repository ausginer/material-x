## W7. Every public .js entry of core and material-x exports each value its .d.ts declares
import { readFileSync } from 'node:fs';
import { join } from 'node:path';
import { parseSync } from 'oxc-parser';
for (const pkg of ['packages/core', 'packages/material-x']) {
  const manifest = JSON.parse(readFileSync(join(pkg, 'package.json'), 'utf8'));
  for (const [key, value] of Object.entries<any>(manifest.exports)) {
    if (typeof value !== 'object' || !value.default?.endsWith('.js') || !value.types) continue;
    const names = (file: string, dts: boolean) => {
      const { program } = parseSync(file, readFileSync(join(pkg, file), 'utf8'));
      const out = new Set<string>();
      for (const n of program.body as any[]) {
        if (n.type === 'ExportNamedDeclaration') {
          if (n.exportKind === 'type' && !n.declaration) continue;
          const d = n.declaration;
          if (d) {
            if (d.type === 'TSTypeAliasDeclaration' || d.type === 'TSInterfaceDeclaration') continue;
            if (d.id?.name) out.add(d.id.name);
            for (const v of d.declarations ?? []) if (v.id?.name) out.add(v.id.name);
          }
          for (const s of n.specifiers ?? []) if (s.exportKind !== 'type') out.add(s.exported.name ?? s.exported.value);
        }
        if (n.type === 'ExportDefaultDeclaration') out.add('default');
      }
      return out;
    };
    const js = names(value.default, false);
    const ts = names(value.types, true);
    const missing = [...ts].filter((n) => !js.has(n));
    if (missing.length) console.log(`${pkg} ${key}: declared in .d.ts, absent from .js: ${missing.join(', ')}`);
  }
}
