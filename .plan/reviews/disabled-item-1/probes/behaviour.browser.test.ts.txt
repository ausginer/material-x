type Any = any;
const lines: string[] = [];
const str = (v: unknown) => JSON.stringify(v);
const is = (name: string, got: unknown, want: unknown) =>
  lines.push(
    str(got) === str(want)
      ? `ok   ${name} | ${str(got)}`
      : `DIFF ${name} | got ${str(got)} | want ${str(want)}`,
  );
const read = (name: string, got: unknown) =>
  lines.push(`read ${name} | ${str(got)}`);
const wait = async (ms: number) =>
  await new Promise((r) => setTimeout(r, ms));
const errors: string[] = [];
addEventListener('error', (e) => errors.push(e.message));

const NAMES = [
  'tabindex',
  'role',
  'aria-label',
  'aria-description',
  'aria-disabled',
];
const sem = (h: Any) =>
  `${NAMES.map((n) => String(h.getAttribute(n))).join(' ')} /${h.style.touchAction}`;
const ON = '0 button Reorder Use the arrow keys to move null /none';
const OFF = '-1 button Reorder null true /';
const PLAIN = 'null null null null null /';
const box = (e: Any) => {
  const b = (e.shadowRoot?.querySelector('.host') ?? e).getBoundingClientRect();
  return [b.x + b.width / 2, b.y + b.height / 2] as const;
};
const ctl = (e: Any) => e.shadowRoot.querySelector('.host');

function makeHandle(kind: string, slot = 'lead'): Any {
  const h = document.createElement(kind);
  h.dataset['handle'] = '';
  h.slot = slot;
  h.textContent = '=';
  return h;
}
async function create(
  tag: string,
  kind: string,
  { disabled = [0], commit = true, slot = 'lead' }: Any = {},
) {
  const list: Any = document.createElement('mx-sortable-list');
  list.style.setProperty('--_drag-duration', '0s');
  list.log = [];
  list.addEventListener('reorder', (event: Event) => {
    const { item, next, from, to } = event as ReorderEvent;
    list.log.push(`${item.id}:${from}>${to}`);
    list.onmove?.(item);
    if (commit) list.insertBefore(item, next);
  });
  const items = ['a', 'b', 'c'].map((name, i) => {
    const item: Any = document.createElement(tag);
    item.id = name;
    if (tag === 'mx-list-link-item') item.setAttribute('href', '#x');
    if (kind !== 'none') item.append(makeHandle(kind, slot));
    item.append(`Item ${name}`);
    if (disabled.includes(i)) item.toggleAttribute('disabled', true);
    return item;
  });
  const before: Any = document.createElement('button');
  before.id = 'before';
  list.append(...items);
  const after: Any = document.createElement('button');
  after.id = 'after';
  document.body.append(before, list, after);
  await wait(60);
  return {
    list,
    items,
    before,
    after,
    handle: items[0].querySelector('[data-handle]') as Any,
  };
}
const active = () => {
  const e: Any = document.activeElement;
  return e === document.body
    ? 'body'
    : e.dataset.handle === ''
      ? `h(${e.parentElement.id})`
      : e.id || e.localName;
};
const ids = (list: Any) =>
  [...list.children]
    .filter((c: Any) => !c.hasAttribute('data-drag-placeholder'))
    .map((c: Any) => c.id)
    .join('');
const rest = (list: Any) =>
  ![...list.children].some(
    (c: Any) =>
      c.hasAttribute('data-drag-placeholder') ||
      c.matches(':state(drag)') ||
      c.hasAttribute('popover'),
  );
let keys: string[] = [];
addEventListener('keydown', (e) => {
  keys.push(`${e.key}:${e.defaultPrevented}`);
});
async function press(key: string) {
  keys = [];
  await userEvent.keyboard(`{${key}}`);
  await wait(60);
  return keys[0];
}
async function drag(from: Any, dy: number, hook?: () => unknown) {
  from.scrollIntoView({ block: 'center' });
  const [x, y] = box(from);
  await commands.pointerPress(x, y);
  await commands.pointerMove(x, y + 12, 3);
  await hook?.();
  await commands.pointerMove(x, y + dy, 8);
  await wait(40);
  const during = document.querySelector('mx-sortable-list > :state(drag)')?.id;
  await commands.pointerRelease();
  await wait(120);
  return during ?? null;
}
async function tabs(from: Any, n: number, key = TAB) {
  from.focus();
  const out: string[] = [];
  for (let i = 0; i < n; i += 1) {
    await userEvent.keyboard(key);
    out.push(active());
  }
  return out;
}

  if (errors.length) lines.push(`DIFF errors ${str(errors)}`);
  document.body.replaceChildren();
  (task.meta as Any).lines = lines.splice(0);
});

it('P2 the page\'s values', async () => {
  const VALUES = {
    tabindex: '3',
    role: 'link',
    'aria-label': 'Own',
    'aria-description': 'Own',
    'aria-disabled': 'false',
  };
  // Each value alone, set before the claim, while enabled, while disabled, and in the same task as the change.
  for (const kind of ['span', 'button'])
    for (const [name, value] of [...Object.entries(VALUES), ['touch-action', 'pan-x'] as const]) {
      const put = (h: Any) =>
        name === 'touch-action' ? (h.style.touchAction = value) : h.setAttribute(name, value);
      const get = (h: Any) => (name === 'touch-action' ? h.style.touchAction : h.getAttribute(name));
      const at = `P2 ${kind} ${name}`;
      const run = async (when: string) => {
        const r = await create('mx-list-button-item', 'none', { disabled: [] });
        const [a] = r.items;
        const h = makeHandle(kind);
        if (when === 'before') put(h);
        a.append(h);
        await wait(40);
        if (when === 'enabled') put(h);
        await wait(40);
        if (when === 'task, before') put(h);
        a.disabled = true;
        if (when === 'task, after') put(h);
        await wait(40);
        if (when === 'disabled') put(h);
        await wait(40);
        const one = get(h);
        a.disabled = false;
        if (when === 'task, after enabling') put(h);
        await wait(40);
        const two = get(h);
        h.remove();
        await wait(40);
        const left = get(h);
        is(
          `${at} set ${when}`,
          when === 'task, after enabling' ? [two, left] : [one, two, left],
          when === 'task, after enabling' ? [value, value] : [value, value, value],
        );
        document.body.replaceChildren();
      };
      for (const when of ['before', 'enabled', 'task, before', 'task, after', 'disabled', 'task, after enabling'])
        await run(when);
    }
  // An attribute that the page removed.
  for (const name of NAMES.slice(0, 4)) {
    const r = await create('mx-list-button-item', 'span', { disabled: [] });
    r.handle.removeAttribute(name);
    await wait(40);
    const before = sem(r.handle);
    r.items[0].disabled = true;
    await wait(40);
    r.items[0].disabled = false;
    await wait(40);
    // The same removal and a slot change, as on main.
    const q = await create('mx-list-button-item', 'span', { disabled: [] });
    q.handle.removeAttribute(name);
    await wait(40);
    const extra = document.createElement('i');
    extra.slot = 'trail';
    q.items[0].append(extra);
    await wait(40);
    read(`P2 the page removed ${name}, then a slot change`, sem(q.handle));
    read(`P2 the page removed ${name}: before, after disabled and enabled`, [before, sem(r.handle)]);
    document.body.replaceChildren();
  }
  // A host's aria-description of an item without a handle.
  const r = await create('mx-list-button-item', 'none', { disabled: [] });
  const [a] = r.items;
  a.setAttribute('aria-description', 'Mine');
  await wait(40);
  a.disabled = true;
  await wait(40);
  const off = ctl(a).getAttribute('aria-description');
  a.disabled = false;
  await wait(40);
  is('P4 the host\'s description', [off, ctl(a).getAttribute('aria-description')], ['Mine', 'Mine']);
});

it('P2 life of the item', async () => {
  const r = await create('mx-list-button-item', 'span');
  const [a, b] = r.items;
  const h = r.handle;
  // Handles swapped.
  const h2 = makeHandle('span', 'trail');
  h.remove();
  await wait(40);
  is('no handle, disabled', [sem(h), ctl(a).getAttribute('aria-description')], [PLAIN, null]);
  a.append(h2);
  await wait(40);
  is('another handle, disabled', sem(h2), OFF);
  // Out of the list and disabled or enabled there.
  const plain = document.createElement('mx-list');
  document.body.append(plain);
  plain.append(a);
  await wait(60);
  is('in mx-list, disabled', sem(h2), PLAIN);
  a.disabled = false;
  await wait(60);
  is('in mx-list, enabled', [sem(h2), ctl(a).getAttribute('aria-description')], [PLAIN, null]);
  a.disabled = true;
  await wait(60);
  is('in mx-list, disabled again', sem(h2), PLAIN);
  r.list.append(a);
  await wait(60);
  is('back, disabled', sem(h2), OFF);
  a.remove();
  a.disabled = false;
  r.list.prepend(a);
  await wait(60);
  is('back, enabled while out', sem(h2), ON);
  // Disabled in the task of its removal.
  a.disabled = true;
  a.remove();
  await wait(60);
  is('disabled and removed in one task', sem(h2), PLAIN);
  r.list.prepend(a);
  await wait(60);
  is('and back', sem(h2), OFF);
  a.disabled = false;
  a.disabled = true;
  a.disabled = false;
  await wait(40);
  is('toggled in one task', sem(h2), ON);
  a.setAttribute('disabled', '');
  a.setAttribute('disabled', 'false');
  await wait(40);
  is('another value of the attribute', sem(h2), OFF);
  // Into a second sortable list, with focus kept by nothing: an enabled neighbour is untouched.
  is('neighbour', sem(b.querySelector('[data-handle]')), ON);
  // From HTML.
  document.body.insertAdjacentHTML(
    'beforeend',
    `<mx-sortable-list id="html"><mx-list-link-item disabled href="#x" id="l"><span data-handle slot="lead">=</span>L</mx-list-link-item><mx-list-button-item disabled id="m"><span data-handle slot="trail">=</span>M</mx-list-button-item><mx-list-button-item disabled id="n">N</mx-list-button-item></mx-sortable-list>`,
  );
  await wait(60);
  is(
    'from HTML',
    ['l', 'm'].map((i) => sem(document.querySelector(`#${i} [data-handle]`))),
    [OFF, OFF],
  );
  is('from HTML, without a handle', ctl(document.querySelector('#n')).getAttribute('aria-description'), null);
});

    is(`P8 ${kind} Tab`, (await tabs(r.before, 1))[0], r.handle ? 'h(a)' : 'a');
    target.focus();
    await press('ArrowDown');
    await wait(80);

    document.body.replaceChildren();
  }
  const r = await create('mx-list-item', 'none', { disabled: [] });
  r.items[0].removeAttribute('tabindex');
  await wait(40);
  const before = r.items[0].hasAttribute('tabindex');
  r.items[0].setAttribute('disabled', '');
  await wait(40);
  is('P8 the page removed tabindex, then set the attribute', [before, r.items[0].hasAttribute('tabindex')], [false, false]);
});

it('nested', async () => {
  // A sortable list in an item, without a handle, of a sortable list.
  for (const disabled of [false, true])
    for (const kind of ['none', 'span']) {
      const outer = await create('mx-list-item', 'none', { disabled: [] });
      const inner = await create('mx-list-button-item', kind, { disabled: disabled ? [1] : [] });
      inner.items.forEach((e: Any, i: number) => (e.id = `i${i}`));
      const holder = document.createElement('div');
      holder.slot = 'support';
      holder.append(inner.list);
      outer.items[1].append(holder);
      await wait(80);
      const target = inner.items[1].querySelector('[data-handle]') ?? inner.items[1];
      let name = 'nothing focusable';
      if (target !== inner.items[1] || !disabled) {
        target.focus();
        name = (await press('ArrowDown')) ?? 'no key';
      }
      const logs = [[...outer.list.log], [...inner.list.log]];
      const during = await drag(target, -70);
      read(
        `nested, inner ${kind}${disabled ? ' disabled' : ''}: key ${name}, outer and inner moves; then a drag`,
        [...logs, during, outer.list.log, inner.list.log],
      );
      document.body.replaceChildren();
    }
});

  it('touch', async () => {
    for (const kind of ['span', 'button']) for (const when of ['enabled', 'disabled', 'disabled after the touch', 'disabled during the drag']) {
      const r = await create('mx-list-button-item', kind, { disabled: when === 'disabled' ? [0] : [] });
      const tall = document.createElement('div');
      tall.style.height = '4000px';
      document.body.append(tall);
      scrollTo(0, 0);
      const log: string[] = [];
      r.list.addEventListener('pointercancel', () => log.push('pointercancel'), true);
      const off = () => (r.items[0].disabled = true);
      const [x, y] = box(r.handle);
      const touch = (commands as Any).touch;
      await touch('touchStart', [x, y]);
      if (when === 'disabled after the touch') off();
      await wait(40);
      for (let i = 1; i <= 3; i += 1) await touch('touchMove', [x, y + i * 8]);
      await wait(40);
      const lifted = r.items[0].matches(':state(drag)');
      if (when === 'disabled during the drag') off();
      await wait(40);
      for (let i = 1; i <= 6; i += 1) await touch('touchMove', [x, y + 24 + i * 12]);
      await wait(40);
      await touch('touchEnd');
      await wait(200);
      is(
        `touch ${kind} (${getComputedStyle(r.handle).display}) ${when}: lifted, moves, rest, canceled`,
        [lifted, r.list.log.length, rest(r.list), log.length > 0],
        when === 'disabled' ? [false, 0, true, true] : [true, 1, true, false],
      );
      scrollTo(0, 0);
      document.body.replaceChildren();
      await wait(60);
    }
  });
