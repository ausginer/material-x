# aube-manifest-edit-probe.sh
#!/usr/bin/env bash
# A two-package hoisted workspace with no registry dependency. It reads what aube
# does after a hand edit of a workspace manifest: a peer-only edit, then a
# dependency edit. Usage: aube-manifest-edit-probe.sh <empty-directory>
set -uo pipefail
P="${1:?}"
mkdir -p "${P:?}/packages/a" "${P:?}/packages/b"
cd "${P:?}" || exit 1
git init -q .
printf '%s\n' 'node-linker=hoisted' 'defaultLockfile=package-lock.json' 'savePrefix=' 'aubeNoAutoInstall=true' > .npmrc
echo '{ "name": "probe-root", "private": true, "workspaces": ["packages/*"] }' > package.json
echo '{ "name": "@p/a", "version": "0.1.0" }' > packages/a/package.json
echo '{ "name": "@p/b", "version": "0.1.0" }' > packages/b/package.json

run() {
  echo "\$ $*"
  "$@" > out.log 2>&1
  local e=$?
  grep -vE '^\s*$' out.log | tail -n 3
  echo "exit=$e"
}
links() { echo "\$ ls node_modules/@p"; ls node_modules/@p 2>&1; }
lockdiff() { echo "\$ git diff --stat package-lock.json"; git diff --stat package-lock.json; }

aube --version | head -n 1
run aube install
run aube install --frozen-lockfile --force
links
git add -A && git commit -qm base

echo
echo '## A. A peer-only edit: @p/a gains react ^19 as an optional peer'
echo '{ "name": "@p/a", "version": "0.1.0", "peerDependencies": { "react": "^19" }, "peerDependenciesMeta": { "react": { "optional": true } } }' > packages/a/package.json
run aube install --frozen-lockfile
run aube ci
links
run aube install
lockdiff
links
run aube install --fix-lockfile
lockdiff
links
run aube install --frozen-lockfile --force
links

echo
echo '## B. A dependency edit: @p/a depends on @p/b'
git checkout -q -- . && aube install --frozen-lockfile --force > /dev/null 2>&1
links
echo '{ "name": "@p/a", "version": "0.1.0", "dependencies": { "@p/b": "0.1.0" } }' > packages/a/package.json
run aube install --frozen-lockfile
run aube ci
run aube install
lockdiff
links
run aube install --frozen-lockfile --force
links

echo '## C. The flag of aube add that saves a peer'
