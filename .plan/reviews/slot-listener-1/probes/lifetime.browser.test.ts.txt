const oldUseSlot = (host, sel, cb) => {
  const slot = host.shadowRoot.querySelector(sel);
  useEvents(host, { slotchange() { cb(slot, slot.assignedNodes()); } }, slot);
};

function define(extra, opts = {}) {
  const tag = `x-proof-${(n += 1)}`;
  const body = class extends ControlledElement {
    constructor() {
      super();
      if (!this.shadowRoot) {
        this.attachShadow({ mode: 'open' }).innerHTML = opts.html ?? '<slot></slot>';
      }
      useSlot(this, 'slot', (s, nodes) => log.new.push(this.text(nodes)));
      oldUseSlot(this, 'slot', (s, nodes) => log.old.push(this.text(nodes)));
      extra?.(this);
    }
    text(nodes) {
      return `${this.isConnected ? '' : 'out:'}${nodes.map((x) => x.textContent || x.localName).join('')}`;
    }
  };
  if (opts.late) return [tag, () => customElements.define(tag, body)];
  customElements.define(tag, body);
  return [tag];
}

describe('proof', () => {
  it('log', async () => {
    const out = [`# ${server.browser}`];
    for (const [k, f] of Object.entries(C)) {
      document.body.replaceChildren();
      await task();
      log = { new: [], old: [] };
      document.body.replaceChildren();
