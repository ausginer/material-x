// Feature proof of RF-143. Prints one line a reading: the member that Tab and Shift+Tab reach, the
// member of each key, the tab stops, and the members that have the outer marks.
import { afterAll, describe, it } from 'vitest';
import { server, userEvent } from 'vitest/browser';
import '../../src/button-group/connected-button-group.ts';
import '../../src/button/switch-button.ts';
import '../../src/button/switch-icon-button.ts';
import { nextFrame } from '../browser.ts';

const OUT: string[] = [];
afterAll(() => { throw new Error(`\nPROOF\n${OUT.join('\n')}\nEND`); });
const E = server.browser;
const V = ['a', 'b', 'c', 'd'];
let style: HTMLStyleElement;
function make(n = 4, tag = 'mx-switch-button'): [HTMLElement, HTMLElement[]] {
  const g = document.createElement('mx-connected-button-group');
  const m = V.slice(0, n).map((v) => { const e = document.createElement(tag); e.setAttribute('value', v); e.id = v; if (tag === 'mx-switch-button') e.textContent = v; return e; });
  g.append(...m); return [g, m];
}
function page(css = ''): void { document.body.replaceChildren(); style = document.createElement('style'); style.textContent = css; document.body.append(style); }
const frames = async (): Promise<void> => { await nextFrame(); await nextFrame(); };
const act = (): string => (document.activeElement as HTMLElement | null)?.id || document.activeElement?.localName || '-';
const stops = (g: HTMLElement): string => [...g.children].filter((e) => e.getAttribute('tabindex') === '0').map((e) => e.id).join('') || '-';
const marks = (g: HTMLElement): string => `${[...g.children].filter((e) => (e as HTMLElement).dataset['first'] !== undefined).map((e) => e.id).join('') || '-'}${[...g.children].filter((e) => (e as HTMLElement).dataset['last'] !== undefined).map((e) => e.id).join('') || '-'}`;
async function tab(g: HTMLElement, back = false): Promise<string> {
  const s = document.createElement('button'); s.id = 'side'; if (back) g.after(s); else g.before(s);
  s.focus(); await userEvent.keyboard(back ? '{Shift>}{Tab}{/Shift}' : '{Tab}'); const r = act(); s.remove(); return r;
}
async function say(tag: string, g: HTMLElement): Promise<void> {
  const st = stops(g); const mk = marks(g); const a = await tab(g); const b = await tab(g, true);
  OUT.push(`${E} ${tag} | stop ${st} | tab ${a} back ${b} | ends ${mk}`);
}
const KEYS = ['{ArrowRight}', '{ArrowDown}', '{ArrowLeft}', '{ArrowUp}', '{Home}', '{End}'];
async function keys(tag: string, g: HTMLElement, from: HTMLElement): Promise<void> {
  const r: string[] = [];
  for (const k of KEYS) { from.focus(); await userEvent.keyboard(k); r.push(act()); }
  OUT.push(`${E} ${tag} keys from ${from.id}: ${r.join(' ')}`); (document.activeElement as HTMLElement).blur();
}
const set: Record<string, [(e: HTMLElement) => void, (e: HTMLElement) => void]> = {
  hidden: [(e) => { e.hidden = true; }, (e) => { e.hidden = false; }],
  disabled: [(e) => { e.toggleAttribute('disabled', true); }, (e) => { e.toggleAttribute('disabled', false); }],
  'until-found': [(e) => { e.setAttribute('hidden', 'until-found'); }, (e) => { e.removeAttribute('hidden'); }],
  'display:none': [(e) => { e.style.display = 'none'; }, (e) => { e.style.display = ''; }],
};
for (const tag of ['mx-switch-button', 'mx-switch-icon-button']) for (const [w, [on, off]] of Object.entries(set)) describe(`${tag} ${w}`, () => {
  const t = (s: string): string => `${tag.slice(3)} ${w} ${s}`;
  it('markup', async () => { page(); const [g, m] = make(4, tag); on(m[0]!); on(m[3]!); document.body.append(g); await frames(); await say(t('a d in the markup'), g); await keys(t('a d in the markup'), g, m[1]!); await keys(t('a d in the markup'), g, m[2]!); });
  it('later', async () => { page(); const [g, m] = make(4, tag); document.body.append(g); await frames(); on(m[0]!); await frames(); await say(t('a later'), g); off(m[0]!); await frames(); await say(t('a later, off again'), g); });
  it('middle', async () => { page(); const [g, m] = make(4, tag); document.body.append(g); await frames(); on(m[1]!); on(m[2]!); await frames(); await keys(t('b c later'), g, m[0]!); await say(t('b c later'), g); });
  it('value', async () => { page(); const [g, m] = make(4, tag); g.setAttribute('value', 'c'); document.body.append(g); await frames(); on(m[2]!); await frames(); await say(t('the value c later'), g); off(m[2]!); await frames(); await say(t('the value c, off again'), g); });
  it('focus chose', async () => { page(); const [g, m] = make(4, tag); document.body.append(g); await frames(); m[2]!.focus(); m[2]!.blur(); on(m[2]!); await frames(); await say(t('c chosen by focus, later'), g); });
  it('focused', async () => { page(); const [g, m] = make(4, tag); document.body.append(g); await frames(); m[1]!.focus(); on(m[1]!); await frames(); OUT.push(`${E} ${t('the focused b')} | active ${act()} stop ${stops(g)}`); await userEvent.keyboard('{ArrowRight}'); OUT.push(`${E} ${t('the focused b')} then ArrowRight ${act()}`); await userEvent.keyboard('{Tab}'); OUT.push(`${E} ${t('the focused b')} then Tab ${act()}`); });
  it('all', async () => { page(); const [g, m] = make(2, tag); document.body.append(g); await frames(); m.forEach(on); await frames(); await say(t('all'), g); off(m[1]!); await frames(); await say(t('all but b'), g); });
  it('out', async () => { page(); const [g, m] = make(4, tag); on(m[0]!); on(m[3]!); await frames(); document.body.append(g); await frames(); await say(t('built out'), g); });
  it('changed out', async () => { page(); const [g, m] = make(4, tag); document.body.append(g); await frames(); g.remove(); on(m[0]!); on(m[3]!); await frames(); document.body.append(g); await frames(); await say(t('changed out'), g); g.remove(); off(m[0]!); off(m[3]!); document.body.append(g); await frames(); await say(t('changed back in one task'), g); });
  it('added', async () => { page(); const [g] = make(3, tag); document.body.append(g); await frames(); const n = document.createElement(tag); n.id = 'n'; n.setAttribute('value', 'n'); on(n); g.prepend(n); await frames(); await say(t('n added first'), g); off(n); await frames(); await say(t('n added first, off'), g); });
  it('removed', async () => { page(); const [g, m] = make(4, tag); document.body.append(g); await frames(); m[0]!.remove(); on(m[1]!); await frames(); await say(t('a removed and b in one task'), g); });
  it('one task', async () => { page(); const [g, m] = make(4, tag); document.body.append(g); await frames(); on(m[0]!); await Promise.resolve(); await Promise.resolve(); OUT.push(`${E} ${t('a, two microtasks later')} | stop ${stops(g)} ends ${marks(g)}`); });
  it('rtl', async () => { page(); const [g, m] = make(4, tag); g.dir = 'rtl'; document.body.append(g); await frames(); on(m[1]!); await frames(); await keys(t('rtl, b later'), g, m[0]!); });
});
describe('page', () => {
  for (const [n, css] of [['own display', '#a[hidden],#d[hidden]{display:contents}'], ['own display block', '#a[hidden]{display:block}'], ['a class', '.gone{display:none}'], ['visibility', '.gone{visibility:hidden}']] as const) it(n, async () => {
    page(css); const [g, m] = make(); document.body.append(g); await frames();
    if (n.startsWith('own')) { m[0]!.hidden = true; m[3]!.hidden = true; } else { m[0]!.classList.add('gone'); m[2]!.classList.add('gone'); }
    await frames(); await say(`page ${n}`, g); await keys(`page ${n}`, g, m[1]!);
  });
  it('an attribute of the state changes the display too', async () => { page('mx-connected-button-group[data-narrow] > #a{display:none}'); const [g] = make(); document.body.append(g); await frames(); g.toggleAttribute('data-narrow', true); await frames(); await say('page narrow', g); g.toggleAttribute('data-narrow', false); await frames(); await say('page wide', g); });
  it('the group hidden', async () => { page(); const [g, m] = make(); document.body.append(g); await frames(); g.hidden = true; m[0]!.hidden = true; await frames(); g.hidden = false; await frames(); await say('group hidden while a is', g); });
  it('an ancestor display none', async () => { page(); const d = document.createElement('div'); d.style.display = 'none'; const [g, m] = make(); m[0]!.hidden = true; d.append(g); document.body.append(d); await frames(); d.style.display = ''; await frames(); await say('ancestor display none at first', g); });
  it('moved', async () => { page(); const [g, m] = make(); document.body.append(g); await frames(); m[2]!.focus(); m[2]!.blur(); const d = document.createElement('div'); document.body.append(d); d.append(g); await frames(); await say('moved, c chosen by focus', g); });
  it('moved with a value', async () => { page(); const [g, m] = make(); g.setAttribute('value', 'a'); document.body.append(g); await frames(); m[2]!.focus(); m[2]!.blur(); m[1]!.hidden = true; await frames(); await say('value a, focus moved to c, b hidden', g); });
  it('template', async () => { page(); const t = document.createElement('template'); t.innerHTML = '<mx-connected-button-group><mx-switch-button id="a" value="a" hidden>a</mx-switch-button><mx-switch-button id="b" value="b">b</mx-switch-button><mx-switch-button id="c" value="c" hidden>c</mx-switch-button></mx-connected-button-group>'; const g = t.content.cloneNode(true).firstChild as HTMLElement; document.body.append(g); await frames(); await say('template clone', g); });
  it('parsed', async () => { page(); const d = document.createElement('div'); document.body.append(d); d.innerHTML = '<mx-connected-button-group><mx-switch-button id="a" value="a" hidden>a</mx-switch-button><mx-switch-button id="b" value="b">b</mx-switch-button><mx-switch-button id="c" value="c" hidden>c</mx-switch-button></mx-connected-button-group>'; await frames(); await say('innerHTML', d.firstChild as HTMLElement); });
  it('shadow', async () => { page(); const h = document.createElement('div'); const r = h.attachShadow({ mode: 'open' }); r.innerHTML = '<mx-connected-button-group><slot></slot></mx-connected-button-group>'; for (const v of 'abc') { const e = document.createElement('mx-switch-button'); e.id = v; e.setAttribute('value', v); e.textContent = v; h.append(e); } document.body.append(h); await frames(); (h.children[0] as HTMLElement).hidden = true; await frames(); OUT.push(`${E} members through a nested slot | stop ${stops(h)} ends ${marks(h)}`); });
  it('one', async () => { page(); const [g, m] = make(1); document.body.append(g); await frames(); await new Promise((r) => { setTimeout(r, 600); }); const cs = getComputedStyle(m[0]!.shadowRoot!.querySelector('.host')!); OUT.push(`${E} one member: ${[cs.borderTopLeftRadius, cs.borderTopRightRadius, cs.borderBottomRightRadius, cs.borderBottomLeftRadius]}`); });
  it('cost', async () => {
    page(); const r: number[] = [];
    for (let k = 0; k < 7; k += 1) { const d = document.createElement('div'); for (let i = 0; i < 200; i += 1) d.append(make(4)[0]); const t0 = performance.now(); document.body.append(d); await Promise.resolve(); document.body.offsetHeight; r.push(performance.now() - t0); await frames(); d.remove(); await frames(); }
    OUT.push(`${E} 200 groups of 4: median ${r.sort((x, y) => x - y)[3]!.toFixed(0)} ms`);
  });
});
