const rows: string[] = [];

const task = () => new Promise((r) => setTimeout(r));
const ctl = (e: Element) => e.shadowRoot!.querySelector('.input, .host')!;
type Own = (e: Element, seen: Set<string>) => string | null;
const on = (a: string, v = 'true'): Own => (e) => (e.hasAttribute(a) ? v : null);
const bool = (a: string): Own => (e) => String(e.hasAttribute(a));
const NAMES: Record<string, Record<string, { own: Own; src: string; free?: boolean }>> = {
  'mx-text-field': { 'aria-invalid': { own: on('error'), src: 'error' }, 'aria-required': { own: on('required'), src: 'required' } },
  'mx-multiline-text-field': { 'aria-invalid': { own: on('error'), src: 'error' }, 'aria-required': { own: on('required'), src: 'required' } },
  'mx-switch-button': { 'aria-checked': { own: bool('checked'), src: 'checked' }, 'aria-invalid': { own: on('error'), src: 'error' }, 'aria-required': { own: on('required'), src: 'required' } },
  'mx-switch-icon-button': { 'aria-checked': { own: bool('checked'), src: 'checked' }, 'aria-invalid': { own: on('error'), src: 'error' }, 'aria-required': { own: on('required'), src: 'required' } },
  'mx-link-button': { 'aria-disabled': { own: on('disabled'), src: 'disabled' } },
  'mx-list-link-item': { 'aria-disabled': { own: on('disabled'), src: 'disabled' }, 'aria-current': { own: on('selected', 'page'), src: 'selected' } },
  'mx-list-button-item': { 'aria-selected': { own: (e, seen) => (seen.has('selected') ? String(e.hasAttribute('selected')) : null), src: 'selected' } },
};
const PLAIN = ['mx-button', 'mx-fab', 'mx-checkbox', 'mx-radio'];
it('random', async () => {
  for (const [tag, map] of Object.entries(NAMES)) {
    const names = [...Object.keys(map), 'aria-description'];
    const srcs = [...new Set(Object.values(map).map((m) => m.src))];
    let reads = 0, wrong = 0, hostWrong = 0, first = '';
    const vals = ['', 'true', 'false', 'mixed', null];
    for (let seed = 1; seed <= 120; seed++) {
      let s = seed * 2654435761;
      const R = (n: number) => { s = (Math.imul(s ^ (s >>> 15), 2246822519) + 0x9e3779b9) | 0; return ((s >>> 8) % n + n) % n; };
      const H = new Map<string, string | null>(); const seen = new Set<string>();
      // Three ways to exist: created, parsed with attributes, cloned from one with attributes.
      let el: HTMLElement;
      if (seed % 3 === 0) el = document.createElement(tag);
      else {
        const n = names[R(names.length)]!, src = srcs[R(srcs.length)]!;
        const box = document.createElement('div');
        box.innerHTML = `<${tag} ${R(2) ? `${n}="mixed" ${src}` : `${src} ${n}="mixed"`}></${tag}>`;
        H.set(n, 'mixed'); seen.add(src);
        el = box.firstChild as HTMLElement;
        if (seed % 3 === 1) { customElements.upgrade(el); el = el.cloneNode(true) as HTMLElement; }
        else if (R(2)) customElements.upgrade(el); else document.body.append(box);
        if (!el.shadowRoot) { customElements.upgrade(el); }
        // the parsed value reaches an unconnected control at the first connect: start there.
        document.body.append(el);
      }
      const log: string[] = [];
      const check = () => { const c = ctl(el);
        for (const n of names) { reads++; if (el.getAttribute(n) !== (H.get(n) ?? null)) hostWrong++;
          const want = H.get(n) ?? map[n]?.own(el, seen) ?? null;
          if (c.getAttribute(n) !== want) { wrong++; first ||= `seed ${seed} ${log.join(' ')} => ${n} ${c.getAttribute(n)} want ${want}`; } } };
      for (let k = 0; k < 20; k++) {
        const op = R(9), n = names[R(names.length)]!, v = vals[R(5)]!, src = srcs[R(srcs.length)]!;
        log.push(op < 3 ? `${n}=${v}` : op < 6 ? `~${src}` : op === 6 ? 'connect' : op === 7 ? 'out' : 'task');
        if (op < 3) { if (v == null) el.removeAttribute(n); else el.setAttribute(n, v); H.set(n, v); }
        else if (op < 6) { el.toggleAttribute(src); seen.add(src); }
        else if (op === 6) { document.body.append(el); check(); }
        else if (op === 7) el.remove();
        else { await task(); check(); }
      }
      await task(); check(); document.body.replaceChildren();
    }
    say(`random ${tag}: reads, wrong, host attributes wrong, first`, [reads, wrong, hostWrong, first]);
  }
});
