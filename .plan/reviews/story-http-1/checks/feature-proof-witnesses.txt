// Counts the ids WithLabel and the playground use, in the story's canvas and
// the checkbox docs page, served from a non-secure origin as docs-smoke does.
import { createServer } from 'node:http';
import { createRequire } from 'node:module';
import { readFile, stat } from 'node:fs/promises';
import { extname, join } from 'node:path';
const tree = process.argv[2];
const { chromium } = createRequire(join(tree, 'package.json'))('playwright');
const docs = join(tree, '.docs');
const base = '/material-x/';
const types = { '.html': 'text/html', '.js': 'text/javascript', '.css': 'text/css', '.json': 'application/json', '.svg': 'image/svg+xml', '.woff2': 'font/woff2' };
const server = createServer(async (req, res) => {
  const p = new URL(req.url, 'http://x').pathname;
  if (!p.startsWith(base)) return res.writeHead(404).end();
  let f = join(docs, decodeURIComponent(p.slice(base.length)));
  try { if ((await stat(f)).isDirectory()) f = join(f, 'index.html'); res.writeHead(200, { 'content-type': types[extname(f)] ?? 'application/octet-stream' }).end(await readFile(f)); }
  catch { res.writeHead(404).end(); }
});
await new Promise((r) => server.listen(0, '127.0.0.1', r));
const origin = `http://mx-docs.test:${server.address().port}`;
const browser = await chromium.launch({ args: ['--host-resolver-rules=MAP mx-docs.test 127.0.0.1'] });
for (const [id, mode] of [['checkbox--with-label', 'story'], ['checkbox--docs', 'docs']]) {
  const page = await browser.newPage();
  const errors = [];
  page.on('pageerror', (e) => errors.push(String(e)));
  page.on('console', (m) => m.type() === 'error' && errors.push(m.text()));
  await page.route((u) => u.origin !== origin, (r) => r.fulfill({ status: 200, body: '' }));
  await page.goto(`${origin}${base}iframe.html?id=${id}&viewMode=${mode}`, { waitUntil: 'load' });
  await page.waitForFunction(() => document.querySelectorAll('mx-checkbox').length > 0, null, { timeout: 30000 });
  await page.waitForTimeout(1500);
  const r = await page.evaluate(() => ({
    secure: isSecureContext,
    randomUUID: typeof crypto.randomUUID,
    withLabel: document.querySelectorAll('[id="mx-with-label-checkbox"]').length,
    playground: document.querySelectorAll('[id="mx-playground-checkbox"]').length,
    checkboxes: document.querySelectorAll('mx-checkbox').length,
    labelFor: [...document.querySelectorAll('label[for]')].map((l) => `${l.htmlFor}->${l.control?.localName ?? 'none'}`),
    duplicateIds: (() => { const c = {}; for (const e of document.querySelectorAll('[id]')) c[e.id] = (c[e.id] ?? 0) + 1; return Object.entries(c).filter(([, n]) => n > 1); })(),
  }));
  console.log(id, JSON.stringify(r), 'errors:', errors.length);
  await page.close();
}
await browser.close();
server.close();
