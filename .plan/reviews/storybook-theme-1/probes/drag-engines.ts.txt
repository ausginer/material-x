// Feature proof of #323 (repo:RF-155): every drag story, light and dark, in one
// engine. For every element under #storybook-root it reads color,
// background-color, border-top-color (where a border is drawn) and box-shadow,
// and lists each element whose value is the same in both themes, ignoring
// transparent and none. For the sortable stories it also starts a drag and
// reads the placeholder. Run against a running `storybook dev`, from a tree
// with Playwright installed: `node drag-engines.ts <engine> [base]`.
import * as pw from '/workspaces/worktrees/c/node_modules/playwright/index.mjs';
import type { Page } from 'playwright';

const engine = process.argv[2] as 'chromium' | 'firefox' | 'webkit';
const base = process.argv[3] ?? 'http://127.0.0.1:6123';
const stories = [
  'drag-custom-behavior--resize-handle',
  'drag-free-drag--interactive',
  'drag-free-drag--async-drop-confirmation',
  'drag-free-drag--transformed-stage',
  'drag-sortable--list',
  'drag-sortable--grid',
  'drag-sortable--custom-placeholder',
  'drag-sortable--zoomed-context',
];
const props = ['color', 'background-color', 'border-top-color', 'box-shadow'];

type Reading = Record<string, Record<string, string>>;

async function readAll(page: Page): Promise<Reading> {
  return page.evaluate((props: string[]) => {
    const out: Record<string, Record<string, string>> = {};
    [...document.querySelectorAll('#storybook-root *')].forEach((el, i) => {
      const cs = getComputedStyle(el);
      const key = `${i}:${el.tagName.toLowerCase()}.${String(el.className).split('_')[1] ?? el.className}`;
      out[key] = Object.fromEntries(props.map((p) => [p, cs.getPropertyValue(p)]));
      out[key]!['border-style'] = cs.borderTopStyle;
    });
    return out;
  }, props);
}

async function placeholder(page: Page): Promise<string> {
  const row = page.locator('#storybook-root [class*="row"], #storybook-root [class*="tile"]').first();
  const box = await row.boundingBox();
  if (!box) return 'no row';
  await page.mouse.move(box.x + box.width / 2, box.y + box.height / 2);
  await page.mouse.down();
  for (let i = 1; i <= 8; i++) {
    await page.mouse.move(box.x + box.width / 2 + i * 2, box.y + box.height / 2 + i * 8);
    await page.waitForTimeout(30);
  }
  await page.waitForTimeout(300);
  const r = await page.evaluate(() => {
    const el = document.querySelector('[data-drag-placeholder]');
    if (!el) return 'no placeholder';
    const cs = getComputedStyle(el);
    return `placeholder bg=${cs.backgroundColor} border=${cs.borderTopColor} ${cs.borderTopStyle} color=${cs.color}`;
  });
  await page.mouse.up();
  return r;
}

const browser = await pw[engine].launch();
try {
  const page = await browser.newPage();
  for (const id of stories) {
    const r: Record<string, Reading> = {};
    const ph: Record<string, string> = {};
    for (const theme of ['light', 'dark']) {
      await page.goto(`${base}/iframe.html?id=${id}&viewMode=story&globals=theme:${theme}`);
      await page.waitForTimeout(2500);
      const html = await page.evaluate(() => document.documentElement.dataset['theme']);
      if (html !== theme) console.log(`${engine} ${id} ${theme}: html data-theme=${html}`);
      r[theme] = await readAll(page);
      if (id.startsWith('drag-sortable--')) ph[theme] = await placeholder(page);
    }
    const same: string[] = [];
    for (const [k, v] of Object.entries(r['light']!)) {
      const d = r['dark']![k];
      if (!d) continue;
      for (const p of props) {
        const a = v[p]!;
        if (a !== d[p] || /^(rgba\(0, 0, 0, 0\)|none|transparent)$/.test(a)) continue;
        if (p === 'border-top-color' && v['border-style'] === 'none') continue;
        same.push(`${k} ${p}=${a}`);
      }
    }
    const n = Object.keys(r['light']!).length;
    console.log(`${engine.padEnd(8)} ${id.padEnd(40)} elements=${n} unchanged=${same.length ? same.join('; ') : 'none'}`);
    if (ph['light']) console.log(`${engine.padEnd(8)}   light ${ph['light']}\n${engine.padEnd(8)}   dark  ${ph['dark']}`);
  }
} finally {
  await browser.close();
}
