# storybook-imports.ts
import { readdir, readFile } from 'node:fs/promises';
import { builtinModules } from 'node:module';

const dir = '.storybook';
const root = JSON.parse(await readFile('package.json', 'utf8'));
const declared = new Set([...Object.keys(root.dependencies ?? {}), ...Object.keys(root.devDependencies ?? {})]);
const files = (await readdir(dir, { recursive: true })).filter((f) => /\.(ts|tsx)$/u.test(f)).sort();
const specifier = /(?:\bfrom\s*|\bimport\s*\(\s*|\bimport\s+)['"]([^'"]+)['"]/gu;
const names = new Map<string, string[]>();
for (const file of files) {
  const text = await readFile(`${dir}/${file}`, 'utf8');
  for (const [, spec] of text.matchAll(specifier)) {
    if (spec.startsWith('.') || spec.startsWith('/')) {
      continue;
    }
    const name = spec.startsWith('@') ? spec.split('/').slice(0, 2).join('/') : spec.split('/')[0];
    names.set(name, [...(names.get(name) ?? []), `${file}: ${spec}`]);
  }
}
console.log(`# Bare specifiers imported by every .ts and .tsx file under ${dir}/, reduced to package names`);
for (const [name, uses] of [...names].sort()) {
  console.log(`${name}\t${uses.join(', ')}`);
}
console.log('\n# Names that are neither a node: builtin nor a key of the root dependencies or devDependencies');
for (const name of [...names.keys()].sort()) {
  if (name.startsWith('node:') || builtinModules.includes(name) || declared.has(name)) {
    continue;
  }
  console.log(name);
}
console.log('# end');
