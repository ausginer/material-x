// property on its host, and the computed style of every element and
// pseudo-element of S is compared with and without it: a difference is a leak.
// The card is read twice, with its slot's resets and without them.

type Container = Readonly<{
  name: string;
  html: string;
  select: string;
  slot?: string;
  prepare?: (element: HTMLElement) => void;
  only?: RegExp;
}>;

const noResets = (card: HTMLElement) => {
  for (const sheet of card.shadowRoot!.adoptedStyleSheets) {
    const copy = new CSSStyleSheet();
    copy.replaceSync([...sheet.cssRules].map((r) => r.cssText).join('\n'));
    for (const rule of copy.cssRules) {
      if (rule instanceof CSSStyleRule && rule.selectorText === 'slot') {
        rule.style.removeProperty('--_level');
        rule.style.removeProperty('--_container-color');
      }
    }
    card.shadowRoot!.adoptedStyleSheets = card.shadowRoot!.adoptedStyleSheets.map(
      (s) => (s === sheet ? copy : s),
    );
  }
};

const READ = /var\(\s*(--_[\w-]+)/gu;

function sheetsOf(element: Element): readonly CSSStyleSheet[] {
  return element.shadowRoot?.adoptedStyleSheets ?? [];
}

const exposedCache = new Map<string, readonly string[]>();
function exposed(element: Element): readonly string[] {
  const cached = exposedCache.get(element.localName);
  if (cached) {
    return cached;
  }
  const read = new Set<string>();
  const declared = new Set<string>();
  const visit = (rules: CSSRuleList) => {
    for (const rule of rules) {
      for (const [, name] of rule.cssText.matchAll(READ)) {
        read.add(name!);
      }
      if (rule instanceof CSSStyleRule && rule.selectorText === ':host') {
        for (const name of rule.style) {
          declared.add(name);
        }
      }
    }
  };
  for (const sheet of sheetsOf(element)) {
    visit(sheet.cssRules);
  }
  const result = [...read].filter((name) => !declared.has(name)).sort();
  exposedCache.set(element.localName, result);
  return result;
}

/** The element and every custom element in its shadow or light tree. */
function family(root: Element): readonly Element[] {
  const out: Element[] = [];
  const visit = (element: Element) => {
    if (element.shadowRoot) {
      out.push(element);
      for (const child of element.shadowRoot.querySelectorAll('*')) {
        visit(child);
      }
    }
    for (const child of element.children) {
      visit(child);
    }
  };
  visit(root);
  return [...new Set(out)];
}

function snapshot(root: Element): Map<string, string> {
  const out = new Map<string, string>();
  for (const [index, host] of family(root).entries()) {
    const elements = [host, ...host.shadowRoot!.querySelectorAll('*')];
    for (const [i, element] of elements.entries()) {
      for (const pseudo of ['', '::before', '::after']) {
        const style = getComputedStyle(element, pseudo || undefined);
        if (pseudo && style.content === 'none') {
          continue;
        }
        for (const property of style) {
          if (!property.startsWith('--')) {
            out.set(
              `${host.localName}[${index}] ${i === 0 ? ':host' : `${element.localName}.${element.className}`}#${i}${pseudo} ${property}`,
              style.getPropertyValue(property),
            );
          }
        }
      }
    }
  }
  return out;
}

const frame = async () => {
  await new Promise((resolve) => {
    requestAnimationFrame(() => requestAnimationFrame(resolve));
  });
  for (const animation of document.getAnimations()) {
    animation.finish();
  }
};

const DERIVED = /^(inline-size|block-size|width|height|perspective-origin|transform-origin|-webkit|-moz|outline-offset-|border-(inline|block)|padding-(inline|block)|margin-(inline|block)|caret-color|column-rule|text-decoration|text-emphasis|min-|max-|inset|border-(start|end))/u;

it('reads the leaks', { timeout: 600_000 }, async () => {
  const lines: string[] = [];
  const summary = new Map<string, Set<string>>();
  let compared = 0;
  const style = document.createElement('style');
  style.textContent = '*,*::before,*::after{transition:none!important;animation:none!important}';
  document.head.append(style);

  lines.push('# exposed properties: read, and declared by no unconditional :host rule');
  for (const [, html] of CONTENT) {
    const holder = document.createElement('div');
    holder.innerHTML = html;
    document.body.append(holder);
    await frame();
    for (const element of family(holder.firstElementChild!)) {
      if (!exposedCache.has(element.localName)) {
        lines.push(`${element.localName}: ${exposed(element).join(' ') || '-'}`);
      }
    }
    holder.remove();
  }

  // A reset of every exposed property, outside any container, changes nothing.
  for (const [label, html] of CONTENT) {
    const holder = document.createElement('div');
    holder.innerHTML = html;
    document.body.append(holder);
    await frame();
    const before = snapshot(holder.firstElementChild!);
    for (const element of family(holder.firstElementChild!)) {
      reset(element, exposed(element));
    }
    await frame();
    for (const [key, value] of snapshot(holder.firstElementChild!)) {
      if (before.get(key) !== value) {
        lines.push(`CONTROL CHANGED ${label}: ${key}: ${before.get(key)} -> ${value}`);
      }
    }
    holder.remove();
  }

  function reset(element: Element, names: readonly string[]): () => void {
    const root = element.shadowRoot!;
    const previous = root.adoptedStyleSheets;
    const sheet = new CSSStyleSheet();
    sheet.replaceSync(`:host{${names.map((n) => `${n}:initial`).join(';')}}`);
    root.adoptedStyleSheets = [sheet, ...previous];
    return () => {
      root.adoptedStyleSheets = previous;
    };
  }

  for (const container of CONTAINERS) {
    for (const [label, html] of CONTENT) {
      if (container.only && !container.only.test(label)) {
        continue;
      }
      const holder = document.createElement('div');
      holder.innerHTML = container.html;
      const host = holder.querySelector<HTMLElement>(container.select)!;
      const template = document.createElement('template');
      template.innerHTML = html;
      const content = document.importNode(template.content, true).firstElementChild!;
      if (container.slot) {
        content.setAttribute('slot', container.slot);
      }
      host.append(content);
      document.body.append(holder);
      container.prepare?.(host);
      await frame();
      const members = family(content);
      const inherited = getComputedStyle(host);
      const names = new Set(members.flatMap((m) => exposed(m)));
      for (const name of names) {
        if (inherited.getPropertyValue(name) === '') {
          continue;
        }
        compared += 1;
        const before = snapshot(content);
        const undo = members.map((m) => (exposed(m).includes(name) ? reset(m, [name]) : undefined));
        const after = snapshot(content);
        for (const u of undo) {
          u?.();
        }
        const changed = [...after].filter(([key, value]) => before.get(key) !== value);
        const visible = changed.filter(([key]) => !DERIVED.test(key.split(' ').at(-1)!));
        if (changed.length > 0) {
          const key = `${container.name} | ${name}`;
          summary.set(key, (summary.get(key) ?? new Set()).add(label));
          const [first, out] = visible[0] ?? changed[0]!;
          lines.push(
            `LEAK ${container.name} > ${label}: ${name}=${inherited.getPropertyValue(name)} changes ${changed.length} values, such as ${first}: ${out} -> ${before.get(first)}`,
          );
        }
      }
      holder.remove();
    }
  }
  const text = [
    `# ${server.browser}, ${compared} (container, content, property) triples compared`,
    '# summary: container | property: contents',
    ...[...summary].map(([key, contents]) => `${key} (${contents.size}): ${[...contents].join(', ')}`),
    ...lines,
  ].join('\n');
  await commands.writeFile(`./leak-${server.browser}.txt`, text);
});
