## wf-read.mjs
import { readFileSync } from 'node:fs';
const data = JSON.parse(readFileSync(process.argv[2], 'utf8'));
const f = data.features;
const want = process.argv.slice(3);
// find features by id or by compat key substring
for (const w of want) {
  const hits = Object.entries(f).filter(([id, v]) => id === w || (v.compat_features ?? []).some((k) => k === w));
  if (!hits.length) { console.log(`${w}: no feature`); continue; }
  for (const [id, v] of hits) {
    const st = v.status ?? {};
    let per = '';
    if (st.by_compat_key && st.by_compat_key[w]) {
      const k = st.by_compat_key[w];
      per = ` | key ${w}: baseline=${k.baseline} low=${k.baseline_low_date ?? '-'} support=${JSON.stringify(k.support)}`;
    }
    console.log(`${w} -> ${id}: baseline=${st.baseline} low=${st.baseline_low_date ?? '-'} support=${JSON.stringify(st.support)}${per}`);
  }
}

## Reading
#
# What the code needs to run, the latest Baseline date first:
#   individual transform properties (`style.rotate`, `style.scale`, `style.translate`)   2022-08-05  Chrome 104, Firefox 72, Safari 14.1
#     composeNode compares each with 'none'. Without the property the value is undefined, so `get2DRotation(undefined)`
#     and `style.scale.split` throw on every element. This is the hard floor.
#   Array.prototype.at (`values.at(-1)`)                                                 2022-03-14  Chrome 92, Firefox 90, Safari 15.4
#   `??` and `??=`                                                                        2020-09-16
#   DOMMatrix with a CSS transform string, scaleSelf, rotateSelf, multiplySelf,
#     preMultiplySelf, is2D                                                               2020-01-15
#   assignedSlot, getRootNode, ShadowRoot.host                                            2020-01-15
#   ES modules, template literals, getClientRects, getComputedStyle, Number.*             2020 or earlier
# So `baseline 2022` is the lowest year in which every function runs.
#
# What the code needs for its CSS `zoom` results to be right:
#   CSS `zoom` (web-features `zoom`)                                                      2024-05-14  Firefox 126
#     In a Firefox before 126, `style.zoom` is undefined, zoomOf returns 1, and the engine applies no zoom either, so the
#     result is right: the page has no zoom to account for.
#   Chrome's standardized zoom                                                            2024-08-20  Chrome 128
#     Before Chrome 128, getClientRects divided by the zoom of the element and its ancestors. box-quad reads the client
#     rect as scaled and composes zoom into the matrix, so in Chromium 108-127 an element under a CSS `zoom` other than 1
#     gets a wrong box, and nothing returns false for it. `baseline 2024` lists Chrome and Edge from 130, past 128.
#   WebKit's zoom before Safari 18.2 was not read here.
# So `baseline 2024` is the lowest year in which the whole contract, zoom included, holds in every core browser by
# this reading, and `baseline 2022` and `baseline 2023` hold it only for a page with no CSS `zoom` other than 1.
