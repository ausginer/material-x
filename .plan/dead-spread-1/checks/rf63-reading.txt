// RF-63 probe: bundles each material-x runtime entry with rolldown (the size
// preset's options) and esbuild (W1's options), and reports bytes and markers.
// Usage: node probe.ts <repo-root> [out-dir]
import { brotliCompressSync, gzipSync, constants } from 'node:zlib';
import { mkdir, readFile, writeFile } from 'node:fs/promises';
import { join } from 'node:path';
import { createRequire } from 'node:module';

const root = process.argv[2]!;
const outDir = process.argv[3];
const pkg = join(root, 'packages/material-x');
const require = createRequire(join(root, 'package.json'));
const { rolldown } = await import(require.resolve('rolldown'));
const esbuild = await import(require.resolve('esbuild'));
const files = JSON.parse(await readFile(join(pkg, 'files.json'), 'utf8'));
const entries: string[] = [...files.runtime];

const markers: Record<string, RegExp> = {
  ffSym: /Symbol\(["`]FocusForwarding["`]\)/g,
  traitRt: /Trait transformer must return/g,
  spread: /\[\.\.\.[\w$]+\]?/g,
};

async function viaRolldown(file: string): Promise<string> {
  const bundle = await rolldown({ input: [file], platform: 'neutral', logLevel: 'silent' });
  try {
    const { output } = await bundle.generate({ format: 'es', minify: true });
    return output.filter((c: any) => c.type === 'chunk').map((c: any) => c.code).join('');
  } finally {
    await bundle.close();
  }
}
async function viaEsbuild(file: string): Promise<string> {
  const r = await esbuild.build({ entryPoints: [file], bundle: true, minify: true, format: 'esm', write: false, logLevel: 'silent' });
  return r.outputFiles.map((f: any) => f.text).join('');
}
const sizes = (code: string) => {
  const b = Buffer.from(code);
  return [b.length, brotliCompressSync(b, { params: { [constants.BROTLI_PARAM_QUALITY]: 11 } }).length, gzipSync(b, { level: 9 }).length];
};

console.log(['bundler', 'entry', 'min', 'brotli', 'gzip', ...Object.keys(markers)].join('\t'));
for (const [name, fn] of [['rolldown', viaRolldown], ['esbuild', viaEsbuild]] as const) {
  for (const entry of entries) {
    const code = await fn(join(pkg, `${entry}.js`));
    const counts = Object.values(markers).map((re) => (code.match(re) ?? []).length);
    console.log([name, entry, ...sizes(code), ...counts].join('\t'));
    if (outDir) {
      await mkdir(join(outDir, name), { recursive: true });
      await writeFile(join(outDir, name, `${entry.replace('/', '_')}.js`), code);
    }
  }
}
