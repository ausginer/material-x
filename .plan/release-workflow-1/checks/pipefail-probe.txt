// Runs the pack and stage steps of release.yml with a failing producer, under
// the shell the workflow gives each step: `bash -e {0}` with no `shell`, and
// `bash --noprofile --norc -eo pipefail {0}` with `shell: bash`.
import { parse } from '/workspaces/worktrees/d/node_modules/yaml/dist/index.js';
import { readFile, writeFile, mkdir } from 'node:fs/promises';
import { spawnSync } from 'node:child_process';
const [workflow, dir] = process.argv.slice(2);
const wf = parse(await readFile(workflow, 'utf8'));
const shell = wf.defaults?.run?.shell;
const steps = {
  pack: wf.jobs.prepare.steps.find((s) => s.id === 'package'),
  stage: wf.jobs.stage.steps.find((s) => s.name === 'Stage'),
};
const subs = {
  pack: [['aube pack --ignore-scripts --json --pack-destination "$RUNNER_TEMP/release"', 'false'], ['node .scripts/release-check.ts package', 'echo "REACHED release-check" #']],
  stage: [['npm stage publish "./$TARBALL" --access public --tag "$DIST_TAG" --provenance', 'false']],
};
const args = shell === 'bash' ? ['--noprofile', '--norc', '-eo', 'pipefail'] : ['-e'];
for (const [name, step] of Object.entries(steps)) {
  let script = step.run;
  for (const [a, b] of subs[name]) { if (!script.includes(a)) throw new Error(`${name}: no ${a}`); script = script.replace(a, b); }
  const temp = `${dir}/${name}`; await mkdir(`${temp}/release`, { recursive: true });
  const summary = `${temp}/summary.md`; await writeFile(summary, '');
  await writeFile(`${temp}/step.sh`, script);
  const r = spawnSync('bash', [...args, `${temp}/step.sh`], { cwd: temp, encoding: 'utf8', env: { ...process.env, RUNNER_TEMP: temp, GITHUB_STEP_SUMMARY: summary, TARBALL: 'x.tgz', DIST_TAG: 'latest', DIRECTORY: '.' } });
  console.log(`${name}: shell=${shell ?? '(none)'} bash ${args.join(' ')} -> exit ${r.status}; stdout: ${JSON.stringify(r.stdout.trim())}; summary: ${JSON.stringify((await readFile(summary, 'utf8')).trim())}`);
}
