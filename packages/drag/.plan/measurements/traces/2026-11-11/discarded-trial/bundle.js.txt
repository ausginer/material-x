	/**
	* Callbacks, each with the receiver it is called on, kept in two adjacent slots
	* of one array, so an entry allocates no object. A callback added with no
	* receiver is called with none.
	*
	* A list holds only the name of the array method it adds with. `queue()` makes
	* a list that adds at the start of the array, and `stack()` one that adds at
	* the end. `run` takes the entry at the end for both, so a queue runs its
	* oldest entry and a stack its newest, and `run` does not branch on the kind.
	*
	* The list decides nothing about when or whether its callbacks run, or what
	* happens when one throws. An owner whose entries coalesce checks `has` before
	* `add`. An entry is removed before its callback runs, so a throw leaves the
	* list consistent and a callback that adds itself again is added as a new
	* entry.
	*/
	var CallbackList = class CallbackList {
		/** A list run first in first out. */
		static queue() {
			return new CallbackList("unshift");
		}
		/** A list run last in first out. */
		static stack() {
			return new CallbackList("push");
		}
		#entries = [];
		#insertName;
		constructor(insertName) {
			this.#insertName = insertName;
		}
		/** Whether no entry is waiting. */
		get empty() {
			return this.#entries.length === 0;
		}
		/** Whether the same callback is waiting with the same receiver. */
		has(callback, receiver) {
			for (let i = 0; i < this.#entries.length; i += 2) if (this.#entries[i] === callback && this.#entries[i + 1] === receiver) return true;
			return false;
		}
		/**
		* Adds an entry, at the start of a queue's array and at the end of a stack's,
		* whether or not the same pair is waiting.
		*/
		add(callback, receiver) {
			this.#entries[this.#insertName](callback, receiver);
		}
		/**
		* Removes the entry at the end of the array and calls its callback with
		* `argument`, on its receiver. Call it only when an entry is waiting.
		*/
		run(argument) {
			const receiver = this.#entries.pop();
			this.#entries.pop().call(receiver, argument);
		}
		/** Drops every entry and the references it holds. */
		clear() {
			this.#entries.length = 0;
		}
	};

	* The public authoring API and the single-pass kernel behind it.

	/**
	* Whether a hop before the subject is marked `data-drag-ignore`. The consumer
	* owns that attribute, and the library infers nothing about element types.
	*/
	function ownsInteraction(event, subject) {
		for (const hop of event.composedPath()) {
			if (hop === subject) return false;
			if (hop.matches?.("[data-drag-ignore]")) return true;
		}
		return false;
	}

			}

				if (!this.#steps.has(step, receiver)) this.#steps.add(step, receiver);
			} else this.enter(step, this.#session ? this.#session.stepTarget() : null, receiver);

			if (this.#session) {
				this.#session.decide(reason, "request");

			}

		* A request made by anything run here is seen on the next turn.
		*/
		#applyRequests() {
			for (;;) try {
				if (this.#resolveDestroyed) {
					this.#teardown();

				}
				if (!this.#steps.empty) this.#steps.run(this.#session ? this.#session.stepTarget() : null);

	/**
	* One flat-tree node's own linear contribution, its CSS `zoom` included.
	*
	* `false` is *not representable in 2D*; `undefined` is *contributes nothing*.
	* The zoom factor is passed in rather than read here because the two callers
	* account for it differently — an ancestor's accumulates into the space, the
	* measured element's does not — and it seeds the node because zoom applies
	* outside everything the element itself authors.
	*
	* A `display: contents` node generates no box and therefore takes no transform,
	* but its zoom still applies, which is why the seed is returned rather than
	* discarded.
	*/
	function composeNode(style, Matrix, zoom) {
		let node = zoom === 1 ? void 0 : new Matrix().scaleSelf(zoom);
		if (style.display === "contents") return node;
		if (style.perspective !== "none" || style.transformStyle === "preserve-3d") return false;
		if (style.rotate !== "none") {
			const angle = get2DRotation(style.rotate);
			if (angle == null) return false;
			(node ??= new Matrix()).rotateSelf(angle);
		}
		if (style.scale !== "none") {
			const values = style.scale.split(" ");
			const scaleX = parse(values[0]);
			const scaleY = parse(values[1] ?? values[0]);
			if (values.length > 2 && parse(values[2]) !== 1) return false;
			(node ??= new Matrix()).scaleSelf(scaleX, scaleY);
		}
		if (style.transform !== "none") {
			const transform = new Matrix(style.transform);
			if (!transform.is2D) return false;
			if (node) node.multiplySelf(transform);
			else node = transform;
		}
		if (style.translate !== "none") {
			const values = style.translate.split(" ");
			if (values.length > 2 && parse(values[2]) !== 0) return false;
		}
		return node;
	}

	/**
	* Reads the space `element` inherits into `out`: the linear part of every
	* flat-tree node strictly above it, composed, together with the cumulative CSS
	* `zoom` of those same nodes.
	*
	* **It never touches the element's own box.** It reads computed style walking
	* upward and takes no layout, so it answers for an element that has no
	* principal box of its own — one that is `display: contents`, fragmented across
	* lines, or disconnected — where {@link coordinates} must refuse.
	*
	* Returns `false` and leaves `out` **untouched** when some ancestor is not
	* representable in 2D: a 3D transform, `perspective`, or `preserve-3d`. That is
	* a recognized limit, not an error.
	*/
	function ancestry(element, out) {
		const view = element.ownerDocument.defaultView;
		const Matrix = view.DOMMatrix;
		const matrix = new Matrix();
		let current = flatParentOf(element);
		let ancestorZoom = 1;
		while (current) {
			const style = view.getComputedStyle(current);
			const zoom = zoomOf(style);
			const node = composeNode(style, Matrix, zoom);
			if (node === false) return false;
			ancestorZoom *= zoom;
			if (node) matrix.preMultiplySelf(node);
			current = flatParentOf(current);
		}
		out[SPACE_A] = matrix.a;
		out[SPACE_B] = matrix.b;
		out[SPACE_C] = matrix.c;
		out[SPACE_D] = matrix.d;
		out[SPACE_ANCESTOR_ZOOM$1] = ancestorZoom;
		return true;
	}
	/**
	* Measures one element into `out`. This is the only operation that touches
	* layout.
	*
	* It reads the element's client rect and its own computed style, composes its
	* own transform and CSS `zoom` onto the space it inherits, and derives the
	* untransformed border-box size the resulting matrix acts on.
	*
	* Pass `above` — the element's own {@link ancestry} — when you already hold it,
	* and the walk is not repeated. Omitting it reads the ancestry internally, so
	* the two forms differ only in who owns the intermediate value.
	*
	* **Precondition:** `above` must be *this* element's inherited space. Nothing
	* can check that: a `Space` is a caller-owned numeric array holding no element,
	* so a space read from somewhere else produces a confidently wrong matrix, in
	* the same way {@link projection}'s same-viewport requirement does.
	*
	* Returns `false` and leaves `out` **untouched** for geometry this library does
	* not model: a disconnected element, a fragmented one (`getClientRects()`
	* length other than 1), 3D transforms, `perspective`, `preserve-3d`, or a
	* non-finite result. A `false` return is a recognized limit, not an error;
	* unexpected platform exceptions are allowed to escape.
	*
	* There is deliberately no `relativeTo` here. Relative coordinates are not a
	* measurement concern — they are a basis conversion between two boxes that have
	* already been measured, which is {@link projection}.
	*/
	function coordinates(element, out, above) {
		const view = element.ownerDocument.defaultView;
		const rects = element.getClientRects();
		if (rects.length !== 1) return false;
		let inherited = above;
		if (!inherited) {
			inherited = space();
			if (!ancestry(element, inherited)) return false;
		}
		const style = view.getComputedStyle(element);
		const node = composeNode(style, view.DOMMatrix, zoomOf(style));
		if (node === false) return false;
		const ia = inherited[SPACE_A];
		const ib = inherited[SPACE_B];
		const ic = inherited[SPACE_C];
		const id = inherited[SPACE_D];
		const na = node ? node.a : 1;
		const nb = node ? node.b : 0;
		const nc = node ? node.c : 0;
		const nd = node ? node.d : 1;
		const a = ia * na + ic * nb;
		const b = ib * na + id * nb;
		const c = ia * nc + ic * nd;
		const d = ib * nc + id * nd;
		const rect = rects[0];
		const absoluteA = abs(a);
		const absoluteB = abs(b);
		const absoluteC = abs(c);
		const absoluteD = abs(d);
		const diagonal = absoluteA * absoluteD;
		const antiDiagonal = absoluteB * absoluteC;
		const determinant = diagonal - antiDiagonal;
		const determinantScale = diagonal + antiDiagonal;
		let width;
		let height;
		if (abs(determinant) > determinantScale * .05) {
			width = (rect.width * absoluteD - rect.height * absoluteC) / determinant;
			height = (rect.height * absoluteA - rect.width * absoluteB) / determinant;
		} else {
			width = getBorderSize(style, "width", "Left", "Right");
			height = getBorderSize(style, "height", "Top", "Bottom");
		}
		if (!finite(a + b + c + d + width + height)) return false;
		out[BOX_A$1] = a;
		out[BOX_B$1] = b;
		out[BOX_C$1] = c;
		out[BOX_D$1] = d;
		out[BOX_E$1] = rect.left - min(0, a * width) - min(0, c * height);
		out[BOX_F$1] = rect.top - min(0, b * width) - min(0, d * height);
		out[BOX_WIDTH$1] = width;
		out[BOX_HEIGHT$1] = height;
		return true;
	}
	/**
	* Converts a viewport delta into `basis` and writes it to `out`. Pure: no DOM
	* reads, no allocation.
	*
	* The basis selects what the result means:
	*
	* - An element's {@link Space}, from {@link ancestry}: the value to write to
	*   that element's `translate`, `left` or `top`. A `Space` excludes the
	*   element's own `zoom`, so an element with its own `zoom` divides the result
	*   by that `zoom`.
	* - An element's {@link Box}, from {@link coordinates}: the change in that
	*   element's own untransformed box, such as `width` or `height`.
	*
	* Returns `false` and leaves `out` untouched when the basis is singular or the
	* result is not finite.
	*/
	function localDelta(basis, dx, dy, out) {
		const a = basis[0];
		const b = basis[1];
		const c = basis[2];
		const d = basis[3];
		const inverseDeterminant = 1 / (a * d - b * c);
		const x = (d * dx - c * dy) * inverseDeterminant;
		const y = (a * dy - b * dx) * inverseDeterminant;
		if (!finite(x + y)) return false;
		out[0] = x;
		out[1] = y;
		return true;
	}

		#drag;
		#lift;
		/** `config.targets`' state for this operation, released with it. */
		#targets;
		/** The visual's viewport delta, axis-projected and clamped. */
		#motion = {

			y: 0

		#offsetX = 0;
		#offsetY = 0;
		/** `null` until the drop: a cancel before it has none. */
		#request = null;
		/** Recorded at `settle` and reported at `end`. */

			this.#lift = lift;

		* at their next read.
		*/

			this.#behavior.bounds?.invalidate();
			this.#targets?.invalidate();
		}
		/** Computes the motion and moves the visual by it. */
		translate() {
			this.#computeMotion();
			this.#lift.translate(this.#motion.x, this.#motion.y);
		}

			this.#offsetX = point.x - this.#lift.rect.left - (this.#drag.x - this.#drag.originX);

		}

			this.#behavior.config.onStart?.(this.#geometry());
		}
		move() {
			this.translate();
			this.#targets?.move();
			if (this.#behavior.config.onMove) this.#behavior.config.onMove(this.#geometry());
		}

			this.#computeMotion();
			this.#targets?.drop();
			const request = buildRequest(this.#behavior.root, this.#drag, this.#lift, this.#motion, this.#home(), this.#targets?.current ?? null, this.#behavior.window);
			this.#request = request;
			this.#lift.translate(this.#motion.x, this.#motion.y);
			return (signal) => this.#behavior.config.onDrop(request, { signal });
		}

					reason: outcome.reason,
					origin: outcome.origin

				request: this.#request,

			};
			else {

					type: "accepted",
					request: this.#request
				};

			}

		}

				item: this.#behavior.root,
				visual: this.#lift.visual
			}) : {
				x: this.#lift.rect.left,
				y: this.#lift.rect.top
			};
		}

	* new positions instead of jumping.

	* easing and the animations it starts. A rebuild never cancels an animation, because

	function linearShift(index) {
		/** A prediction is valid only against a buffer that describes the DOM. */
		let dirty = true;
		/** The destination gap the buffer reflects, or `-1` when nothing is known. */
		let lastGap = -1;
		/** The collection version the buffer holds, or `-1` for nothing. */
		let snapshotVersion = -1;
		/**
		* The distance one crossed row travels along the axis, or `-1` while unmeasured.
		* Every invalidation discards it, because a zoom, a resize or a replaced
		* collection all change it, and the cache has one staleness flag with no reason
		* attached. The cost is one row read on the next committed move.
		*/
		let rowDistance = -1;
		/** A committed move left the cached placeholder rect out of date. */
		let placeholderIsStale = false;
		return {
			refresh(snapshot, dragged, getBox, placeholder, displacement, gap) {
				if (placeholderIsStale) {
					placeholderIsStale = false;
					index.remeasurePlaceholder(placeholder);
				}
				index.refresh(snapshot, dragged, getBox, placeholder, displacement);
				snapshotVersion = snapshot.version;
				dirty = false;
				if (gap >= 0) lastGap = gap;
			},
			moved(gap, snapshot, context) {
				if (dirty || snapshotVersion !== snapshot.version || index.count === 0 || lastGap < 0 || lastGap === gap) {
					dirty = true;
					lastGap = -1;
					placeholderIsStale = false;
					index.invalidate();
					return;
				}
				const lo = lastGap < gap ? lastGap : gap;
				const hi = lastGap < gap ? gap : lastGap;
				let delta = lastGap < gap ? -rowDistance : rowDistance;
				if (rowDistance < 0) {
					const probe = index.items[lo];
					let observed = (context.box ? context.box(probe) : probe).getBoundingClientRect().top;
					if (context.displacement) {
						const scratch = /* @__PURE__ */ new Float64Array(6);
						scratch[1] = observed;
						context.displacement.settle(scratch, [probe], 1);
						observed = scratch[1];
					}
					delta = observed - index.values[lo * 6 + 1];
					rowDistance = delta < 0 ? -delta : delta;
				}
				index.advance(lo, hi, delta);
				if (context.displacement) {
					const { items } = index;
					for (let i = lo; i < hi; i += 1) context.displacement.report(items[i], 0, -delta, context.space);
				}
				lastGap = gap;
				placeholderIsStale = true;
			},
			invalidate() {
				dirty = true;
				rowDistance = -1;
				placeholderIsStale = false;
				index.invalidate();
			},
			retire() {
				dirty = true;
				lastGap = -1;
				snapshotVersion = -1;
				rowDistance = -1;
				placeholderIsStale = false;
				index.retire();
			}
		};

		if (!incumbent.after) return destination.at(-1) === incumbent.before ? insertionAt(destination, destination.length) : null;
		const beforeIndex = destination.indexOf(incumbent.before);
		return beforeIndex >= 0 && destination[beforeIndex + 1] === incumbent.after ? insertionAt(destination, beforeIndex + 1) : null;
	}
	/**
	* The gap the dragged item itself occupies, from the snapshot and without
	* creating a destination view: removing the item leaves earlier elements in place
	* and shifts later ones down by one.
	*/
	function homeInsertion(snapshot, item) {
		const from = snapshot.items.indexOf(item);
		return {
			index: from,
			before: snapshot.items[from - 1] ?? null,
			after: snapshot.items[from + 1] ?? null
		};
	}
	/** The reorder a gap proposes. `insertion` comes from the same snapshot. */
	function buildReorderRequest(snapshot, item, insertion) {
		return {
			item,
			from: snapshot.items.indexOf(item),
			to: insertion.index,
			previous: insertion.before,
			next: insertion.after
		};
	}
	const ARROW_UP = "ArrowUp";
	const ARROW_LEFT = "ArrowLeft";
	const ARROW_DOWN = "ArrowDown";
	const ARROW_RIGHT = "ArrowRight";
	/** The command a key names, or `null` for every other key. */
	function directionOf(key) {
		if (key === ARROW_UP || key === ARROW_LEFT) return 110;
		if (key === ARROW_DOWN || key === ARROW_RIGHT) return 111;
		return null;
	}
	/**
	* The destination gap for moving `item` one slot, or `null` at the edge, which
	* makes the command inert so that the arrow key keeps its native meaning. Moving
	* up takes the predecessor's slot. Moving down passes the successor, which has
	* already shifted into `from`, so the gap is `from + 1`. `item` is in `snapshot`.
	*/
	function keyboardInsertion(snapshot, item, direction) {
		const from = snapshot.items.indexOf(item);
		const up = direction === 110;
		if (up ? from === 0 : from === snapshot.items.length - 1) return null;
		return insertionAt(snapshot.items.filter((candidate) => candidate !== item), up ? from - 1 : from + 1);
	}

	/**
	* Creates the operation's placeholder, detached. The mechanics here cannot be
	* configured away: one insertion position, hidden from assistive technology, in
	* the item's slot, sized from the footprint the visual removed. Beyond this the
	* library writes no styling.
	*/
	function createPlaceholder(document, context, width, height, factory) {
		const placeholder = factory ? factory(context) : document.createElement("div");
		const slot = context.item.getAttribute("slot");
		placeholder.setAttribute("data-drag-placeholder", "");
		placeholder.setAttribute("aria-hidden", "true");
		if (slot === null) placeholder.removeAttribute("slot");
		else placeholder.setAttribute("slot", slot);
		placeholder.style.boxSizing = "border-box";
		placeholder.style.width = `${width}px`;
		placeholder.style.height = `${height}px`;
		return placeholder;
	}
	/** Whether the placeholder already occupies `insertion`. */
	function placeholderAt(placeholder, insertion) {
		if (insertion.after) return insertion.after.previousElementSibling === placeholder;
		return !insertion.before || insertion.before.nextElementSibling === placeholder;
	}
	/**
	* Moves the placeholder into `insertion`, or does nothing when it is there: a
	* re-insertion would reset transitions and force a reflow. An anchor the consumer
	* moved out of the container throws, so that the footprint does not silently go
	* with it.
	*/
	function movePlaceholder(placeholder, insertion) {
		if (placeholderAt(placeholder, insertion)) return;
		if ((insertion.after ?? insertion.before).parentNode !== placeholder.parentNode) throw new Error("sortable/anchor-outside-container");
		if (insertion.after) insertion.after.before(placeholder);
		else insertion.before.after(placeholder);
	}

	const RECOVERY_DESTINATION = 1;
	const RECOVERY_HOME = 2;
	const RECOVERY_IMMEDIATE = 3;
	/**
	* One activated sortable operation, from `start` to `end`. The kernel calls each
	* hook as a method of this instance.
	*/
	var SortableOperation = class {
		#behavior;
		#drag;
		#lift;
		/** A command's direction, or `null` for a pointer operation. */
		#direction;
		#frame;
		#context;
		/** The operation state the axis rule reads. */
		#query;
		/**
		* The item's gap in the pull that is the operation's collection. A landing at
		* home reads it, so no cancel, verdict or release reads the consumer's array.
		*/
		#home;
		/** `null` until the drop: a cancel before it has none. */
		#request = null;
		/** Recorded at `settle` and reported at `end`. */
		#result = null;
		constructor(behavior, drag, lift, direction, snapshot, home, insertion, frame, context) {
			this.#behavior = behavior;
			this.#drag = drag;
			this.#lift = lift;
			this.#direction = direction;
			this.#frame = frame;
			this.#context = context;
			this.#home = home;
			this.#query = {
				item: drag.item,
				snapshot,
				insertion,
				pointerX: 0,
				pointerY: 0
			};
		}
		start() {
			this.#behavior.config.onStart?.(this.#query.item);
		}
		move() {
			this.#lift.translate(this.#drag.x - this.#drag.originX, this.#drag.y - this.#drag.originY);
			this.#frame.schedule();
		}
		drop() {
			const isPointerOperation = this.#direction === null;
			this.#frame.cancel();
			if (isPointerOperation) {
				this.#behavior.axis.invalidate();
				this.#query.pointerX = this.#drag.x;
				this.#query.pointerY = this.#drag.y;
				this.#query.insertion = this.#behavior.axis.resolve(this.#query, this.#context) ?? this.#query.insertion;
			}
			const request = buildReorderRequest(this.#query.snapshot, this.#drag.item, this.#query.insertion);
			this.#request = request;
			movePlaceholder(this.#context.placeholder, this.#query.insertion);
			if (isPointerOperation) this.#lift.translate(this.#drag.x - this.#drag.originX, this.#drag.y - this.#drag.originY);
			return request.to === request.from ? null : (signal) => this.#behavior.config.onReorder(request, { signal });
		}
		settle(outcome) {
			const recovery = this.#recordResult(outcome);
			if (this.#behavior.landing && recovery !== RECOVERY_IMMEDIATE) {
				const { left, top } = this.#landingTarget(recovery);
				this.#lift.land(left, top, this.#behavior.landing);
			}
		}
		end() {
			this.#behavior.config.onEnd?.(this.#result);
		}
		/** The collection was pulled again: rebases the gap, or cancels. */
		rebase(next) {
			this.#query.snapshot = next;
			this.#home = homeInsertion(next, this.#drag.item);
			if (next.items.includes(this.#drag.item)) {
				const insertion = reconcileCollection(next, this.#drag.item, this.#query.insertion);
				if (insertion) this.#query.insertion = insertion;
				else this.#drag.requestCancel(CANCEL_COLLECTION_INVALIDATED);
			} else this.#drag.requestCancel(CANCEL_ITEM_REMOVED);
			this.#behavior.axis.invalidate();
		}
		/** The gap under the pointer, and the placeholder moved to it. */
		updateInsertion() {
			this.#query.pointerX = this.#drag.x;
			this.#query.pointerY = this.#drag.y;
			const resolved = this.#behavior.axis.resolve(this.#query, this.#context);
			if (resolved === null) return;
			this.#query.insertion = resolved;
			if (placeholderAt(this.#context.placeholder, resolved)) return;
			let stale = true;
			try {
				movePlaceholder(this.#context.placeholder, resolved);
				this.#behavior.axis.moved(this.#query, this.#context);
				stale = false;
			} finally {
				if (stale) this.#behavior.axis.invalidate();
			}
		}
		/** Records the terminal result, and says where the visual goes. */
		#recordResult(outcome) {
			if (outcome.type === "canceled") {
				this.#result = {
					type: "canceled",
					reason: outcome.reason,
					origin: outcome.origin,
					request: this.#request
				};
				return outcome.origin === "fault" ? RECOVERY_IMMEDIATE : RECOVERY_HOME;
			}
			if (outcome.type === "skipped") {
				this.#result = {
					type: "noop",
					request: this.#request
				};
				return RECOVERY_IMMEDIATE;
			}
			if (outcome.value === ACCEPTED) {
				this.#result = {
					type: "accepted",
					request: this.#request
				};
				return RECOVERY_DESTINATION;
			}
			this.#result = {
				type: "rejected",
				reason: outcome.value[0],
				request: this.#request
			};
			return RECOVERY_HOME;
		}
		/**
		* Only with a landing composed: the one forced layout a drop adds. Returns the
		* placeholder's rect, whose top-left is where the visual lands.
		*/
		#landingTarget(recovery) {
			if (recovery === RECOVERY_DESTINATION) {
				if (this.#query.item.isConnected && this.#query.item.parentElement === this.#context.placeholder.parentElement && this.#context.placeholder.nextElementSibling !== this.#query.item) this.#query.item.before(this.#context.placeholder);
			} else movePlaceholder(this.#context.placeholder, this.#home);
			if (!this.#context.placeholder.isConnected || this.#context.placeholder.parentElement !== this.#query.item.parentElement) throw new Error("sortable/landing-placeholder-lost");
			return this.#context.placeholder.getBoundingClientRect();
		}
	};

	/** Step, once per frame: the gap under the pointer, and the placeholder moved to it. */

	/**
	* The behavior `draggable` runs for one root: admission and activation. Each
	* step it requests is the same function on every request, or the axis's method
	* passed with the axis, so the kernel coalesces a step requested again before it
	* runs. A function it creates to hand elsewhere is created once per instance, or
	* once per operation when it closes over the operation. It holds only
	* controller-lifetime values, and the operation reads the ones it needs through
	* it.
	*/
	var SortableBehavior = class {
		config;
		#window;
		#lifter;
		axis;
		displacement;
		landing;
		#measuredBox;
		/** The version the next pulled snapshot carries. */
		#version = 0;
		constructor(root, config, displacement, lifter) {
			if (config.threshold !== void 0) this.threshold = config.threshold;
			this.config = config;
			this.#window = root.ownerDocument.defaultView;
			this.#lifter = lifter;
			this.axis = config.axis();
			this.displacement = displacement;
			this.landing = config.landing ?? null;
			this.#measuredBox = config.box ?? config.visual ?? null;
		}
		/** Step, on `invalidate()`: re-reads the collection and rebases the gap. */
		invalidate = (operation) => {
			if (operation) operation.rebase(this.#pull());
		};
		admit(event) {
			return event.altKey ? null : this.#admission(event, null);
		}
		command(event) {
			const direction = event.isComposing ? null : directionOf(event.key);
			return direction === null ? null : this.#admission(event, direction);
		}
		activate(drag, state) {
			const lift = this.#lifter.acquire(drag, state.visual, state.box);
			const snapshot = this.#pull();
			if (!snapshot.items.includes(drag.item)) return null;
			const home = homeInsertion(snapshot, drag.item);
			const insertion = state.direction === null ? home : keyboardInsertion(snapshot, drag.item, state.direction);
			if (!insertion) return null;
			const placeholder = createPlaceholder(this.#window.document, {
				item: drag.item,
				visual: lift.visual,
				box: lift.box,
				rect: lift.rect
			}, lift.boxWidth, lift.visual.contains(lift.box) ? lift.boxHeight : lift.boxHeight - lift.box.offsetHeight, this.config.placeholder ?? null);
			drag.use(() => {
				placeholder.remove();
			});
			drag.item.after(placeholder);
			if (!placeholder.isConnected || drag.item.nextElementSibling !== placeholder) throw new Error("sortable/insertion-placeholder-lost");
			const frame = createFrameTask(this.#window, () => {
				drag.requestStep(updateInsertion);
			});
			drag.use(() => {
				frame.cancel();
			});
			drag.use(this.axis.retire, this.axis);
			listenForInvalidation(this.#window, drag.signal, () => {
				drag.requestStep(this.axis.invalidate, this.axis);
			});
			const operation = new SortableOperation(this, drag, lift, state.direction, snapshot, home, insertion, frame, {
				placeholder,
				box: this.#measuredBox,
				displacement: this.displacement ?? null,
				space: lift.itemSpace
			});
			this.axis.invalidate();
			return operation;
		}
		#pull() {
			this.#version += 1;
			return {
				items: this.config.items(),
				version: this.#version
			};
		}
		/**
		* The pressed item from a fresh collection, and the admission for it. A command
		* names its direction and is refused at the collection's edge, so that the key
		* keeps its native meaning there.
		*/
		#admission(event, direction) {
			const snapshot = this.#pull();
			const path = event.composedPath();
			const item = path.find((node) => snapshot.items.includes(node));
			if (!item) return null;
			let handle;
			if (this.config.handle) {
				const named = this.config.handle(item);
				if (!named || !path.includes(named)) return null;
				handle = named;
			}
			if (direction !== null && !keyboardInsertion(snapshot, item, direction)) return null;
			const visual = this.config.visual?.(item) ?? item;
			return {
				item,
				...handle ? { handle } : {},
				state: {
					visual,
					box: this.config.box?.(item) ?? visual,
					direction
				}
			};
		}
	};

	/**
	* The sortable, written on the public `draggable` with no private path. The
	* consumer owns the DOM order: a drop proposes a reorder and the consumer commits or
	* rejects it.
	*/
	/** Makes `root` a sortable collection. */
	function sortable(root, config) {
		const displacement = config.displacement?.();
		const lifter = new Lifter(root.ownerDocument.defaultView);
		const behavior = new SortableBehavior(root, config, displacement, lifter);
		const kernel = draggable(root, behavior);
		if (displacement) kernel.use(displacement.retire, displacement);
		kernel.use(lifter.cancelLanding);
		return {
			invalidate: () => {
				kernel.requestStep(behavior.invalidate);
			},
			requestCancel: (reason) => {
				kernel.requestCancel(reason);
			},
			requestDestroy: () => kernel.requestDestroy()
		};
	}
