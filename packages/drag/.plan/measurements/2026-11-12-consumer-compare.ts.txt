/* Compares the base suite's titles and verdicts with the final suite's, per project. */
import { readFile } from 'node:fs/promises';

const [baseJson, finalLog] = process.argv.slice(2) as [string, string];

type Row = { project: string; file: string; name: string; status: string };

function projectOf(file: string): string {
  if (file.endsWith('hot-path.browser.test.ts')) return 'shipped';
  if (file.endsWith('.browser.test.ts')) return 'browser';
  if (file.endsWith('.node.test.ts')) return 'node';
  return 'declaration';
}

const base: Row[] = [];
const report = JSON.parse(await readFile(baseJson, 'utf8')) as {
  testResults: {
    name: string;
    assertionResults: { fullName: string; status: string }[];
  }[];
};

for (const result of report.testResults) {
  const file = result.name.replace(/^.*\/packages\/drag\//u, '');

  for (const test of result.assertionResults) {
    base.push({ project: projectOf(file), file, name: test.fullName.replace(/^tests\/\S+ /u, ''), status: test.status });
  }
}

const final: Row[] = [];
const line =
  /^ (✓|×|↓) \|(\w+)(?: \(chromium\))?\|\s+(?:TS\s+)?(tests\/\S+?)(?::\d+)? > (.*?)(?: \d+ms)?$/u;

for (const text of (await readFile(finalLog, 'utf8')).split('\n')) {
  const match = line.exec(text);

  if (match) {
    const [, mark, project, file, path] = match as unknown as string[];

    final.push({
      project: project!,
      file: file!,
      name: path!.split(' > ').join(' ').replace(/^tests\/\S+ /u, ''),
      status: mark === '✓' ? 'passed' : mark === '×' ? 'failed' : 'skipped',
    });
  }
}

function summary(rows: Row[]): Record<string, string> {
  const out: Record<string, string> = {};

  for (const project of [...new Set(rows.map((row) => row.project))].sort()) {
    const own = rows.filter((row) => row.project === project);
    const files = new Set(own.map((row) => row.file)).size;
    const failed = own.filter((row) => row.status === 'failed');

    out[project] =
      `${files} files, ${own.length} tests, ${failed.length} failed` +
      (failed.length ? `: ${failed.map((row) => row.name).join(' | ')}` : '');
  }

  return out;
}

console.log('base', summary(base));
console.log('final', summary(final));

const key = (row: Row): string => `${row.name}\t${row.status}`;
const multiset = (rows: Row[]): Map<string, number> => {
  const map = new Map<string, number>();
  for (const row of rows) map.set(key(row), (map.get(key(row)) ?? 0) + 1);
  return map;
};

for (const project of ['browser', 'shipped', 'node', 'declaration']) {
  const a = multiset(base.filter((row) => row.project === project));
  const b = multiset(final.filter((row) => row.project === project));
  const diff = [...new Set([...a.keys(), ...b.keys()])].filter(
    (name) => a.get(name) !== b.get(name),
  );

  console.log(
    `${project}: titles and verdicts ${diff.length === 0 ? 'identical' : 'DIFFER'}`,
    diff,
  );
}

const moved = new Map<string, string>();
const baseFile = new Map(base.map((row) => [row.name, row.file]));

for (const row of final.filter((row) => row.project === 'browser')) {
  const from = baseFile.get(row.name);
  if (from && from !== row.file) moved.set(row.name, `${from} -> ${row.file}`);
}

console.log('moved', moved);

const consumerFiles = new Set(final.filter((row) => row.project === 'consumer').map((row) => row.file));
const browserFiles = new Set(final.filter((row) => row.project === 'browser').map((row) => row.file));
console.log('browser files not in consumer', [...browserFiles].filter((file) => !consumerFiles.has(file)).sort());
