#!/usr/bin/env bash
# rec.sh <out> <command...>: the machine record at the head and foot of <out>, and the command's output between.
out=$1; shift
m() { echo "nproc $(nproc), node $(node --version), cpu $(grep -m1 'model name' /proc/cpuinfo | cut -d: -f2 | xargs), cpu_core $(cat /sys/devices/cpu_core/cpus), cpu_atom $(cat /sys/devices/cpu_atom/cpus), uptime: $(uptime)"; }
{ echo "# command: $*"; echo "# revision: $(git rev-parse HEAD)$(git diff --quiet HEAD -- . || echo ', with uncommitted changes')"; echo "# before: $(m)"; } > "$out"
"$@" >> "$out" 2>&1; code=$?
echo "# after: $(m), exit $code" >> "$out"
exit $code
