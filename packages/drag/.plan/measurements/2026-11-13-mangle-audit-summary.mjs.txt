import { readFileSync } from 'node:fs';
const blocks = readFileSync(process.argv[2], 'utf8').split(/\n(?=## )/).slice(1);
for (const b of blocks) {
  const lines = b.split('\n');
  const name = lines[0].slice(3);
  const decls = lines.filter(l => l.startsWith('  decl ')).map(l => l.slice(7).trim());
  const dts = /dts-member: YES/.test(b) ? 'D' : '-';
  const plat = /platform: none/.test(b) ? '-' : 'P';
  const unctx = /uncontextual literal/.test(b) ? 'U' : '-';
  const str = /strings: none/.test(b) ? '-' : 'S';
  const src = b.match(/refs src\((\d+)\)/)[1], tests = b.match(/refs tests\((\d+)\)/)[1];
  console.log(`${dts}${plat}${unctx}${str}\t${name}\tsrc ${src} tests ${tests}\t${decls.join(' ; ')}`);
}
