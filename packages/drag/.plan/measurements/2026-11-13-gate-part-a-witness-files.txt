/* Scratch witness for DRF-23, an entry's source by two forms. Not committed. */
import { expect, it } from 'vitest';
import { draggable as published } from '@ydinjs/drag/draggable.js';
import { draggable as relative } from '../src/draggable.ts';
import { draggable as absolute } from '/workspaces/material-x/packages/drag/src/draggable.ts';

it('should give the same draggable by the relative and absolute forms', () => {
  expect(absolute).toBe(relative);
});

it('should give the emitted draggable by both forms', () => {
  expect([relative, absolute]).toEqual([published, published]);
});
