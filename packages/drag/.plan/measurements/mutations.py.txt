        cfg='vitest.engines.config.ts' if '/' in project else 'vitest.config.ts'

        try:
            data=json.load(open(path))
            for t in data['testResults']:
                for r in t['assertionResults']:
                    if r['status']=='failed':
                        failed.append(r['fullName'])
        except Exception as e:
            failed=['RUNNER ERROR '+str(e)]
    for f,s in originals.items(): open(f,'w').write(s)
    print(f"{dri}\t{name}\t{engine}\t{len(failed)}\t{failed[0] if failed else 'SURVIVED'}",flush=True)
