# mutate.py
# Usage: mutate.py <file> <head-nodrop|new-leak> <classes|destruction>.
import sys
path, kind, which = sys.argv[1:4]
s = open(path).read()
def sub(old, new):
    global s
    assert s.count(old) == 1, old
    s = s.replace(old, new, 1)
if kind == 'head-nodrop':
    # The test keeps its own reference to the destroyed controller.
    sub("      controller = null;\n", "")
else:
    # The controller escapes the function, so the test's frame still holds it.
    name = 'dragThenDestroy' if which == 'classes' else 'operateThenDestroy'
    sub(f"      const {name} = async (): Promise<void> => {{\n", f"      const leaked: unknown[] = [];\n      const {name} = async (): Promise<void> => {{\n")
    call = "freeDrag(box, {" if which == 'classes' else "sortable(list, {"
    sub(f"        const controller = {call}", f"        const controller = {call}")
    sub("        await controller.requestDestroy();\n      };", "        await controller.requestDestroy();\n        leaked.push(controller);\n      };")
open(path, 'w').write(s)
