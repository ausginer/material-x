# mutate.py
# Usage: mutate.py <lift.ts> <M1|M2|M3>. Rewrites Lifter.acquire's measurement.
import sys
path, kind = sys.argv[1], sys.argv[2]
s = open(path).read()
anchor = "      const rect = visual.getBoundingClientRect();\n"
assert s.count(anchor) == 1
pre = {
    # Cancels the running landing before anything is measured.
    'M1': "      this.cancelLanding();\n",
    # Detaches every animation's effect while measuring, so the flight's offset is absent.
    'M2': "      const running = visual.getAnimations();\n      const effects = running.map((animation) => animation.effect);\n      for (const animation of running) { animation.effect = null; }\n",
    # Moves every animation to its start while measuring: the release offset, not the current one.
    'M3': "      const running = visual.getAnimations();\n      for (const animation of running) { animation.currentTime = 0; }\n",
}[kind]
post = {
    'M1': "",
    'M2': "      running.forEach((animation, i) => { animation.effect = effects[i]!; });\n",
    'M3': "",
}[kind]
s = s.replace(anchor, pre + anchor + post, 1)
if kind == 'M2':
    # The rest of the measurement (coordinates) also runs before the effects return.
    s = s.replace(post, "", 1)
    a2 = "      const style = this.#window.getComputedStyle(visual);\n"
    assert s.count(a2) == 1
    s = s.replace(a2, post + a2, 1)
open(path, 'w').write(s)

# Usage: delay.py <test file>. Waits 1 s between the read before the grab and the grab.
