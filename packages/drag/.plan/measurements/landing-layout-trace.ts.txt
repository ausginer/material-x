async function bundle(lab: string, srcRoot: string) {
  const b = await rolldown({ input: lab, platform: 'browser', resolve: { alias: { '@ydinjs/box-quad': BQ } }, transform: { define: { __DEV__: 'false' } },
    plugins: [{ name: 'a', resolveId(id: string) { return id.startsWith('../src/') ? join(srcRoot, id.slice(7)) : null; } }] });
  const { output } = await b.generate({ format: 'iife', name: 'labExports' });
  await b.close();
  return output[0].code;
}

import { cpSync, readFileSync, writeFileSync } from 'node:fs';
async function variant(tag: string, file: string, find: string, rep: string) {
  const dir = join(scratch, tag); cpSync(join(ROOT, 'src'), dir, { recursive: true });
  const f = join(dir, file); const t = readFileSync(f, 'utf8');
  if (!t.includes(find)) throw new Error('no match ' + tag);
  writeFileSync(f, t.replace(find, rep));
  return bundle(join(ROOT, 'bench/lab.ts'), dir);
}
