/**
 * Before/after cost in one browser process: drag2, the prototype, production, and
 * production again as the noise floor. Arms are interleaved and every run has a
 * fresh page. CDP metrics, so Chromium only and headless, with no real compositing
 * cost.
 *
 * `node bench/compare.ts [--rounds n] [--rows validation] [--only text] [--out file]`
 * runs the comparison. `--before <revision>` replaces drag2 and the prototype with
 * production's `src` at that revision.
 * `node bench/compare.ts --prove [--out file]` injects a known cost into a scratch
 * copy of production, and exits non-zero unless the comparison flags each one
 * expected to flag and flags none expected to stay quiet, or if the control flags
 * against itself. The one quiet injection, the step-list entry, is the bytes rule's
 * specificity witness: kernel code that runs on every frame and allocates no object
 * stays under the rule. Its quiet result counts only when an injection expected to
 * flag ran on its row in the same run. With `--only a|b`, only the injections whose
 * name contains one of the texts run, and the A/A pass runs in full. Every selected
 * injection is applied before a browser starts, and the first one whose text does
 * not occur exactly once, or whose patch leaves the bundle unchanged, ends the run.
 * `--prove --dry` stops after that check: it launches no browser, and exits 0 when
 * every selected injection applies.
 * `node bench/compare.ts --floors [--only a|b] [--out file]` runs the detection
 * floors: graded injections, each run as a `--prove` injection is, that record
 * their verdicts and readings and never change the exit code. `--prove` doesn't
 * run them, and `--floors --dry` applies them with no browser. Their work is a
 * fixed number of loop iterations, calibrated once at 60 per µs; the size is
 * nominal, and the reading's median script difference is the measured one.
 * `node bench/compare.ts --rejudge <file>` applies `judge` to a recorded run.
 * `node bench/compare.ts --profile --only <row>` profiles one run of a row per arm,
 * and `--attribute` gives the retained row's growth per constructor per arm.
 *
 * A row whose id starts with `in the page:` mirrors the CDP row named by the rest
 * of its id, and drives its timed window from inside the page: one evaluation
 * dispatches every event as a `PointerEvent` of Chrome's mouse and awaits the
 * frames the CDP row awaits. Each CDP row's event costs an input command and, on
 * a move, an evaluation, and their script is most of the row's own, so an
 * in-page row's script is closer to the library's. A move row's press and
 * crossing are CDP input before the window; every event of a cycle is in the
 * window, because the press, activation and drop are what it measures. The
 * driver-alone row dispatches the 20-item move row's events to a list with no
 * behavior, which measures the driver's own share. These rows are validation
 * rows, and run only with arms built from this package's source: `--before`,
 * `--prove` and `--floors`, never with drag2 and the prototype.
 *
 * The rules are stated before running, in `judge`'s comment.
 */

/** The 0-based line at which a module's code starts in a bundle, and its key. */
type ModuleStart = Readonly<{ line: number; key: string }>;
type Bundle = Readonly<{ code: string; modules: readonly ModuleStart[] }>;

/**
 * Lab source bundled into an IIFE, unminified so that functions keep their names.
 * The bundler writes a line `// <path from the working directory>` where each
 * module starts. A module's key is its path under `sourceRoot`, such as
 * `kernel/session.ts`, so that every arm built from this package's source keys a
 * module alike. Any other module is keyed by its path from the package, such as
 * `bench/lab.ts`.
 */
async function bundle(lab: string, sourceRoot: string): Promise<Bundle> {
  const built = await rolldown({
    input: lab,
    platform: 'browser',
    resolve: { alias: { '@ydinjs/box-quad': BOX_QUAD } },
    transform: { define: { __DEV__: 'false' } },
    plugins: [
      {
        name: 'arm-source',
        resolveId(id) {
          return id.startsWith('../src/')
            ? join(sourceRoot, id.slice('../src/'.length))
            : null;
        },
      },
    ],
    onLog(level, log, handler) {
      if (log.code === 'UNRESOLVED_IMPORT') {
        throw new Error(`compare: ${log.message}`);
      }

      handler(level, log);
    },
  });

  try {
    const { output } = await built.generate({
      format: 'iife',
      name: 'labExports',
    });

    const [chunk] = output;
    const ids = new Set(chunk.moduleIds);
    const modules: ModuleStart[] = [];

    chunk.code.split('\n').forEach((text, line) => {
      const marker = /^\t\/\/ (\S+)$/u.exec(text)?.[1];
      const path = marker && resolve(marker);

      if (path && ids.has(path)) {
        const key = relative(sourceRoot, path);

        modules.push({
          line,
          key: key.startsWith('..') ? relative(ROOT, path) : key,
        });
      }
    });

    return { code: chunk.code, modules };
  } finally {
    await built.close();
  }
}

/** The key of the module whose code holds `line`, or `null` before the first one. */
function moduleAt(
  modules: readonly ModuleStart[],
  line: number,
): string | null {
  let found: string | null = null;

  for (const start of modules) {
    if (start.line > line) {
      break;
    }

    found = start.key;
  }

  return found;
}

// `/proc` from Node; nothing adds a CDP command or a page evaluation between
// a scenario's two `metrics()` calls.

/** One reading's record: its window, its renderer's main thread, the system, the gauge. */
type WindowRecord = {
  start: string;
  end: string;
  windowMs: number;
  rendererPid: number;
  renderers: number;
  thread: string;
  onCpuMs: number;
  waitMs: number;
  samples: number;
  shareCpu0to3: number;
  pressureUs: number;
  load1: number;
  gaugeBeforeMs: number;
  gaugeAfterMs: number;
  settled?: Readonly<{
    wait: boolean;
    placement: boolean;
    gauge: boolean;
    all: boolean;
  }>;
};

const SAMPLE_MS = 20;
const P_CPUS = new Set([0, 1, 2, 3]);
/** The gauge: the floor entries' loop body, 20 ms nominal at `ITERATIONS_PER_US`. */
const GAUGE_ITERATIONS = 20 * 1000 * 60;

/** The fields after a `/proc/<pid>/stat` name, so that index 0 is field 3. */
const statFields = (text: string): string[] =>
  text.slice(text.lastIndexOf(')') + 2).split(' ');

async function readOr(path: string): Promise<string | null> {
  try {
    return await readFile(path, 'utf8');
  } catch {
    return null;
  }
}

/** Every renderer process descended from this one. */
async function renderers(): Promise<number[]> {
  const parents = new Map<number, number>();
  const commands = new Map<number, string>();

  await Promise.all(
    (await readdir('/proc'))
      .filter((name) => /^\d+$/u.test(name))
      .map(async (name) => {
        const stat = await readOr(`/proc/${name}/stat`);

        if (stat) {
          parents.set(Number(name), Number(statFields(stat)[1]));
        }
      }),
  );

  const descendants: number[] = [];
  const isDescendant = (pid: number): boolean => {
    for (let at = parents.get(pid); at !== undefined && at > 1; at = parents.get(at)) {
      if (at === process.pid) {
        return true;
      }
    }

    return false;
  };

  for (const pid of parents.keys()) {
    if (isDescendant(pid)) {
      descendants.push(pid);
    }
  }

  await Promise.all(
    descendants.map(async (pid) => {
      commands.set(pid, (await readOr(`/proc/${pid}/cmdline`)) ?? '');
    }),
  );

  return descendants.filter((pid) =>
    commands.get(pid)!.includes('--type=renderer'),
  );
}

/** On-CPU and run-queue wait of a process's main thread, in ns. */
async function schedstat(
  pid: number,
): Promise<Readonly<{ onCpu: number; wait: number }> | null> {
  const text = await readOr(`/proc/${pid}/task/${pid}/schedstat`);

  if (!text) {
    return null;
  }

  const [onCpu, wait] = text.trim().split(' ').map(Number);

  return { onCpu: onCpu!, wait: wait! };
}

async function pressureTotal(): Promise<number> {
  const text = (await readOr('/proc/pressure/cpu'))!;

  return Number(/^some .*total=(\d+)$/mu.exec(text)![1]);
}

async function load1(): Promise<number> {
  return Number((await readFile('/proc/loadavg', 'utf8')).split(' ')[0]);
}

type Recording = Readonly<{
  stop(): Promise<
    Omit<WindowRecord, 'gaugeBeforeMs' | 'gaugeAfterMs' | 'settled'>
  >;
}>;

/**
 * Starts recording a window over `candidates`: schedstat at both ends, and the
 * CPU each main thread last ran on, every 20 ms. The page's renderer is the
 * candidate whose main thread has the most on-CPU time in the window.
 */
async function startRecording(candidates: readonly number[]): Promise<Recording> {
  const [load, pressure0, ...stats0] = await Promise.all([
    load1(),
    pressureTotal(),
    ...candidates.map(schedstat),
  ]);
  const start = new Date();
  const t0 = performance.now();
  const cpus = new Map<number, number[]>(candidates.map((pid) => [pid, []]));
  let inFlight: Promise<void> | null = null;
  const sample = (): void => {
    if (inFlight) {
      return;
    }

    inFlight = Promise.all(
      candidates.map(async (pid) => {
        const stat = await readOr(`/proc/${pid}/task/${pid}/stat`);

        if (stat) {
          cpus.get(pid)!.push(Number(statFields(stat)[36]));
        }
      }),
    ).then(() => {
      inFlight = null;
    });
  };
  const timer = setInterval(sample, SAMPLE_MS);

  return {
    async stop() {
      clearInterval(timer);
      await inFlight;

      const windowMs = performance.now() - t0;
      const end = new Date();
      const [pressure1, ...stats1] = await Promise.all([
        pressureTotal(),
        ...candidates.map(schedstat),
      ]);
      let best = -1;
      let onCpu = -1;
      let wait = 0;

      candidates.forEach((pid, i) => {
        const before = stats0[i] as Awaited<ReturnType<typeof schedstat>>;
        const after = stats1[i] as Awaited<ReturnType<typeof schedstat>>;

        if (before && after && after.onCpu - before.onCpu > onCpu) {
          best = pid;
          onCpu = after.onCpu - before.onCpu;
          wait = after.wait - before.wait;
        }
      });

      if (best < 0 || onCpu <= 0) {
        throw new Error(
          `recorder: no renderer main thread ran in the window, candidates ${JSON.stringify(candidates)}`,
        );
      }

      const placed = cpus.get(best)!;
      const comm = ((await readOr(`/proc/${best}/task/${best}/comm`)) ?? '').trim();

      return {
        start: start.toISOString(),
        end: end.toISOString(),
        windowMs,
        rendererPid: best,
        renderers: candidates.length,
        thread: comm,
        onCpuMs: onCpu / 1e6,
        waitMs: wait / 1e6,
        samples: placed.length,
        shareCpu0to3:
          placed.length === 0
            ? 0
            : placed.filter((cpu) => P_CPUS.has(cpu)).length / placed.length,
        pressureUs: (pressure1 as number) - (pressure0 as number),
        load1: load as number,
      };
    },
  };
}

/**
 * The floor entries' loop body, timed in the page, in ms. The entries store into
 * a module-level variable of the lab bundle, a slot of an enclosing function's
 * context, and V8 compiles that loop at about 60 iterations per µs; a local
 * variable compiles to a far faster loop. So `work` is captured by `body`.
 */
function fixedWork(iterations: number): number {
  let work = 0;
  const body = (): void => {
    for (let i = 0; i < iterations; i += 1) {
      work = Math.imul(work ^ i, 0x5bd1e995);
    }
  };
  const t0 = performance.now();

  body();

  const ms = performance.now() - t0;

  (globalThis as Record<string, unknown>)['__gauge'] = work;
  return ms;
}

/** The gauge's duration in the page, in ms, by its own evaluation. */
const gauge = (page: Page): Promise<number> =>
  page.evaluate(fixedWork, GAUGE_ITERATIONS);

/** The record of the last window a scenario's page completed, taken by `runScenarios`. */
let lastRecord: WindowRecord | null = null;
const records: Record<string, Record<string, WindowRecord[]>> = {};

async function openPage(browser: Browser, arm: Arm): Promise<PageContext> {
  const page = await browser.newPage({ viewport: { width: 900, height: 700 } });

  await page.setContent('<!doctype html><body style="margin:40px"></body>');
  await page.addScriptTag({ content: arm.code });
  page.on('pageerror', (error) => {
    console.log(`page error: ${error.message}`);
  });

  const cdp = await page.context().newCDPSession(page);

  await cdp.send('Performance.enable');

  const read = async (): Promise<Metrics> =>
    Object.fromEntries(
      (await cdp.send('Performance.getMetrics')).metrics.map(
        ({ name, value }) => [name, value],
      ),
    );
  // A scenario's first call opens its window and its second closes it. The
  // gauge runs before the first `getMetrics` and after the second, and the
  // recorder starts after the first returns and stops before the second.
  let recording: Recording | null = null;
  let gaugeBefore = 0;
  const metrics = async (): Promise<Metrics> => {
    if (!recording) {
      const candidates = await renderers();

      gaugeBefore = await gauge(page);

      const before = await read();

      recording = await startRecording(candidates);
      return before;
    }

    const window = await recording.stop();
    const after = await read();

    lastRecord = {
      ...window,
      gaugeBeforeMs: gaugeBefore,
      gaugeAfterMs: await gauge(page),
    };
    return after;
  };

  return { browser, arm, page, cdp, metrics };
}

const delta = (before: Metrics, after: Metrics, count: number): Reading => ({
  layouts: after['LayoutCount']! - before['LayoutCount']!,
  // Not judged. These say where a task-time difference went.
  layoutUs:
    ((after['LayoutDuration']! - before['LayoutDuration']!) * 1e6) / count,
  styleUs:
    ((after['RecalcStyleDuration']! - before['RecalcStyleDuration']!) * 1e6) /
    count,
  scriptUs:
    ((after['ScriptDuration']! - before['ScriptDuration']!) * 1e6) / count,
  taskUs: ((after['TaskDuration']! - before['TaskDuration']!) * 1e6) / count,
});

  cached: boolean;
  landed: boolean;
  refocus: boolean;

const mount = (page: Page, options: LabOptions): Promise<void> =>
  page.evaluate((opts) => {
    lab.sortable(opts);

async function grabAndCross(page: Page): Promise<void> {
  await page.mouse.move(X, Y0);
  await page.mouse.down();
  await page.mouse.move(X, Y0 + 12, { steps: 2 });
  await page.mouse.move(X, Y0 + 20);
  await page.waitForTimeout(50);
}

const movePhase = (i: number): number =>
  Y0 + 20 + Math.abs(((i / 60) % 2) - 1) * ROWS * 40;

interface Scenario {
  readonly id: string;
  /** The rule stated before running, in the handoff's terms. */
  readonly rule: 'exact-and-floor' | 'floor' | 'report' | 'retained' | 'bytes';
  readonly arms: 'sortable' | 'free-drag' | 'no-lift' | 'all';
  /** Runs only with arms built from this package's source, never drag2's or the prototype's. */
  readonly packageArmsOnly?: true;
  run(context: PageContext): Promise<Reading>;
}

function movesScenario(count: number): Scenario {
  return {
    id: `300 moves, one frame each, ${count} items`,
    rule: 'exact-and-floor',
    arms: 'sortable',
    async run({ page, metrics }) {
      await mount(page, { count, cached: true, landed: true, refocus: false });
      await grabAndCross(page);

      const before = await metrics();

      for (let i = 0; i < MOVES; i += 1) {
        await page.mouse.move(X, movePhase(i));
        await page.evaluate(() => new Promise(requestAnimationFrame));
      }

      const after = await metrics();

      await page.mouse.up();
      return delta(before, after, MOVES);
    },
  };
}

/**
 * The fields of Chrome's mouse, which is pointer 1 and which CDP's input drives, so
 * that an in-page event reaches the library as that pointer's input does.
 */
const MOUSE = {
  pointerId: 1,
  pointerType: 'mouse',
  isPrimary: true,
  bubbles: true,
  cancelable: true,
  composed: true,
} as const;

/**
 * The in-page mirror of `movesScenario`: the same press, crossing and positions.
 * Each move is dispatched to the list, where the root's pointer capture sends a
 * real one, with the button held. With `mounted` false, no behavior is mounted,
 * and the row measures the driver's own share.
 */
function inPageMovesScenario(count: number, mounted: boolean): Scenario {
  return {
    id: mounted
      ? `in the page: 300 moves, one frame each, ${count} items`
      : `in the page: driver alone, 300 moves, ${count} items, no behavior`,
    rule: mounted ? 'exact-and-floor' : 'report',
    arms: 'sortable',
    packageArmsOnly: true,
    async run({ page, metrics }) {
      if (mounted) {
        await mount(page, {
          count,
          cached: true,
          landed: true,
          refocus: false,
        });
      } else {
        await page.evaluate((rows) => {
          lab.list(rows);
        }, count);
      }

      await grabAndCross(page);

      const before = await metrics();

      await page.evaluate(
        async ({ mouse, x, ys }) => {
          const root = document.querySelector('#i0')!.parentElement!;

          for (const y of ys) {
            root.dispatchEvent(
              new PointerEvent('pointermove', {
                ...mouse,
                button: -1,
                buttons: 1,
                clientX: x,
                clientY: y,
              }),
            );
            await new Promise(requestAnimationFrame);
          }
        },
        {
          mouse: MOUSE,
          x: X,
          ys: Array.from({ length: MOVES }, (_, i) => movePhase(i)),
        },
      );

      const after = await metrics();

      await page.mouse.up();
      return delta(before, after, MOVES);
    },
  };
}

type HeapNode = Readonly<{
  callFrame: Readonly<{
    functionName: string;
    scriptId: string;
    lineNumber: number;
  }>;
  selfSize: number;
  children: readonly HeapNode[];
}>;

/**
 * Bytes per move by function, with no harness `evaluate` during the run. A frame
 * in the lab bundle is keyed `<module>:<function>`, such as
 * `kernel/session.ts:#move`.
 */
async function samplingProfile(
  context: PageContext,
  moves: (i: number) => Promise<void>,
): Promise<Reading> {
  const { cdp, arm } = context;

  // A sampled frame of an inline script has no url, so the lab bundle's script is
  // found from one of its functions, before sampling starts.
  const { result } = await cdp.send('Runtime.evaluate', {
    expression:
      "Object.values(lab).find((value) => typeof value === 'function')",
  });
  const { internalProperties } = await cdp.send('Runtime.getProperties', {
    objectId: result.objectId!,
    ownProperties: true,
  });
  const labScript = (
    internalProperties?.find(({ name }) => name === '[[FunctionLocation]]')
      ?.value?.value as { scriptId: string } | undefined
  )?.scriptId;

  await cdp.send('HeapProfiler.enable');
  await cdp.send('HeapProfiler.startSampling', {
    samplingInterval: 64,
    includeObjectsCollectedByMajorGC: true,
    includeObjectsCollectedByMinorGC: true,
  });

  for (let i = 0; i < MOVES; i += 1) {
    await moves(i);
  }

  const { profile } = await cdp.send('HeapProfiler.stopSampling');
  const bytes = new Map<string, number>();
  // A built-in such as `push` has no script, and its bytes belong to its caller.
  // Counted under its own name, the kernel's allocation would escape the rule.
  const walk = (node: HeapNode, caller: string): void => {
    const { scriptId, lineNumber, functionName } = node.callFrame;
    const module =
      scriptId === labScript ? moduleAt(arm.modules, lineNumber) : null;
    const name =
      scriptId === '0' && caller
        ? caller
        : `${module ? `${module}:` : ''}${functionName || '(anonymous)'}`;

    bytes.set(name, (bytes.get(name) ?? 0) + node.selfSize);
    node.children.forEach((child) => {
      walk(child, name);
    });
  };

  walk(profile.head, '');
  return Object.fromEntries(
    [...bytes]
      .map(([name, size]) => [name, Math.round(size / MOVES)] as const)
      .filter(([, size]) => size > 0),
  );
}

/**
 * Mounts a 20-item sortable, runs `operations` landed cycles, destroys it and reads
 * the retained JavaScript objects after forced GC, with `JSHeapUsedSize` beside
 * them for continuity.
 */
async function operateAndDestroy(
  { page, cdp, metrics }: PageContext,
  operations: number,
): Promise<
  Readonly<{
    objects: number;
    heap: number;
    byName: Readonly<Record<string, number>>;
  }>
> {
  await mount(page, { count: 20, cached: true, landed: true, refocus: false });

  for (let i = 0; i < operations; i += 1) {
    await page.mouse.move(X, Y0);
    await page.mouse.down();
    await page.mouse.move(X, Y0 + 12, { steps: 2 });
    await page.mouse.move(X, Y0 + 45);
    await page.evaluate(() => new Promise(requestAnimationFrame));
    await page.mouse.up();
    await page.waitForTimeout(ANIMATION_WAIT);
  }

  await page.evaluate(async () => {
    await lab.destroy();
  });
  await page.waitForTimeout(300);
  await page.mouse.move(800, 600);
  await page.waitForTimeout(50);

  const { bytes, byName } = await retainedObjects(cdp);

  return {
    objects: bytes,
    heap: (await metrics())['JSHeapUsedSize']!,
    byName,
  };
}

/** The middle value, or the mean of the two middle values. */
const median = (values: readonly number[]): number => {
  const sorted = values.toSorted((x, y) => x - y);
  const middle = Math.floor(sorted.length / 2);

  return sorted.length % 2 === 1
    ? sorted[middle]!
    : (sorted[middle - 1]! + sorted[middle]!) / 2;
};

/** A bytes key in a kernel module: one under `kernel/`, or `draggable.ts`. */
const isKernelKey = (key: string): boolean =>
  key.startsWith('kernel/') || key.startsWith('draggable.ts:');

/** Whether a bytes key names its module, as `samplingProfile` keys a lab frame. */
const hasModule = (key: string): boolean => /^[^:]+\.[cm]?[jt]sx?:/u.test(key);

type ArmReadings = Readonly<Record<string, readonly Reading[]>>;
export type Verdict = Readonly<{
  against: string;
  measure: string;
  result: string;
  /** Per round, production minus the before arm. */
  differences?: readonly number[];
  /** Per round, the A/A difference the bound comes from. */
  aa?: readonly number[];
  higher?: number;
  lower?: number;
  median?: number;
  bound?: number;
}>;

/** Rounds in which a timing difference must have one sign: 6 of 7. */
const timingSigns = (rounds: number): number => Math.ceil((rounds * 6) / 7);

/**
 * One paired comparison. `differences` and `aa` are per round. A verdict needs the
 * sign in `required` rounds **and** a median beyond the largest absolute A/A
 * difference. Anything else is no difference shown, which is not equality.
 */
function pairedVerdict(
  against: string,
  measure: string,
  differences: readonly number[],
  aa: readonly number[],
  required: number,
  words: readonly [more: string, less: string],
  unit: string,
): Verdict {
  const higher = differences.filter((value) => value > 0).length;
  const lower = differences.filter((value) => value < 0).length;
  const middle = median(differences);
  const bound = Math.max(...aa.map((value) => Math.abs(value)));
  const label =
    higher >= required && middle > bound
      ? `FLAG ${words[0]}`
      : lower >= required && -middle > bound
        ? words[1]
        : 'no difference shown';

  return {
    against,
    measure,
    result: `${label}: higher in ${higher} of ${differences.length}, median ${Math.round(middle)} ${unit}, A/A bound ${Math.round(bound)} ${unit}`,
    differences: differences.map((value) => Math.round(value)),
    aa: aa.map((value) => Math.round(value)),
    higher,
    lower,
    median: Math.round(middle),
    bound: Math.round(bound),
  };
}

/**
 * The rules, fixed before running (remediation handoff, step 6):
 *
 * - **Layouts, exact:** production's median at or below each before arm's.
 * - **Script and task, paired:** per round, production minus the before arm, and
 *   production minus control. Slower when higher in at least 6 of 7 rounds and the
 *   median difference exceeds the largest absolute A/A difference. Faster is the
 *   mirror.
 * - **Retained objects, over 3 rounds:** an arm's growth is its 200-operation sum
 *   minus its 0-operation sum in the same round. Per round, production's growth
 *   minus the before arm's, and control's minus production's. Retains more when
 *   positive in every round and the median exceeds the largest absolute A/A
 *   difference, which may be 0 B. Retains less is the mirror.
 * - **Bytes, production only:** a function in a kernel module, under `kernel/` or
 *   `draggable.ts`, flags at 16 B per move or more. No other module is judged.
 *
 * Readings are paired by index, which is the round.
 */
export function judge(
  rule: Scenario['rule'],
  arms: ArmReadings,
  after = 'production',
  control = 'control',
  against: readonly string[] = ['drag2', 'prototype'],
): readonly Verdict[] {
  const verdicts: Verdict[] = [];
  const of = (arm: string, key: string): number[] =>
    (arms[arm] ?? []).map((reading) => reading[key] ?? 0);

  if (rule === 'bytes') {
    for (const reading of arms[after] ?? []) {
      for (const [name, bytes] of Object.entries(reading)) {
        if (isKernelKey(name) && bytes >= 16) {
          verdicts.push({
            against: 'rule',
            measure: name,
            result: `FLAG ${bytes} B per move`,
          });
        }
      }
    }

    return verdicts;
  }

  const minus = (a: readonly number[], b: readonly number[]): number[] =>
    a.map((value, round) => value - b[round]!);

  for (const before of against) {
    if (!arms[before]) {
      continue;
    }

    if (rule === 'retained') {
      const growth = of(after, 'growth');

      verdicts.push(
        pairedVerdict(
          before,
          'retained objects',
          minus(growth, of(before, 'growth')),
          minus(of(control, 'growth'), growth),
          growth.length,
          ['retains more', 'retains less'],
          'B',
        ),
      );
      continue;
    }

    if (rule === 'exact-and-floor') {
      const layouts = median(of(after, 'layouts'));
      const beforeLayouts = median(of(before, 'layouts'));

      verdicts.push({
        against: before,
        measure: 'layouts',
        result: `${layouts > beforeLayouts ? 'FLAG more' : 'at or below'}: ${layouts} against ${beforeLayouts}`,
      });
    }

    for (const key of ['scriptUs', 'taskUs']) {
      const timed = of(after, key);

      verdicts.push(
        pairedVerdict(
          before,
          key,
          minus(timed, of(before, key)),
          minus(timed, of(control, key)),
          timingSigns(timed.length),
          ['slower', 'faster'],
          'µs',
        ),
      );
    }
  }

  return verdicts;
}

function applicableArms(
  scenario: Scenario,
  all: Readonly<Record<string, Arm>>,
  noLift: Arm,
): Record<string, Arm> {
  if (scenario.arms === 'no-lift') {
    return { production: noLift, control: noLift };
  }

  return { ...all };
}

/** Rounds for a rule: the retained rule is fixed at 3, everything else at `rounds`. */
const roundsFor = (scenario: Scenario, rounds: number): number =>
  scenario.rule === 'retained' ? Math.min(rounds, 3) : rounds;

async function runScenarios(
  browser: Browser,
  scenarios: readonly Scenario[],
  arms: Readonly<Record<string, Arm>>,
  noLift: Arm,
  rounds: number,
): Promise<Record<string, Record<string, Reading[]>>> {
  const raw: Record<string, Record<string, Reading[]>> = {};

  for (let round = 0; round < rounds; round += 1) {
    for (const scenario of scenarios) {
      if (round >= roundsFor(scenario, rounds)) {
        continue;
      }

      const armsHere = applicableArms(scenario, arms, noLift);
      const names = Object.keys(armsHere);
      // A different arm goes first each round, so no arm always runs on a cold cache.
      const order = names.map((_, i) => names[(i + round) % names.length]!);

      for (const arm of order) {
        const context = await openPage(browser, armsHere[arm]!);

        try {
          lastRecord = null;

          const reading = await scenario.run(context);
          const record = lastRecord as WindowRecord | null;

          if (!record) {
            throw new Error(`recorder: no window recorded: ${scenario.id}`);
          }

          ((raw[scenario.id] ??= {})[arm] ??= []).push(reading);
          ((records[scenario.id] ??= {})[arm] ??= []).push(record);
          console.log(
            `reading: round ${round + 1}, ${arm}, ${scenario.id}: ${JSON.stringify({ scriptUs: reading['scriptUs'], taskUs: reading['taskUs'], layouts: reading['layouts'], ...record })}`,
          );
        } finally {
          await context.page.close();
        }
      }

      console.log(`round ${round + 1}/${rounds}: ${scenario.id}`);
    }
  }

  return raw;
}

function summarize(
  raw: Readonly<Record<string, ArmReadings>>,
  rules: Readonly<Record<string, Scenario['rule']>>,
  against: readonly string[],
) {
  return Object.keys(rules)
    .filter((id) => raw[id])
    .map((id) => {
      const arms = raw[id]!;
      const rule = rules[id]!;
      const medians = Object.fromEntries(
        Object.entries(arms).map(([arm, readings]) => [
          arm,
          Object.fromEntries(
            [...new Set(readings.flatMap((reading) => Object.keys(reading)))]
              .map((key) => {
                const values = readings.map((reading) => reading[key] ?? 0);

                return [
                  key,
                  {
                    median: Math.round(median(values)),
                    min: Math.round(Math.min(...values)),
                    max: Math.round(Math.max(...values)),
                  },
                ];
              })
              .filter(
                ([, value]) =>
                  rule !== 'bytes' || (value as { median: number }).median > 0,
              ),
          ),
        ]),
      );

      return {
        scenario: id,
        rule,
        medians,
        verdicts: judge(rule, arms, 'production', 'control', against),
      };
    });
}

/** The selected entries: those whose name contains one of `only`'s `|`-separated texts. */
const select = <Entry extends Readonly<{ name: string }>>(
  entries: readonly Entry[],
  only: string | undefined,
): Entry[] =>
  entries.filter(
    (each) => !only || only.split('|').some((part) => each.name.includes(part)),
  );

/**
 * Production's lab bundle from a copy of `src` with `injection` applied. Throws
 * unless `find` occurs exactly once and the patch changes the bundle. The
 * comparison is with the copy's own bundle before the patch, because a bundle of
 * another directory differs by its module paths.
 */
async function applyInjection(injection: Injection): Promise<Arm> {
  const { name, file, find, replace, declaration = '' } = injection;
  const directory = await mkdtemp(join(tmpdir(), 'drag-inject-'));

  try {
    const source = join(directory, 'src');

    await cp(join(ROOT, 'src'), source, { recursive: true });

    const path = join(source, file);
    const text = await readFile(path, 'utf8');

    if (text.split(find).length !== 2) {
      throw new Error(`injection does not apply in ${file}: ${find}`);
    }

    const lab = join(ROOT, 'bench/lab.ts');
    const unpatched = (await bundle(lab, source)).code;

    await writeFile(path, text.replace(find, () => replace) + declaration);

    const patched = await bundle(lab, source);

    if (patched.code === unpatched) {
      throw new Error(
        `injection does not apply: ${name} leaves the bundle unchanged`,
      );
    }

    return { ...patched, restoresFocus: true };
  } finally {
    await rm(directory, { force: true, recursive: true });
  }
}

/** Every entry applied, in order, before any browser starts. The first failure throws. */
async function applyAll<Entry extends Injection>(
  entries: readonly Entry[],
): Promise<Array<Readonly<{ entry: Entry; arm: Arm }>>> {
  const applied: Array<Readonly<{ entry: Entry; arm: Arm }>> = [];

  for (const entry of entries) {
    applied.push({ entry, arm: await applyInjection(entry) });
    console.log(
      `applies: ${entry.name}${'expect' in entry && entry.expect === 'quiet' ? ', expected to stay quiet' : ''}`,
    );
  }

  return applied;
}

/**
 * Each injection runs as `production` against unpatched production, with a second
 * copy of the injected bundle as its control, and must give the verdict it
 * expects. A quiet injection's result is counted only when an injection expected
 * to flag on its row ran in the same run. Then the control runs
 * against itself on every validation row: three copies of one bundle, none of which
 * may flag.
 */
async function prove(
  browser: Browser,
  injections: ReadonlyArray<
    Readonly<{ entry: (typeof INJECTIONS)[number]; arm: Arm }>
  >,
  production: Arm,
  noLift: Arm,
  rounds: number,
  out?: string,
): Promise<boolean> {
  const results: unknown[] = [];
  let proven = true;

  for (const { entry: injection, arm: injected } of injections) {
    const scenario = SCENARIOS.find((each) => each.id === injection.scenario)!;
    const scenarioRounds = scenario.rule === 'bytes' ? 3 : rounds;
    const raw = await runScenarios(
      browser,
      [scenario],
      { production: injected, base: production, control: injected },
      noLift,
      scenarioRounds,
    );
    const arms = raw[scenario.id]!;
    const verdicts = judge(scenario.rule, arms, 'production', 'control', [
      'base',
    ]);
    const { expect } = injection;
    const quiet = expect === 'quiet';
    const flagged = verdicts.some((verdict) =>
      quiet ? verdict.result.startsWith('FLAG') : expect.test(verdict.result),
    );
    const met = quiet ? !flagged : flagged;
    const counted =
      !quiet ||
      injections.some(
        ({ entry }) =>
          entry.scenario === injection.scenario && entry.expect !== 'quiet',
      );
    let line: string;

    if (quiet) {
      line = met
        ? `quiet as expected${counted ? '' : ', not counted: no injection expected to flag ran on its row'}`
        : 'FLAGGED, expected quiet';
    } else {
      line = met ? 'flagged' : 'NOT FLAGGED, expected to flag';
    }

    proven &&= met;
    results.push({
      injection: injection.name,
      scenario: scenario.id,
      expect: quiet ? 'quiet' : 'flag',
      flagged,
      met,
      counted,
      verdicts,
      raw: arms,
    });
    console.log(
      `${line}: ${injection.name} ${JSON.stringify(verdicts.map((verdict) => verdict.result))}`,
    );
  }

  const scenarios = SCENARIOS.filter((each) =>
    VALIDATION_ROWS.includes(each.id),
  );
  const raw = await runScenarios(
    browser,
    scenarios,
    { production, base: production, control: production },
    noLift,
    rounds,
  );

  for (const scenario of scenarios) {
    const verdicts = judge(
      scenario.rule,
      raw[scenario.id]!,
      'production',
      'control',
      ['base'],
    );
    const flagged = verdicts.filter((verdict) =>
      verdict.result.startsWith('FLAG'),
    );

    proven &&= flagged.length === 0;
    results.push({
      aa: scenario.id,
      flagged: flagged.length > 0,
      verdicts,
      raw: raw[scenario.id],
    });
    console.log(
      `${flagged.length === 0 ? 'quiet' : 'A/A FLAGGED'}: ${scenario.id} ${JSON.stringify(verdicts.map((verdict) => verdict.result))}`,
    );
  }

  if (out) {
    await writeFile(out, `${JSON.stringify(results, null, 1)}\n`);
  }

  return proven;
}

type ProfileNode = Readonly<{
  id: number;
  callFrame: Readonly<{
    functionName: string;
    url: string;
    lineNumber: number;
  }>;
}>;

/**
 * One run of `scenario` under the CPU profiler, as self time in µs per function.
 * The whole run is profiled, including the page's setup.
 */
async function profileOnce(
  browser: Browser,
  scenario: Scenario,
  arm: Arm,
): Promise<Record<string, number>> {
  const context = await openPage(browser, arm);

  try {
    const { cdp } = context;

    await cdp.send('Profiler.enable');
    await cdp.send('Profiler.setSamplingInterval', { interval: 50 });
    await cdp.send('Profiler.start');
    await scenario.run(context);

    const { profile } = (await cdp.send('Profiler.stop')) as unknown as {
      profile: {
        nodes: readonly ProfileNode[];
        samples: readonly number[];
        timeDeltas: readonly number[];
      };
    };
    const self = new Map<number, number>();

    profile.samples.forEach((id, i) => {
      self.set(id, (self.get(id) ?? 0) + (profile.timeDeltas[i] ?? 0));
    });

    const byName: Record<string, number> = {};

    for (const node of profile.nodes) {
      const time = self.get(node.id);

      if (time) {
        const { functionName, lineNumber } = node.callFrame;
        const name = functionName || `(anonymous, line ${lineNumber})`;

        byName[name] = (byName[name] ?? 0) + time;
      }
    }

    return byName;
  } finally {
    await context.page.close();
  }
}

/** Growth in retained self bytes per constructor, 200 operations against 0. */
async function attributeRetained(
  browser: Browser,
  arm: Arm,
): Promise<Record<string, number>> {
  const run = async (operations: number) => {
    const context = await openPage(browser, arm);

    try {
      return (await operateAndDestroy(context, operations)).byName;
    } finally {
      await context.page.close();
    }
  };
  const zero = await run(0);
  const operated = await run(OPERATIONS);
  const growth: Record<string, number> = {};

  for (const name of new Set([
    ...Object.keys(zero),
    ...Object.keys(operated),
  ])) {
    const change = (operated[name] ?? 0) - (zero[name] ?? 0);

    if (change !== 0) {
      growth[name] = change;
    }
  }

  return growth;
}

/** The comparison itself, printed and written to `--out`. */
async function compareArms(
  browser: Browser,
  arms: Readonly<Record<string, Arm>>,
  noLift: Arm,
): Promise<void> {
  const scenarios = SCENARIOS.filter(
    (each) =>
      (values.rows !== 'validation' || VALIDATION_ROWS.includes(each.id)) &&
      (!values.only || each.id.includes(values.only)) &&
      (values.before !== undefined || !each.packageArmsOnly),
  );
  const raw = await runScenarios(
    browser,
    scenarios,
    arms,
    noLift,
    Number(values.rounds),
  );
  const against = Object.keys(arms).filter(
    (arm) => arm !== 'production' && arm !== 'control',
  );
  const summary = summarize(
    raw,
    Object.fromEntries(scenarios.map((each) => [each.id, each.rule])),
    against,
  );

  if (values.out) {
    await writeFile(
      values.out,
      `${JSON.stringify({ chrome: browser.version(), rounds: Number(values.rounds), moves: MOVES, arms: Object.keys(arms), before: values.before ?? null, summary, raw }, null, 1)}\n`,
    );
  }

  print(summary);
}

/** Production's `src` at `revision`, extracted under `scratch`. */
async function archivedSource(
  scratch: string,
  revision: string,
  path: string,
): Promise<string> {
  const archive = join(scratch, `${revision}.tar`);
  const target = join(scratch, revision);

  await execFile('git', ['archive', revision, '--output', archive, path], {
    cwd: join(ROOT, '../..'),
  });
  await execFile('mkdir', ['-p', target]);
  await execFile('tar', ['-xf', archive, '-C', target]);
  return join(target, path);
}

/** `--profile` or `--attribute`: one diagnostic reading per arm, printed and written. */
async function diagnose(
  browser: Browser,
  arms: Readonly<Record<string, Arm>>,
): Promise<void> {
  const scenario = SCENARIOS.find((each) =>
    each.id.includes(values.only ?? '\u0000'),
  );
  const result: Record<string, Record<string, number>> = {};

  for (const [name, arm] of Object.entries(arms)) {
    result[name] = values.attribute
      ? await attributeRetained(browser, arm)
      : await profileOnce(browser, scenario!, arm);
    console.log(
      `${name}: ${JSON.stringify(
        Object.entries(result[name])
          .toSorted((a, b) => Math.abs(b[1]) - Math.abs(a[1]))
          .slice(0, 25),
      )}`,
    );
  }

  if (values.out) {
    await writeFile(values.out, `${JSON.stringify(result, null, 1)}\n`);
  }
}

/**
 * Each floor runs as a `--prove` injection does, and its verdicts and readings are
 * recorded whatever they are. For the bytes row, the difference per round is the
 * injected arm's kernel bytes per move minus base's.
 */
async function floors(
  browser: Browser,
  entries: ReadonlyArray<
    Readonly<{ entry: (typeof FLOORS)[number]; arm: Arm }>
  >,
  production: Arm,
  noLift: Arm,
  rounds: number,
  out?: string,
): Promise<void> {
  const results: unknown[] = [];

  for (const { entry, arm: injected } of entries) {
    const scenario = SCENARIOS.find((each) => each.id === entry.scenario)!;
    const raw = await runScenarios(
      browser,
      [scenario],
      { production: injected, base: production, control: injected },
      noLift,
      scenario.rule === 'bytes' ? 3 : rounds,
    );
    const arms = raw[scenario.id]!;
    const verdicts = judge(scenario.rule, arms, 'production', 'control', [
      'base',
    ]);
    const kernel = (reading: Reading): number =>
      Object.entries(reading)
        .filter(([key]) => isKernelKey(key))
        .reduce((sum, [, bytes]) => sum + bytes, 0);
    const kernelDifferences =
      scenario.rule === 'bytes'
        ? (arms['production'] ?? []).map(
            (reading, round) => kernel(reading) - kernel(arms['base']![round]!),
          )
        : undefined;

    results.push({
      floor: entry.name,
      scenario: scenario.id,
      flagged: verdicts.some((verdict) => verdict.result.startsWith('FLAG')),
      verdicts,
      ...(kernelDifferences && {
        kernelBytes: {
          differences: kernelDifferences,
          higher: kernelDifferences.filter((value) => value > 0).length,
          median: median(kernelDifferences),
        },
      }),
      raw: arms,
    });
    console.log(
      `floor: ${entry.name} ${JSON.stringify(verdicts.map((verdict) => verdict.result))}${kernelDifferences ? ` kernel bytes per move, injected minus base: ${JSON.stringify(kernelDifferences)}` : ''}`,
    );
  }

  if (out) {
    await writeFile(out, `${JSON.stringify(results, null, 1)}\n`);
  }
}

/** Every mode that launches a browser. */
async function run(
  injections: ReadonlyArray<
    Readonly<{ entry: (typeof INJECTIONS)[number]; arm: Arm }>
  >,
  floorEntries: ReadonlyArray<
    Readonly<{ entry: (typeof FLOORS)[number]; arm: Arm }>
  >,
): Promise<void> {
  const scratch = await mkdtemp(join(tmpdir(), 'drag-compare-'));

  try {
    const lab = join(ROOT, 'bench/lab.ts');
    const production: Arm = {
      ...(await bundle(lab, join(ROOT, 'src'))),
      restoresFocus: true,
    };
    const noLift: Arm = {
      ...(await bundle(join(ROOT, 'bench/lab-no-lift.ts'), join(ROOT, 'src'))),
      restoresFocus: true,
    };
    const browser = await chromium.launch({
      executablePath:
        process.env['CHROME_EXECUTABLE'] ?? '/usr/local/bin/chrome',
    });

    try {
      if (values.prove) {
        process.exitCode = (await prove(
          browser,
          injections,
          production,
          noLift,
          Number(values.rounds),
          values.out,
        ))
          ? 0
          : 1;
      } else if (values.floors) {
        await floors(
          browser,
          floorEntries,
          production,
          noLift,
          Number(values.rounds),
          values.out,
        );
      } else if (values.profile || values.attribute) {
        await diagnose(browser, {
          ...(await beforeArms(scratch, lab)),
          production,
        });
      } else if (values.before) {
        const before: Arm = {
          ...(await bundle(
            lab,
            await archivedSource(scratch, values.before, 'packages/drag/src'),
          )),
          restoresFocus: true,
        };

        await compareArms(
          browser,
          { before, production, control: production },
          noLift,
        );
      } else {
        const { drag2, prototype } = await beforeArms(scratch, lab);

        await compareArms(
          browser,
          { drag2, prototype, production, control: production },
          noLift,
        );
      }
    } finally {
      await browser.close();
    }
  } finally {
    await rm(scratch, { force: true, recursive: true });
  }
}

// Probe (DRD-44): the main block. The two in-page gate injections, each as
// `--prove` runs it, then the A/A of the CDP 20-item move row with three
// production arms, at `--rounds` rounds. `--dry` applies the injections and
// launches no browser. `--self-check` records one 300 ms loop in one page.

const GATES = [
  '30 µs of script per move, in the page',
  '300 µs of script per cycle, in the page',
];
const AA_ROW = '300 moves, one frame each, 20 items';

async function machine(): Promise<Record<string, string>> {
  const run = async (file: string, args: string[] = []): Promise<string> =>
    (await execFile(file, args)).stdout.trim();

  return {
    time: new Date().toISOString(),
    nproc: await run('nproc'),
    node: process.version,
    uptime: await run('uptime'),
    cpuModel: (await run('lscpu'))
      .split('\n')
      .find((line) => line.startsWith('Model name:'))!
      .replace(/^Model name:\s*/u, ''),
    cpuCore: (await readFile('/sys/devices/cpu_core/cpus', 'utf8')).trim(),
    cpuAtom: (await readFile('/sys/devices/cpu_atom/cpus', 'utf8')).trim(),
    affinity: await run('taskset', ['-cp', String(process.pid)]),
    revision: await run('git', ['rev-parse', 'HEAD']),
    gitStatus: await run('git', ['status', '--porcelain']),
  };
}
