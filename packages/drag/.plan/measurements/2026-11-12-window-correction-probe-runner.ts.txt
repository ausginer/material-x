/**
 * The xy() rebuild probe (DRD-50 property 2): one Chromium process, one page, three
 * arms per case, 7 rounds in rotated order, judged by compare.ts's paired rule.
 *
 * usage: node probe.ts <output directory>
 */
import { createRequire } from 'node:module';
import { writeFile } from 'node:fs/promises';
import { dirname, join } from 'node:path';
import { fileURLToPath, pathToFileURL } from 'node:url';

const PACKAGE = '/workspaces/material-x/packages/drag/package.json';
const BOX_QUAD = '/workspaces/material-x/packages/box-quad/src/index.ts';
const here = dirname(fileURLToPath(import.meta.url));
const require = createRequire(PACKAGE);
const { rolldown } = (await import(
  pathToFileURL(require.resolve('rolldown')).href
)) as typeof import('rolldown');
// CommonJS: the namespace's `default` is the module's exports.
const { chromium } = (
  (await import(pathToFileURL(require.resolve('playwright')).href)) as {
    default: typeof import('playwright');
  }
).default;

const ROUNDS = 7;
const ARMS = ['before', 'production', 'control'] as const;
const CASES = [
  { name: 'no sink', withSink: false },
  { name: 'layoutAnimation, nothing in flight', withSink: true },
] as const;
/** A batch lasts at least this long; calibration aims at three times it. */
const MINIMUM_BATCH_MS = 100;

const median = (values: readonly number[]): number => {
  const sorted = values.toSorted((x, y) => x - y);
  const middle = Math.floor(sorted.length / 2);

  return sorted.length % 2 === 1
    ? sorted[middle]!
    : (sorted[middle - 1]! + sorted[middle]!) / 2;
};

/** compare.ts's `pairedVerdict` rule, with `timingSigns(7)` = 6, unrounded. */
function verdict(differences: readonly number[], aa: readonly number[]) {
  const required = Math.ceil((differences.length * 6) / 7);
  const higher = differences.filter((value) => value > 0).length;
  const lower = differences.filter((value) => value < 0).length;
  const middle = median(differences);
  const bound = Math.max(...aa.map((value) => Math.abs(value)));
  const label =
    higher >= required && middle > bound
      ? 'slower'
      : lower >= required && -middle > bound
        ? 'faster'
        : 'no difference shown';

  return { label, higher, lower, required, median: middle, bound };
}

const output = process.argv[2]!;
const built = await rolldown({
  input: join(here, 'page.ts'),
  platform: 'browser',
  resolve: { alias: { '@ydinjs/box-quad': BOX_QUAD } },
  transform: { define: { __DEV__: 'false' } },
});
const { output: chunks } = await built.generate({ format: 'iife' });
const code = chunks[0].code;

await built.close();
await writeFile(join(output, 'bundle.js.txt'), code);

const browser = await chromium.launch({
  executablePath: process.env['CHROME_EXECUTABLE'] ?? '/usr/local/bin/chrome',
});
const result: Record<string, unknown> = {
  browser: browser.version(),
  rounds: ROUNDS,
};

try {
  const page = await browser.newPage();

  await page.setContent('<!doctype html><html><body style="margin:0"></body></html>');
  await page.addScriptTag({ content: code });

  const layout = await page.evaluate(() => (globalThis as any).probe.setup());

  // Two frames, so the page has rendered the laid-out field before any timing.
  await page.evaluate(
    () =>
      new Promise((resolve) => {
        requestAnimationFrame(() => requestAnimationFrame(resolve));
      }),
  );
  result['layout'] = layout;
  console.log(`field: ${JSON.stringify(layout)}, ${browser.version()}`);

  const cases: Record<string, unknown>[] = [];

  for (const { name, withSink } of CASES) {
    const prepared = await page.evaluate(
      (sink) => (globalThis as any).probe.prepare(sink),
      withSink,
    );
    const size: number = await page.evaluate(
      (ms) => (globalThis as any).probe.calibrate(ms),
      MINIMUM_BATCH_MS * 3,
    );
    const rounds: Record<string, { us: number; ms: number }>[] = [];

    for (let r = 0; r < ROUNDS; r += 1) {
      const order = ARMS.map((_, i) => ARMS[(i + r) % ARMS.length]!);
      const readings = await page.evaluate(
        ([o, s]) => (globalThis as any).probe.round(o, s),
        [order, size] as const,
      );

      rounds.push(readings);
      console.log(
        `${name} round ${r + 1} order ${order.join(',')}: ${ARMS.map((arm) => `${arm} ${readings[arm]!.us.toFixed(2)} µs (${readings[arm]!.ms.toFixed(1)} ms)`).join(', ')}`,
      );
    }

    const of = (arm: string): number[] => rounds.map((round) => round[arm]!.us);
    const differences = of('production').map((value, i) => value - of('before')[i]!);
    const aa = of('control').map((value, i) => value - of('production')[i]!);
    const judged = verdict(differences, aa);
    const shortest = Math.min(...rounds.flatMap((round) => ARMS.map((arm) => round[arm]!.ms)));

    cases.push({
      name,
      slots: prepared.count,
      armsPackTheSameValues: prepared.same,
      batchSize: size,
      warmUp: `one discarded batch of ${size} rebuilds per arm per round`,
      shortestTimedBatchMs: shortest,
      everyTimedBatchAtLeastMinimum: shortest >= MINIMUM_BATCH_MS,
      rounds,
      differences,
      aa,
      beforeMedianUs: median(of('before')),
      productionMedianUs: median(of('production')),
      controlMedianUs: median(of('control')),
      verdict: judged,
    });
    console.log(
      `${name}: ${judged.label}: higher in ${judged.higher} of ${ROUNDS}, lower in ${judged.lower}, median ${judged.median.toFixed(3)} µs, A/A bound ${judged.bound.toFixed(3)} µs, before median ${median(of('before')).toFixed(2)} µs, batch ${size} rebuilds, shortest timed batch ${shortest.toFixed(1)} ms, arms pack the same values: ${prepared.same}`,
    );
  }

  result['cases'] = cases;

  if (cases.some((each) => !each['everyTimedBatchAtLeastMinimum'])) {
    console.log(`a timed batch lasted less than ${MINIMUM_BATCH_MS} ms`);
    process.exitCode = 1;
  }
} finally {
  await browser.close();
}

await writeFile(join(output, 'probe.json'), `${JSON.stringify(result, null, 2)}\n`);
