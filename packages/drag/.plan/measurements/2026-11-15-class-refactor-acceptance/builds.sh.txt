# with node_modules, packages/drag/node_modules and packages/drag2 linked to the main checkout,
# so both trees read the main checkout's box-quad build.

    | grep -v -E '\.config\.|eslint' | sort | xargs sha256sum > $S/h-$r.$ext.txt; done); done
