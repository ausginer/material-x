	/**
	* The slots stored before slot 0: the scratch slot, then the placeholder. Slot `s`
	* is at `(s + LEADING) * STRIDE`.
	*/
	const LEADING = 2;
	/**
	* One slack value, sized to the precision at which geometry is reported. Gecko
	* reports a rect in 1/60 px, so a settled reading is off by up to 1/120 px. A
	* predicted slot is a reading plus the difference of two readings, and the rebuild
	* carries its own, so the bound is 1/30 px. The slack is the next power of two above
	* it, far below the smallest disagreement a broken rule produces, which is a row.
	*/
	const EQUIVALENCE_SLACK = 1 / 16;
	/**
	* The hot path: each axis rule reads the slot range it scans once per resolution,
	* then one number per candidate through `at`.
	*/
	var RectIndex = class {
		/**
		* The scratch slot, the placeholder, then the packed slots. It grows only by
		* appending, so it never has a hole, and it never shrinks: it holds the largest
		* collection the index measured, and is released with the index.
		*/
		#values = [];
		/** Destination-ordered elements, parallel to the packed slots. */
		#items = [];
		/**
		* `LEFT` or `TOP`: the axis `shift` moves along and `travel` reads. An index that
		* never shifts keeps the default.
		*/
		#axis;
		/** The window's first slot. */
		#first = 0;
		/** The slot after the window's last one: the window is empty when it equals `#first`. */
		#end = 0;
		/** Starts stale, at a version no collection holds, so the first read measures. */
		#dirty = true;
		#measured = -1;
		constructor(axis = 0) {
			this.#axis = axis;
		}
		get count() {
			return this.#items.length;
		}
		get first() {
			return this.#first;
		}
		get end() {
			return this.#end;
		}
		/**
		* One packed field of `slot`, or of the placeholder at `PLACEHOLDER`. A slot's
		* value is valid inside the window.
		*/
		at(slot, field) {
			return this.#values[(slot + LEADING) * 6 + field];
		}
		/** The element at `slot`. */
		item(slot) {
			return this.#items[slot];
		}
		/** The insertion at gap `gap` of the destination. */
		insertion(gap) {
			return insertionAt(this.#items, gap);
		}
		/**
		* When the cache is stale, collects every slot's element and measures the
		* placeholder, and leaves an empty window at `slot`. `measure` then adds slots to
		* the window.
		*/
		collect(snapshot, dragged, placeholder, slot) {
			if (!this.#dirty && this.#measured === snapshot.version) return;
			const items = this.#items;
			const values = this.#values;
			let n = 0;
			for (const item of snapshot.items) if (item !== dragged) {
				items[n] = item;
				n += 1;
			}
			for (let i = values.length; i < (n + LEADING) * 6; i += 1) values.push(0);
			this.measurePlaceholder(placeholder);
			items.length = n;
			this.#first = slot;
			this.#end = slot;
			this.#measured = snapshot.version;
			this.#dirty = false;
		}
		/**
		* Measures the slots from `lo` to before `hi`, each settled through the sink, and
		* adds them to the window. The range starts or ends at the window's first or end
		* slot, so the window stays one contiguous range.
		*/
		measure(lo, hi, getBox, displacement) {
			const items = this.#items;
			for (let i = lo; i < hi; i += 1) this.#read(i, items[i], getBox, displacement);
			if (lo < this.#first) this.#first = lo;
			if (hi > this.#end) this.#end = hi;
		}
		/**
		* Moves the slots from `lo` to before `hi` by `delta` along the index's axis: both
		* edges move, and the center is recomputed from them, so the arithmetic is the
		* one a measurement performs. The other axis does not change. The range is inside
		* the window.
		*/
		shift(lo, hi, delta) {
			const values = this.#values;
			const axis = this.#axis;
			for (let i = lo; i < hi; i += 1) {
				const offset = (i + LEADING) * 6 + axis;
				const near = values[offset] + delta;
				const far = values[offset + 2] + delta;
				values[offset] = near;
				values[offset + 2] = far;
				values[offset + 4] = (near + far) * .5;
			}
		}
		/**
		* How far `slot`'s element now is, along the index's axis, from where the cache
		* holds it: measured again like a rebuild measures it, settled through the sink.
		* The slot is inside the window.
		*/
		travel(slot, getBox, displacement) {
			const axis = this.#axis;
			this.#read(-2, this.#items[slot], getBox, displacement);
			return this.#values[axis] - this.#values[(slot + LEADING) * 6 + axis];
		}
		/**
		* Measures the placeholder again, packed like a slot. A rebuild calls it, and so
		* may an axis rule whose committed move relocated the placeholder.
		*/
		measurePlaceholder(placeholder) {
			this.#read(-1, placeholder, null, null);
		}
		/**
		* Whether one slot's element is still where the cache holds it: measured again like
		* a rebuild measures it, and compared on left and top within `EQUIVALENCE_SLACK`. A
		* scroll or a layout shift outside the list moves every slot by one vector, so one
		* slot shows it. A stale cache or an empty window reads nothing and answers
		* `true`, because the next resolution measures what it needs anyway.
		*
		* The slot is the end of the window farther from `gap`, the gap the placeholder
		* holds. Committed moves displace the slots next to the gap, so a slot far from it
		* is the least likely to carry a running animation, and a settled reading that
		* misses by more than the slack costs a new measurement.
		*/
		holds(gap, snapshot, getBox, displacement) {
			const first = this.#first;
			const end = this.#end;
			if (this.#dirty || this.#measured !== snapshot.version || first === end) return true;
			const slot = end - gap > gap - first ? end - 1 : first;
			const values = this.#values;
			const offset = (slot + LEADING) * 6;
			this.#read(-2, this.#items[slot], getBox, displacement);
			return Math.abs(values[0] - values[offset + 0]) <= EQUIVALENCE_SLACK && Math.abs(values[1] - values[offset + 1]) <= EQUIVALENCE_SLACK;
		}
		/** One flag, because any reason for staleness forces the same full rebuild. */
		invalidate() {
			this.#dirty = true;
		}
		/**
		* Empties the element array, which would retain DOM between operations, and keeps
		* the slot array for reuse.
		*/
		retire() {
			this.#items.length = 0;
			this.#dirty = true;
			this.#measured = -1;
		}
		/**
		* Reads `element`'s rect, or its box's, into `slot`, settled: what a running
		* animation of the sink currently carries is subtracted, so the slot holds where
		* flow puts the element, not where the animation draws it.
		*/
		#read(slot, element, getBox, displacement) {
			const values = this.#values;
			const offset = (slot + LEADING) * 6;
			const rect = (getBox ? getBox(element) : element).getBoundingClientRect();
			let dx = 0;
			let dy = 0;
			if (displacement?.displaced(element)) ({dx, dy} = displacement);
			const left = rect.left - dx;
			const top = rect.top - dy;
			const right = rect.right - dx;
			const bottom = rect.bottom - dy;
			values[offset + 0] = left;
			values[offset + 1] = top;
			values[offset + 2] = right;
			values[offset + 3] = bottom;
			values[offset + 4] = (left + right) * .5;
			values[offset + 5] = (top + bottom) * .5;
		}
	};

	function xy() {
		return () => {
			const index = new RectIndex();
			/** The gap the buffer reflects, or `-1` when nothing is known. */
			let last = -1;
			/** The origins held before the DOM write, packed `[left, top]` per slot. */
			let before = /* @__PURE__ */ new Float64Array(0);
			const invalidate = () => {
				last = -1;
				index.invalidate();
			};
			return {
				resolve(query, context) {
					const { pointerX, pointerY } = query;
					refresh(index, query.snapshot, query.item, context.box, context.placeholder, context.displacement);
					last = query.insertion.index;
					const { count } = index;
					const dxAnchor = pointerX - index.at(-1, 4);
					const dyAnchor = pointerY - index.at(-1, 5);
					let best = dxAnchor * dxAnchor + dyAnchor * dyAnchor;
					let nearest = -1;
					for (let i = 0; i < count; i += 1) {
						const dx = pointerX - index.at(i, 4);
						const dy = pointerY - index.at(i, 5);
						const distance = dx * dx + dy * dy;
						if (distance < best) {
							best = distance;
							nearest = i;
						}
					}
					if (nearest === -1) return null;
					const follows = (context.placeholder.compareDocumentPosition(index.item(nearest)) & Node.DOCUMENT_POSITION_FOLLOWING) !== 0;
					return index.insertion(follows ? nearest + 1 : nearest);
				},
				invalidate,
				confirm(query, context) {
					if (!index.holds(query.insertion.index, query.snapshot, context.box, context.displacement)) invalidate();
				},
				moved(query, context) {
					const { displacement } = context;
					const gap = query.insertion.index;
					const held = index.count;
					if (!displacement || last < 0 || last === gap || held === 0) {
						invalidate();
						return;
					}
					if (before.length < held * 2) before = new Float64Array(held * 2);
					for (let i = 0; i < held; i += 1) {
						before[i * 2] = index.at(i, 0);
						before[i * 2 + 1] = index.at(i, 1);
					}
					index.invalidate();
					refresh(index, query.snapshot, query.item, context.box, context.placeholder, displacement);
					last = gap;
					const { count } = index;
					const span = count < held ? count : held;
					for (let i = 0; i < span; i += 1) {
						const dx = before[i * 2] - index.at(i, 0);
						const dy = before[i * 2 + 1] - index.at(i, 1);
						if (dx !== 0 || dy !== 0) displacement.report(index.item(i), dx, dy, context.space);
					}
				},
				retire() {
					last = -1;
					index.retire();
				}
			};
		};
	}

	/** `index` was constructed with `TOP` as its axis. */
	function linearShift(index) {
		/** A prediction is valid only against a buffer that describes the DOM. */
		let dirty = true;
		/** The destination gap the buffer reflects, or `-1` when nothing is known. */
		let lastGap = -1;
		/** The collection version the buffer holds, or `-1` for nothing. */
		let snapshotVersion = -1;
		/**
		* The distance one crossed row travels along the axis, or `-1` while unmeasured.
		* Every invalidation discards it, because a zoom, a resize or a replaced
		* collection all change it, and the cache has one staleness flag with no reason
		* attached. The cost is one row read on the next committed move.
		*/
		let rowDistance = -1;
		/** A committed move left the cached placeholder rect out of date. */
		let placeholderIsStale = false;
		return {
			refresh(snapshot, dragged, getBox, placeholder, displacement, gap, pointerY) {
				if (placeholderIsStale) {
					placeholderIsStale = false;
					index.measurePlaceholder(placeholder);
				}
				index.collect(snapshot, dragged, placeholder, gap);
				const { count } = index;
				if (pointerY > index.at(-1, 5)) while (index.end < count && (index.first === index.end || index.at(index.end - 1, 5) < pointerY)) index.measure(index.end, index.end + 1, getBox, displacement);
				else while (index.first > 0 && (index.first === index.end || index.at(index.first, 5) > pointerY)) index.measure(index.first - 1, index.first, getBox, displacement);
				snapshotVersion = snapshot.version;
				dirty = false;
				lastGap = gap;
			},
			moved(gap, snapshot, context) {
				if (dirty || snapshotVersion !== snapshot.version || index.count === 0 || lastGap < 0 || lastGap === gap) {
					dirty = true;
					lastGap = -1;
					placeholderIsStale = false;
					index.invalidate();
					return;
				}
				const lo = lastGap < gap ? lastGap : gap;
				const hi = lastGap < gap ? gap : lastGap;
				let delta = lastGap < gap ? -rowDistance : rowDistance;
				if (rowDistance < 0) {
					delta = index.travel(lo, context.box, context.displacement);
					rowDistance = delta < 0 ? -delta : delta;
				}
				index.shift(lo, hi, delta);
				if (context.displacement) for (let i = lo; i < hi; i += 1) context.displacement.report(index.item(i), 0, -delta, context.space);
				lastGap = gap;
				placeholderIsStale = true;
			},
			invalidate() {
				dirty = true;
				rowDistance = -1;
				placeholderIsStale = false;
				index.invalidate();
			},
			retire() {
				dirty = true;
				lastGap = -1;
				snapshotVersion = -1;
				rowDistance = -1;
				placeholderIsStale = false;
				index.retire();
			}
		};
	}
