#!/bin/bash
# Prints the unit's tool list, resolved from the current directory (packages/drag).
echo "node $(node --version)"
for p in rolldown tsdown vitest @vitest/browser-playwright playwright typescript typedoc size-limit @size-limit/file; do
  node -e "const {createRequire}=require('module');const r=createRequire(process.cwd()+'/');let f;try{f=r.resolve('$p/package.json')}catch{const path=require('path');let d=path.dirname(r.resolve('$p'));while(!require('fs').existsSync(d+'/package.json'))d=path.dirname(d);f=d+'/package.json'}const j=require(f);console.log('$p',j.version,require('path').relative(process.cwd(),f))"
done
