#!/usr/bin/env bash
# Usage: alloc.sh <out dir>. Three rounds, base then end, each probe in its own launch.
S=/tmp/claude-1000/-workspaces-material-x/4851917c-3fc3-4652-853c-37368bf1d84a/scratchpad
OUT=$1; mkdir -p $OUT
C=/workspaces/material-x/packages/drag/bench/campaign.ts
echo "load-before $(cat /proc/loadavg) $(date -Is)" >> $OUT/load.txt
for r in 1 2 3; do for arm in base end; do
  cd $S/$arm/packages/drag
  node $C run --dir $OUT/campaigns --id $arm-r$r -- \
    "npx vitest run -c vitest.config.ts tests/hot-path.browser.test.ts --silent=false -t 'C6 move probe'" \
    "npx vitest run -c vitest.config.ts tests/hot-path.browser.test.ts --silent=false -t 'C6 warm move probe'" \
    "npx vitest run -c vitest.config.ts tests/hot-path.browser.test.ts --silent=false -t 'VERIFY lifetime probe y idle'" \
    "npx vitest run -c vitest.config.ts tests/hot-path.browser.test.ts --silent=false -t 'VERIFY lifetime probe y operated'"
  echo "campaign $arm-r$r exit=$? $(date -Is) load $(cat /proc/loadavg)" >> $OUT/load.txt
done; done
echo done >> $OUT/load.txt
