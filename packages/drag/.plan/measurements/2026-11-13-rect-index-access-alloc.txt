#!/usr/bin/env bash
# Usage: alloc-campaign.sh <worktree> <id>. Each probe test launched alone, 3 times.
set -uo pipefail
cd "$1/packages/drag"
V="npx vitest run -c vitest.config.ts tests/hot-path.browser.test.ts --silent=false -t"
steps=()
for t in "C6 probe y:" "C6 probe xy:" "C6 move probe" "C6 warm move probe" "RIA sink drop probe"; do
  for r in 1 2 3; do steps+=("$V '$t'"); done
done
date; uptime; nproc; git rev-parse HEAD
node bench/campaign.ts run --id "$2" -- "${steps[@]}"
echo "campaign exit $?"; uptime

## alloc-table.ts
// Usage: node alloc-table.ts <campaign dir>...: one line per probe reading, then
// per-probe means, and for the sink drop probe the bytes under rect-index.ts and
// linear-shift.ts, and under layout-animation.ts apart from `report`.
import { readFileSync, readdirSync } from 'node:fs';
for (const dir of process.argv.slice(2)) {
  console.log(`## ${dir}`);
  const groups = new Map<string, number[]>();
  const logs = readdirSync(dir).filter((f) => f.endsWith('.log')).sort((a, b) => parseInt(a) - parseInt(b));
  for (const log of logs) {
    for (const line of readFileSync(`${dir}/${log}`, 'utf8').split('\n')) {
      const m = /^(C6-PROBE (y|xy)|C6-MOVE-PROBE|C6-WARM-MOVE-PROBE|RIA-SINK-DROP-PROBE)\b.*?per(?:Move|Drop)=([\d.]+) keys=(\{.*\})/.exec(line);
      if (!m) continue;
      const name = m[1]!;
      const keys = JSON.parse(m[4]!) as Record<string, number>;
      let value = Number(m[3]);
      let extra = '';
      if (name === 'RIA-SINK-DROP-PROBE') {
        const cycles = Number(/cycles=(\d+)/.exec(line)![1]);
        const insertions = /insertions=(\d+)/.exec(line)![1];
        const cache = Object.entries(keys).filter(([k]) => k.startsWith('sortable/rect-index.ts') || k.startsWith('sortable/linear-shift.ts')).reduce((s, [, v]) => s + v, 0) / cycles;
        const sink = Object.entries(keys).filter(([k]) => k.startsWith('sortable/layout-animation.ts') && !k.endsWith(':report')).reduce((s, [, v]) => s + v, 0) / cycles;
        extra = ` insertions=${insertions} cacheAndShiftPerDrop=${cache.toFixed(2)} sinkExceptReportPerDrop=${sink.toFixed(2)}`;
        (groups.get(`${name} cache+shift`) ?? groups.set(`${name} cache+shift`, []).get(`${name} cache+shift`)!).push(cache);
        (groups.get(`${name} sink-not-report`) ?? groups.set(`${name} sink-not-report`, []).get(`${name} sink-not-report`)!).push(sink);
      }
      (groups.get(name) ?? groups.set(name, []).get(name)!).push(value);
      console.log(`${log}: ${name} per=${value.toFixed(2)}${extra}`);
    }
  }
  for (const [name, values] of groups) {
    const mean = values.reduce((a, b) => a + b, 0) / values.length;
    console.log(`MEAN ${name}: ${mean.toFixed(2)} (n=${values.length}, min ${Math.min(...values).toFixed(2)}, max ${Math.max(...values).toFixed(2)})`);
  }
}
