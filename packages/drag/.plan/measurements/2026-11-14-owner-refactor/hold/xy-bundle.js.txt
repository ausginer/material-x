				local[1] = sy;

				dy: sy

		/**

		* value is valid inside the window.

			return this.#values[(slot + LEADING) * 6 + field];

		item(slot) {

		insertion(gap) {
			return insertionAt(this.#items, gap);
		}
		/**
		* When the cache is stale, collects every slot's element and measures the
		* placeholder, and leaves an empty window at `slot`. `measure` then adds slots to
		* the window.
		*/
		collect(snapshot, dragged, placeholder, slot) {
			if (!this.#dirty && this.#measured === snapshot.version) return;
			const items = this.#items;
			const values = this.#values;
			let n = 0;
			for (const item of snapshot.items) if (item !== dragged) {
				items[n] = item;
				n += 1;
			}
			for (let i = values.length; i < (n + LEADING) * 6; i += 1) values.push(0);
			this.measurePlaceholder(placeholder);

		}

		* adds them to the window. The range starts or ends at the window's first or end
		* slot, so the window stays one contiguous range.
		*/
		measure(lo, hi, getBox, displacement) {
			const items = this.#items;
			for (let i = lo; i < hi; i += 1) this.#read(i, items[i], getBox, displacement);
			if (lo < this.#first) this.#first = lo;
			if (hi > this.#end) this.#end = hi;
		}
		/**

		* edges move, and the center is recomputed from them, so the arithmetic is the
		* one a measurement performs. The other axis does not change. The range is inside
		* the window.
		*/

			const values = this.#values;
			const axis = this.#axis;
			for (let i = lo; i < hi; i += 1) {
				const offset = (i + LEADING) * 6 + axis;
				const near = values[offset] + delta;

				values[offset] = near;
				values[offset + 2] = far;
				values[offset + 4] = (near + far) * .5;
			}
		}

		* How far `slot`'s element now is, along the index's axis, from where the cache
		* holds it: measured again like a rebuild measures it, settled through the sink.
		* The slot is inside the window.
		*/
		travel(slot, getBox, displacement) {
			const axis = this.#axis;
			this.#read(-2, this.#items[slot], getBox, displacement);
			return this.#values[axis] - this.#values[(slot + LEADING) * 6 + axis];
		}
		/**
		* Measures the placeholder again, packed like a slot. A rebuild calls it, and so
		* may an axis rule whose committed move relocated the placeholder.
		*/
		measurePlaceholder(placeholder) {
			this.#read(-1, placeholder);
		}

		* a rebuild measures it, and compared on left and top within `EQUIVALENCE_SLACK`. A
		* scroll or a layout shift outside the list moves every slot by one vector, so one
		* slot shows it. A stale cache or an empty window reads nothing and answers
		* `true`, because the next resolution measures what it needs anyway.
		*

		* holds. Committed moves displace the slots next to the gap, so a slot far from it
		* is the least likely to carry a running animation, and a settled reading that
		* misses by more than the slack costs a new measurement.
		*/

			const end = this.#end;
			if (this.#dirty || this.#measured !== snapshot.version || first === end) return true;
			const slot = end - gap > gap - first ? end - 1 : first;
			const values = this.#values;

			this.#read(-2, this.#items[slot], getBox, displacement);
			return Math.abs(values[0] - values[offset + 0]) <= EQUIVALENCE_SLACK && Math.abs(values[1] - values[offset + 1]) <= EQUIVALENCE_SLACK;
		}

		invalidate() {

		}

		* the slot array for reuse.
		*/

			this.#measured = -1;
		}

		* animation of the sink currently carries is subtracted, so the slot holds where
		* flow puts the element, not where the animation draws it.
		*/

			const values = this.#values;
			const offset = (slot + LEADING) * 6;
			const rect = (getBox?.(element) ?? element).getBoundingClientRect();

			let dy = 0;

			values[offset + 1] = top;
			values[offset + 2] = right;
			values[offset + 3] = bottom;
			values[offset + 4] = (left + right) * .5;
			values[offset + 5] = (top + bottom) * .5;
		}
	};

		const tops = new Set(field.snapshot.items.slice(1).map((cell) => cell.offsetTop));
		return {
			fieldCount: ITEMS,
			fieldRows: tops.size,
			columnCount: ITEMS,
			fieldPointer: [field.pointerX, field.pointerY],
			columnPointer: [column.pointerX, column.pointerY]
		};
	}
	const CASES = {
		A: {
			axis: "xy",
			sink: false,
			kind: "rebuild"
		},
		M: {
			axis: "xy",
			sink: true,
			kind: "move"
		},
		C: {
			axis: "y",
			sink: false,
			kind: "scan"
		}

	function makeArm(kind, name) {
		const { axis, sink, kind: operationKind } = CASES[name];
		const before = kind === "before";
		return {
			geometry: (axis === "xy" ? before ? xy$1 : xy : before ? y$1 : y)()(),
			sink: sink ? (before ? layoutAnimation$1 : layoutAnimation)()() : null,
			surface: axis === "xy" ? field : column,
			kind: operationKind
		};
	}
	function query(surface) {
		return {
			pointerX: surface.pointerX,
			pointerY: surface.pointerY,
			snapshot: surface.snapshot,
			insertion: surface.insertion,
			item: surface.dragged
		};
	}
	function context({ sink, surface }) {
		return {
			placeholder: surface.placeholder,
			box: null,
			displacement: sink,
			space: SPACE
		};
	}
	function resolve(arm) {
		return arm.geometry.resolve(query(arm.surface), context(arm));
	}
	/**
	* One committed move of case M: the placeholder goes to `to`, one gap away, which
	* displaces one cell, then the arm's `moved` rebuilds with its sink and reports.
	*/
	function commit(arm, to) {
		const { surface } = arm;
		const next = gapAt(surface.destination, to);
		next.before.after(surface.placeholder);
		surface.insertion = next;
		arm.geometry.moved(query(surface), context(arm));
	}
	const HOME = 1001;
	/**
	* One operation. Case A rebuilds and resolves. Case M commits a move one gap
	* forward and one back, so the placeholder, the surface and the arm's last gap are
	* where they were, whichever arm runs next. Case C scans a window that is already
	* grown.
	*/
	function operation(arm) {
		if (arm.kind === "move") {
			commit(arm, 1002);
			commit(arm, HOME);
			return arm.surface.insertion.index;
		}
		if (arm.kind === "rebuild") arm.geometry.invalidate();
		const result = resolve(arm);
		return result ? result.index : -1;
	}
	/** Mean µs per operation over `size` operations, and the batch's duration in ms. */
	function batch(arm, size) {
		const start = performance.now();
		for (let i = 0; i < size; i += 1) operation(arm);
		const ms = performance.now() - start;
		return {
			us: ms * 1e3 / size,
			ms
		};
	}
	/**
	* Rect reads and reports during one operation, counted for that call only. In case
	* M, each report is the vector the arm's `xy()` passed to its sink.
	*/
	function readsOf(arm) {
		const read = Element.prototype.getBoundingClientRect;
		const reports = [];
		const report = arm.sink?.report;
		let reads = 0;
		Element.prototype.getBoundingClientRect = function counted() {
			reads += 1;
			return read.call(this);
		};
		if (arm.sink && report) arm.sink.report = function recorded(element, dx, dy, space) {
			reports.push(`${element.textContent} ${dx} ${dy}`);
			report.call(this, element, dx, dy, space);
		};
		try {
			const result = operation(arm);
			return {
				reads,
				result,
				reports
			};
		} finally {
			Element.prototype.getBoundingClientRect = read;
			if (arm.sink && report) arm.sink.report = report;
		}
	}
	const arms = /* @__PURE__ */ new Map();
	/**
	* Creates the three arms of one case, each with its own installer instance. In case
	* C, one resolve at the pointer grows each arm's window to the end of the list. In
	* case M, one resolve builds the arm's index with its sink, and sets its last gap
	* to the placeholder's. Then one operation per arm is counted for rect reads, and
	* its answer recorded.
	*/
	function prepare(name) {
		arms.clear();
		for (const kind of [
			"before",
			"production",
			"control"
		]) {
			const arm = makeArm(kind, name);
			arms.set(kind, arm);
			if (arm.kind !== "rebuild") resolve(arm);
		}
		const check = {};
		for (const [kind, arm] of arms) check[kind] = readsOf(arm);
		return check;
	}
	/**
	* Doubles the batch size until a production batch lasts at least `ms`, after one
	* discarded batch, so the size is set on warm code.
	*/
	function calibrate(ms) {
		let size = 1;
		batch(arms.get("production"), 16);
		while (batch(arms.get("production"), size).ms < ms) size *= 2;
		return size;
	}
	/** One round: each arm in `order` discards a warm-up batch, then times one batch. */
	function round(order, size) {
		const readings = {};
		for (const kind of order) {
			const arm = arms.get(kind);
			batch(arm, size);
			readings[kind] = batch(arm, size);
		}
		return readings;
	}
