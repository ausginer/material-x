import { firefox, webkit } from 'playwright';
for (const [name, type] of [['firefox', firefox], ['webkit', webkit]]) {
  try {
    const b = await type.launch();
    const p = await b.newPage();
    console.log(name, 'launched', b.version(), await p.evaluate(() => navigator.userAgent));
    await b.close();
  } catch (e) {
    console.log(name, 'FAILED', String(e.message).split('\n').slice(0, 6).join('\n'));
  }
}
