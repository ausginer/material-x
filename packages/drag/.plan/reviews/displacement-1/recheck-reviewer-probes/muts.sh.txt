#!/bin/bash
# muts.sh <engine>: every mutation against the rows, and the reviewer's own against the two commit probes.
export Q=/tmp/claude-1000/-workspaces-worktrees-b/e64c54e5-c6df-4cae-91a9-3ec47d1d631e/scratchpad/q
e=${1:?}
cd /workspaces/worktrees/f/packages/drag
node $Q/mut.mjs $e all tests/displacement.browser.test.ts > $Q/mut-rows-$e.txt 2>&1
export VITE_SETS='[{"axis":"y","env":"none","vis":"item"},{"axis":"xy","env":"none","vis":"item"}]'
OWN="own: the drop's gap anchored on its next row alone;own: the gap restored only in a return home;own: the gap restored only for an accepted item;own: the gap restored on the far side of its anchor"
node $Q/mut.mjs $e "$OWN" tests/proof-ends.browser.test.ts > $Q/mut-ends-$e.txt 2>&1
node $Q/mut.mjs $e "$OWN" tests/proof-commit-around.browser.test.ts > $Q/mut-around-$e.txt 2>&1
git -C /workspaces/worktrees/f status --short src
echo done $e
