function build(o: Opt, config: any = {}): void {
  PH.textContent = o.rowCss?.includes('/*ph*/') ? '[data-drag-placeholder]{margin-bottom:8px}' : o.rowCss?.includes('float') ? '[data-drag-placeholder]{float:left}' : '';
  log = [];
  const outer = document.createElement('div');
  outer.style.cssText = `position:absolute;top:30px;left:60px;transform-origin:0 0;${
    o.env === 'scale' ? 'transform:scale(.75,1.25);' : o.env === 'rot' ? 'transform:rotate(15deg) scale(.8);' : o.env === 'zoom' ? 'zoom:1.5;' : ''}`;
  list = document.createElement('div');
  list.id = 'list';
  list.style.cssText = o.axis === 'xy' ? 'display:flex;flex-wrap:wrap;width:220px;align-content:flex-start;' : 'width:120px;';
  list.style.cssText += o.css ?? '';
  for (let i = 0; i < 6; i += 1) {
    const r = document.createElement('div');
    r.id = `r${i}`; r.tabIndex = 0;
    r.style.cssText = `touch-action:none;user-select:none;box-sizing:border-box;${o.rowCss ?? ''}`;
    const w = o.axis === 'xy' ? [60, 90, 70, 60, 80, 50][i] : 100;
    const h = o.axis === 'xy' ? 40 : [40, 60, 30, 50, 40, 36][i];
    if (o.vis === 'item') { r.style.width = `${w}px`; r.style.height = `${h}px`; }
    else { const k = document.createElement('div'); k.className = 'k'; k.style.cssText = `width:${w}px;height:${h}px`; r.append(k); if (o.vis !== 'child') r.style.padding = '6px'; }
    list.append(r);
  }
  outer.append(list); document.body.append(outer);
  controller = sortable(list, {
    items: () => R(),
    axis: o.axis === 'xy' ? xy() : y(),
    displacement: layoutAnimation({ timing: () => ({ duration: HOUR, easing: 'linear' }) }),
    ...(o.land ? { landing: landing({ duration: 80 }) } : {}),
    ...(o.vis === 'item' ? {} : { visual: (i: HTMLElement) => i.firstElementChild as HTMLElement }),
    ...(o.vis === 'shellbox' ? { box: (i: HTMLElement) => i } : {}),
    onStart() { log.push('start'); },
    onReorder({ item, next }) { list.insertBefore(item, next); return ReorderResolution.accept(); },
    onEnd(r: any) { log.push(r.type); },
    ...config,
  } as any);
  onCleanup(() => controller.requestDestroy());
}

/** Samples each frame. `jump` is the greatest change of a row between two samples. */
function trace(item: HTMLElement, withShell = false) {
  const rows = R().filter((r) => r !== item);
  const start = new Map(rows.map((r) => [r, c(r)]));
  let last = new Map(start); let shell = c(item); let shellJump = 0;
  let jump = 0; let at = ''; let n = 0; let live = true;
  const sample = (): void => {
    n += 1;
    for (const r of rows) {
      if (!r.isConnected) continue;
      const p = c(r); const q = last.get(r)!; const d = Math.hypot(p[0] - q[0], p[1] - q[1]);
      if (d > jump) { jump = d; at = `${r.id}@${n}`; }
      last.set(r, p);
    }
    const s = c(item); shellJump = Math.max(shellJump, Math.hypot(s[0] - shell[0], s[1] - shell[1])); shell = s;
  };
  const step = (): void => { if (live) { sample(); requestAnimationFrame(step); } };
  requestAnimationFrame(step);
  return {
    rebase(): void { last = new Map(rows.map((r) => [r, c(r)])); },
    stop(tag: string): void {
      sample(); live = false;
      const left = rowAnims().length;
      finish();
      const moved = Math.max(...rows.filter((r) => r.isConnected).map((r) => { const a = start.get(r)!; const b = c(r); return Math.hypot(a[0] - b[0], a[1] - b[1]); }));
      OUT.push(`${jump < 1 ? 'ok  ' : 'JUMP'} ${engine} ${tag} | jump ${jump.toFixed(2)} ${jump < 1 ? '' : at} | moved ${moved.toFixed(1)} | anims ${left} | ${order()} | ${log.join()} | ${faultMessages().join(';')}${withShell ? ` | shell ${shellJump.toFixed(1)}` : ''}${list.querySelector('[data-drag-placeholder]') ? ' | PLACEHOLDER LEFT' : ''}`);
    },
  };
}
const key = async (id: string, k: string): Promise<void> => { $(id).focus(); await commands.keyPress(k); };
async function hold(id: string, to: string, rest = true): Promise<void> {
  const [sx, sy] = c($(id)); const [tx, ty] = c($(to));
  await commands.pointerMove(sx, sy); await commands.pointerDown();
  await commands.pointerMove(sx + 6, sy + 6, 2); await frames();
  await commands.pointerMove(tx + 3, ty + 3, 3); await frames(3);
  if (rest) finish();
}
const defer = () => { const { promise, resolve, reject } = Promise.withResolvers<any>(); return { fn: () => promise, resolve, reject }; };
const A = () => ReorderResolution.accept(); const X = () => ReorderResolution.reject();
const before: any = ({ item, next }: any) => { list.insertBefore(item, next); return A(); };
const after: any = ({ item, previous }: any) => { if (previous) previous.after(item); else list.prepend(item); return A(); };
const neighbors: any = ({ item, from, to }: any) => { const r = R(); if (to > from) for (const m of r.slice(from + 1, to + 1)) item.before(m); else item.after(...r.slice(to, from)); return A(); };
const toEnd: any = ({ item }: any) => { list.append(item); return A(); };
