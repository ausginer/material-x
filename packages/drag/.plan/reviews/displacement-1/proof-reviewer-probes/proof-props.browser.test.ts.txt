/** Samples each frame. `jump` is the greatest change of a row between two samples. */
function trace(item: HTMLElement, withShell = false) {
  const rows = R().filter((r) => r !== item);
  const start = new Map(rows.map((r) => [r, c(r)]));
  let last = new Map(start); let shell = c(item); let shellJump = 0;
  let jump = 0; let at = ''; let n = 0; let live = true;
  const sample = (): void => {
    n += 1;
    for (const r of rows) {
      if (!r.isConnected) continue;
      const p = c(r); const q = last.get(r)!; const d = Math.hypot(p[0] - q[0], p[1] - q[1]);
      if (d > jump) { jump = d; at = `${r.id}@${n}`; }
      last.set(r, p);
    }
    const s = c(item); shellJump = Math.max(shellJump, Math.hypot(s[0] - shell[0], s[1] - shell[1])); shell = s;
  };
  const step = (): void => { if (live) { sample(); requestAnimationFrame(step); } };
  requestAnimationFrame(step);
  return {
    rebase(): void { last = new Map(rows.map((r) => [r, c(r)])); },
    stop(tag: string): void {
      sample(); live = false;
      const left = rowAnims().length;
      finish();
      const moved = Math.max(...rows.filter((r) => r.isConnected).map((r) => { const a = start.get(r)!; const b = c(r); return Math.hypot(a[0] - b[0], a[1] - b[1]); }));
      OUT.push(`${jump < 1 ? 'ok  ' : 'JUMP'} ${engine} ${tag} | jump ${jump.toFixed(2)} ${jump < 1 ? '' : at} | moved ${moved.toFixed(1)} | anims ${left} | ${order()} | ${log.join()} | ${faultMessages().join(';')}${withShell ? ` | shell ${shellJump.toFixed(1)}` : ''}${list.querySelector('[data-drag-placeholder]') ? ' | PLACEHOLDER LEFT' : ''}`);
    },
  };
}

}

function counted(): { items(): HTMLElement[]; pulls: () => number; reads: () => number } {
  let pulls = 0; let reads = 0;
  return { pulls: () => pulls, reads: () => reads, items: () => { pulls += 1; return new Proxy(R(), { get(target, p, r) { if (p === 'length' || /^[0-9]+$/u.test(String(p)) || typeof target[p as any] === 'function') reads += 1; return Reflect.get(target, p, r); } }); } };
}
const P0: any = Element.prototype; const H0: any = HTMLElement.prototype;
async function census(f: () => Promise<unknown>): Promise<string> {
  const n = { rect: 0, offset: 0, style: 0, mm: 0, cmp: 0, anim: 0 };
  const g = P0.getBoundingClientRect; const cs = window.getComputedStyle; const mm = window.matchMedia; const cmp = Node.prototype.compareDocumentPosition; const an = P0.animate;
  const oh = Object.getOwnPropertyDescriptor(H0, 'offsetHeight')!; const ow = Object.getOwnPropertyDescriptor(H0, 'offsetWidth')!;
  P0.getBoundingClientRect = function (this: Element) { n.rect += 1; return g.call(this); };
  window.getComputedStyle = ((...a: any[]) => { n.style += 1; return (cs as any).apply(window, a); }) as any;
  window.matchMedia = ((q: string) => { n.mm += 1; return mm.call(window, q); }) as any;
  Node.prototype.compareDocumentPosition = function (this: Node, o: Node) { n.cmp += 1; return cmp.call(this, o); };
  P0.animate = function (this: Element, ...a: any[]) { n.anim += 1; return an.apply(this, a); };
  Object.defineProperty(H0, 'offsetHeight', { ...oh, get() { n.offset += 1; return oh.get!.call(this); } });
  Object.defineProperty(H0, 'offsetWidth', { ...ow, get() { n.offset += 1; return ow.get!.call(this); } });
  try { await f(); await frames(3); } finally {
    P0.getBoundingClientRect = g; window.getComputedStyle = cs; window.matchMedia = mm; Node.prototype.compareDocumentPosition = cmp; P0.animate = an;
    Object.defineProperty(H0, 'offsetHeight', oh); Object.defineProperty(H0, 'offsetWidth', ow);
  }
  return JSON.stringify(n);
}

// ---- natural timing: no frame draws a row further from its end than the frame before, and no step beats the speed
const DUR = 500;
async function natural(tag: string, axis: Axis, cfg: any, prep: () => Promise<void>, act: () => unknown, easing = 'linear', id = 'r1', land = false): Promise<void> {
  build(base(axis, land), { displacement: layoutAnimation(opts(DUR, easing)), ...cfg });
  await prep();
  const rows = R().filter((r) => r.id !== id); const samples: [number, Map<HTMLElement, [number, number]>][] = []; let live = true;
  const step = (): void => { if (live) { samples.push([performance.now(), new Map(rows.map((r) => [r, c(r)]))]); requestAnimationFrame(step); } };
  samples.push([performance.now(), new Map(rows.map((r) => [r, c(r)]))]);
  requestAnimationFrame(step);
  const mark = (e: KeyboardEvent): void => { if (e.key === 'Escape') { samples.length = 0; samples.push([performance.now(), new Map(rows.map((r) => [r, c(r)]))]); } }; addEventListener('keydown', mark, { capture: true, once: true });
  await act(); await wait(DUR + 300); live = false; removeEventListener('keydown', mark, { capture: true });
  let back = 0; let fast = 0; let moved = 0;
  for (const r of rows) {
    const end = c(r); let last = Infinity; let prev = samples[0]!;
    for (const s of samples) { const p = s[1].get(r)!; const dd = Math.hypot(p[0] - end[0], p[1] - end[1]); moved = Math.max(moved, dd); if (dd > last + 0.05) back = Math.max(back, dd - last); last = dd;
      const q = prev[1].get(r)!; const stepd = Math.hypot(p[0] - q[0], p[1] - q[1]); const dt = s[0] - prev[0]; fast = Math.max(fast, stepd - (dt + 34) * 0.5); prev = s; }
  }
  const left = [...list.children].flatMap((e) => e.getAnimations()).length;
  say((back < 1 && fast < 1 ? 'ok   ' : 'JUMP ') + tag + ' frames ' + samples.length + ' moved ' + moved.toFixed(1) + ' back ' + back.toFixed(2) + ' over speed ' + fast.toFixed(1) + ' left ' + left + ' ' + order() + ' ' + log.join());
}
