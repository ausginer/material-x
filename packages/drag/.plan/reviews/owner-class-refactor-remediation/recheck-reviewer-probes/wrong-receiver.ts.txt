import type { Drag, DraggableController, BehaviorOperation } from '../src/draggable.ts';
import type { DisplacementSink, InsertionGeometry } from '../src/sortable/axis.ts';

export function swapped(
  drag: Drag,
  controller: DraggableController<BehaviorOperation>,
  geometry: InsertionGeometry,
  sink: DisplacementSink,
): void {
  drag.use(geometry.retire, sink); // W1: the axis's retire, on the sink
  controller.use(sink.retire, geometry); // W2: the sink's retire, on the axis
}
