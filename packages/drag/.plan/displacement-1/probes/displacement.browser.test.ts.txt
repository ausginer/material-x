const D = 300;
const H = [40, 60, 30, 50, 40, 40];
const LA = (): SortableConfig['displacement'] => (layoutAnimation as any)({ duration: D, easing: 'linear', timing: () => ({ duration: D, easing: 'linear' }) });
const lines: string[] = []; const out = (s: string): void => { lines.push(`${engine} | ${s}`); };

function make(css = '', wrap = false): HTMLElement {
  const list = makeList(6, document.body, `position:absolute;top:0;left:0;width:200px;${css}`);
  rowsOf(list).forEach((r, i) => {
    r.tabIndex = 0;
    r.style.cssText = wrap
      ? `box-sizing:border-box;margin:0;width:${[60, 90, 70, 60, 80, 60][i]}px;height:40px;touch-action:none;user-select:none`
      : `box-sizing:border-box;margin:0;width:100px;height:${H[i]}px;touch-action:none;user-select:none`;
  });
  return list;
}

/** Frame-by-frame positions of every row, and whether a row moved in one frame by nearly all it travels. */
function record(list: HTMLElement): () => string {
  const rows = rowsOf(list);
  const tracks: Array<Array<[number, number]>> = rows.map(() => []); const homes = list;
  const anim: string[] = [];
  const orig = Element.prototype.animate;
  Element.prototype.animate = function (this: Element, ...a: any[]): Animation {
    const k = a[0]?.[0]?.translate;
    if (rows.includes(this as HTMLElement) && a[1]?.composite === 'add') anim.push(`${this.id}:${k}`);
    return orig.apply(this, a as any);
  };
  let live = true;
  const tick = (): void => {
    if (!live) return;
    rows.forEach((r, i) => { const b = r.getBoundingClientRect(); const h = homes.getBoundingClientRect(); tracks[i]!.push([b.left - h.left, b.top - h.top]); });
    requestAnimationFrame(tick);
  };
  requestAnimationFrame(tick);
  return () => {
    live = false;
    Element.prototype.animate = orig;
    const verdict = tracks.map((t, i) => {
      const xs = t.map((v) => v[0]); const ys = t.map((v) => v[1]);
      let max = 0;
      for (let k = 1; k < t.length; k += 1) { const d = Math.abs(ys[k]! - ys[k - 1]!) + Math.abs(xs[k]! - xs[k - 1]!); if (d > max) max = d; }
      return max > 12 ? `${rows[i]!.id}:jump${Math.round(max)}` : max > 0.5 ? `${rows[i]!.id}:smooth` : '';
    }).filter(Boolean).join(' ');
    return `${verdict || 'still'} || animate ${anim.join(' ') || '-'}`;
  };
}

async function key(list: HTMLElement, id: string, k = 'ArrowDown'): Promise<void> {
  list.querySelector<HTMLElement>(`#${id}`)!.focus();
  await frames(2);
  await commands.keyPress(k);
}
async function slow(list: HTMLElement, id: string, dy: number): Promise<void> {
  const r = list.querySelector(`#${id}`)!.getBoundingClientRect();
  await commands.pointerMove(r.left + 20, r.top + 10);
  await commands.pointerDown();
  await commands.pointerMove(r.left + 20, r.top + 10 + dy, 25);
  await wait(D + 150);
}
async function flick(list: HTMLElement, id: string, dy: number, dx = 0): Promise<void> {
  const r = list.querySelector(`#${id}`)!.getBoundingClientRect();
  await commands.pointerMove(r.left + 20, r.top + 10);
  await commands.pointerDown();
  // one sample and the release in one task, as a very quick drop
  const el = list.querySelector(`#${id}`)!;
  void el;
  await commands.pointerMove(r.left + 20 + dx, r.top + 10 + dy, 1);
  await commands.pointerUp();
}

type Case = Readonly<{ name: string; run(list: HTMLElement): Promise<void>; cfg?: Partial<SortableConfig>; wrap?: boolean; css?: string }>;
const accept = { onReorder: ({ item, next }: any) => { const l = item.parentElement; l.insertBefore(item, next); return ReorderResolution.accept(); } };
const later = { onReorder: ({ item, next }: any) => new Promise<any>((res) => { setTimeout(() => { item.parentElement.insertBefore(item, next); res(ReorderResolution.accept()); }, 100); }) };
const never = { onReorder: () => new Promise<any>((res) => { setTimeout(() => { res(ReorderResolution.accept()); }, 400); }) };
const reject = { onReorder: () => ReorderResolution.reject() };
const rejectLater = { onReorder: () => new Promise<any>((res) => { setTimeout(() => { res(ReorderResolution.reject()); }, 400); }) };
// A commit that moves the neighbor and not the item, as a keyed renderer may.
const movesNeighbor = { onReorder: ({ item, from, to }: any) => { const l: HTMLElement = item.parentElement; const rs = rowsOf(l); if (to > from) { for (let i = to; i > from; i -= 1) l.insertBefore(rs[i]!, item); } else { l.insertBefore(item, rs[to]!); } return ReorderResolution.accept(); } };
const L = { landing: landing({ duration: D }) };

describe('PROBE', () => { afterAll(async () => { await server.commands.writeFile('../zz-out-' + engine + '.txt', lines.join('\n')); });
  it.each(cases)('$name', async ({ name, run, cfg, wrap = false, css = '' }) => {
    const list = make(css, wrap);
    const arr = rowsOf(list); document.body.style.minHeight = '3000px'; onCleanup(() => { document.body.style.minHeight = ''; document.documentElement.scrollTop = 0; });
    (window as any).ctl = basicSortable(list, { items: () => ((cfg as any)?.__clear ? arr : rowsOf(list)), displacement: LA(), ...cfg } as any);
    if ((cfg as any)?.__clear) setTimeout(() => { arr.length = 0; }, 350);
    await frames(3);
    const stop = record(list);
    await frames(2);
    await run(list);
    await wait(D + 600);
    out(`${name} | ${stop()} | ${rowsOf(list).map((r) => r.id).join('')}${faults.length ? ` | FAULTS ${faults.map((f) => f.message).join()}` : ''}`);
    faults.length = 0;
    expect(true).toBe(true);
  }, 30000);

  it('reads of getBoundingClientRect in a key move, with and without displacement', async () => {
    for (const [label, disp] of [['with', LA()], ['without', undefined]] as const) {
      for (const axis of [undefined, xy()]) {
        const list = make(axis ? 'display:flex;flex-wrap:wrap;' : '', Boolean(axis));
        basicSortable(list, { items: () => rowsOf(list), ...(disp ? { displacement: disp } : {}), ...(axis ? { axis } : {}), ...accept, ...L } as any);
        await frames(2);
        list.querySelector<HTMLElement>('#r0')!.focus();
        const orig = Element.prototype.getBoundingClientRect; let n = 0;
        Element.prototype.getBoundingClientRect = function (this: Element) { n += 1; return orig.call(this); };
        await commands.keyPress('ArrowDown');
        Element.prototype.getBoundingClientRect = orig;
        const keyReads = n; await wait(D + 100);
        const one = async (esc: boolean): Promise<number> => { const r = rowsOf(list)[0]!; const b = r.getBoundingClientRect(); await commands.pointerMove(b.left + 10, b.top + 10); await commands.pointerDown(); await commands.pointerMove(b.left + 10, b.top + 120, 8); await wait(D + 100); let k = 0; Element.prototype.getBoundingClientRect = function (this: Element) { k += 1; return orig.call(this); }; if (esc) await commands.keyPress('Escape'); else await commands.pointerUp(); Element.prototype.getBoundingClientRect = orig; await wait(D + 100); if (esc) await commands.pointerUp(); await wait(50); return k; };
        const drop = await one(false); const esc = await one(true);
        out(`reads ${label} ${axis ? 'xy' : 'y'}: key ${keyReads}, drop after a resolved move ${drop}, Escape ${esc}`);
        await wait(D + 100);
        list.remove();
      }
    }
  }, 30000);

  it('a row in flight that is then moved by key', async () => {
    const list = make();
    basicSortable(list, { items: () => rowsOf(list), displacement: LA(), ...accept, ...L } as any);
    const r = list.querySelector('#r0')!.getBoundingClientRect();
