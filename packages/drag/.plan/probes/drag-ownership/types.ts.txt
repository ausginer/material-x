// drag:DRD-27 type check. A scratch model of the public and kernel types, not production code. See README.md.
type Space = Float64Array;
export type Lift = Readonly<{ visual: HTMLElement; rect: DOMRectReadOnly; visualSpace: Space; translatedX: number; translatedY: number; translate(x: number, y: number): void }>;
export type Landing = Readonly<{ x: number; y: number }>;
export type LiftHold = Readonly<{ view: Lift; release(): void; land(l: Landing): (() => void) | null }>;
export type Admission<State> = Readonly<{ item: HTMLElement; handle?: HTMLElement; visual?: never; box?: never }> &
  (undefined extends State ? Readonly<{ state?: State }> : Readonly<{ state: State }>);
export type LiftedAdmission<State> = Omit<Admission<State>, 'visual' | 'box'> & Readonly<{ visual?: HTMLElement; box?: HTMLElement }>;
export type Drag<State = undefined, L extends Lift | undefined = undefined> = Readonly<{
  item: HTMLElement; state: State; x: number; y: number; lift: L;
  requestStep(step: (drag: Drag<State, L> | null) => void): void;
}>;
export type LiftedDrag<State = undefined> = Drag<State, Lift>;
export function lift<State>(item: HTMLElement, admission: LiftedAdmission<State>): LiftHold {
  const view = { visual: admission.visual ?? item, rect: item.getBoundingClientRect(), visualSpace: new Float64Array(6), translatedX: 0, translatedY: 0,
    translate(x: number, y: number) { view.translatedX = x; view.translatedY = y; } };
  return { view, release() {}, land: () => null };
}
type Hooks<Op, Ad, Settled> = Readonly<{ admit(e: PointerEvent): Ad | null; activate?(d: Op): false | undefined; move?(d: Op): void; settle?(d: Op): Settled; end?(d: Op): void }>;
export type Behavior<S = undefined> = Hooks<Drag<S>, Admission<S>, null | undefined> & Readonly<{ lift?: undefined }>;
export type LiftedBehavior<S = undefined> = Hooks<LiftedDrag<S>, LiftedAdmission<S>, Landing | null | undefined> & Readonly<{ lift: typeof lift }>;
// Kernel's honest view: the operation may or may not have a lift.
type KernelDrag<S> = Drag<S, Lift | undefined>;
type KernelBehavior<S> = Hooks<KernelDrag<S>, LiftedAdmission<S>, Landing | null | undefined> & Readonly<{ lift?: typeof lift | undefined }>;
export type Controller<Op> = Readonly<{ requestStep(step: (d: Op | null) => void): void }>;

class Session<S> {
  #lift: LiftHold | undefined;
  #drag: { -readonly [K in keyof KernelDrag<S>]: KernelDrag<S>[K] } | null = null;
  #x = 0; #y = 0;
  constructor(readonly behavior: KernelBehavior<S>, readonly admission: LiftedAdmission<S>) {}
  activate(): void {
    const admission = this.admission;
    this.#lift = this.behavior.lift?.(admission.item, admission);
    const self = this;
    const drag: { -readonly [K in keyof KernelDrag<S>]: KernelDrag<S>[K] } = {
      item: admission.item, state: admission.state as S /* pre-existing assertion, out of scope */, x: this.#x, y: this.#y, lift: this.#lift?.view,
      requestStep() {},
    };
    if (this.behavior.activate?.(drag) === false) return;
    this.#drag = drag;
    void self;
  }
  sample(x: number): void { const d = this.#drag; if (d) { d.x = x; this.behavior.move?.(d); } }
}
export function draggable<S = undefined>(root: HTMLElement, b: LiftedBehavior<S>): Controller<LiftedDrag<S>>;
export function draggable<S = undefined>(root: HTMLElement, b: Behavior<S>): Controller<Drag<S>>;
export function draggable<S>(root: HTMLElement, b: KernelBehavior<S>): Controller<KernelDrag<S>> {
  void root; const a = b.admit(new PointerEvent('pointerdown')); if (a) new Session(b, a).activate();
  return { requestStep() {} };
}
