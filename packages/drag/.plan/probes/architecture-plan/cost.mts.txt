/**
 * Per-move cost of three hit tests, over real pointer moves with one frame per
 * move, by CDP `Performance.getMetrics`. Three runs per cell; medians and range.
 * argv: <prototype dir> <output json>
 */
import { build } from '/workspaces/material-x/node_modules/esbuild/lib/main.js';
import { chromium } from '/workspaces/material-x/node_modules/playwright/index.mjs';
import { writeFileSync } from 'node:fs';
const [, , proto, out] = process.argv;
const code = (await build({ entryPoints: [`${proto}/cost-lab.ts`], bundle: true, format: 'iife', write: false, target: 'es2022', define: { __DEV__: 'false' }, logLevel: 'error' })).outputFiles[0].text;
const browser = await chromium.launch({ executablePath: '/usr/local/bin/chrome', headless: true });
const MOVES = 300;
const ROWS = 9;
const rows: unknown[] = [];
const median = (v: number[]) => [...v].sort((a, b) => a - b)[Math.floor(v.length / 2)];
for (const n of [20, 2000]) {
  for (const arm of ['y', 'hit', 'independent']) {
    const runs: Record<string, number>[] = [];
    for (let run = 0; run < 3; run += 1) {
      const page = await browser.newPage({ viewport: { width: 900, height: 700 } });
      await page.setContent('<!doctype html><body style="margin:40px"></body>');
      await page.addScriptTag({ content: code });
      await page.evaluate(([a, c]) => (globalThis as any).make(a, c), [arm, n]);
      const cdp = await page.context().newCDPSession(page);
      await cdp.send('Performance.enable');
      const metrics = async () => Object.fromEntries((await cdp.send('Performance.getMetrics')).metrics.map((m: any) => [m.name, m.value]));
      const x = 140;
      const y0 = 40 + 40 * 2 + 20;
      await page.mouse.move(x, y0);
      await page.mouse.down();
      await page.mouse.move(x, y0 + 12, { steps: 2 });
      await page.mouse.move(x, y0 + 20);
      await page.waitForTimeout(50);
      const a = await metrics();
      for (let i = 0; i < MOVES; i += 1) {
        const phase = Math.abs(((i / 60) % 2) - 1);
        await page.mouse.move(x, y0 + 20 + phase * ROWS * 40);
        await page.evaluate(() => new Promise(requestAnimationFrame));
      }
      const b = await metrics();
      await page.mouse.up();
      await page.waitForTimeout(30);
      const log = await page.evaluate(() => (globalThis as any).log);
      if (log.length !== 1) throw new Error(`${arm}/${n}: ${JSON.stringify(log)}`);
      runs.push({
        layouts: b.LayoutCount - a.LayoutCount,
        styles: b.RecalcStyleCount - a.RecalcStyleCount,
        layoutUs: ((b.LayoutDuration - a.LayoutDuration) * 1e6) / MOVES,
        styleUs: ((b.RecalcStyleDuration - a.RecalcStyleDuration) * 1e6) / MOVES,
        scriptUs: ((b.ScriptDuration - a.ScriptDuration) * 1e6) / MOVES,
        taskUs: ((b.TaskDuration - a.TaskDuration) * 1e6) / MOVES,
      });
      await page.close();
    }
    const row: Record<string, unknown> = { n, arm };
    for (const key of Object.keys(runs[0])) {
      const v = runs.map((r) => r[key]);
      row[key] = Math.round(median(v));
      if (key === 'taskUs' || key === 'scriptUs') row[`${key}Range`] = [Math.round(Math.min(...v)), Math.round(Math.max(...v))];
    }
    rows.push(row);
    console.log(JSON.stringify(row));
  }
}
await browser.close();
writeFileSync(out, JSON.stringify(rows, null, 1));
