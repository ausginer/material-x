+/**
+ * `dropTargets()`: drag and drop for `freeDrag`. The target under the pointer
+ * is marked with `data-drop-over` while the drag is over it, and the drop's
+ * request names it. A composition without it carries none of this code.
+ */
+import type { Drag } from '../draggable.ts';
+import { createFrameTask } from '../shared/invalidation.ts';
+
+export type DropTargetsOptions = Readonly<{
+  /**
+   * The targets, pulled once per operation. Where targets overlap, the first in
+   * the array wins.
+   */
+  targets(): readonly HTMLElement[];
+}>;
+
+/** What the capability may use of the operation. Nothing enforces it at runtime. */
+export type DropTargetDrag = Pick<Drag, 'x' | 'y' | 'use' | 'requestStep'>;
+
+/**
+ * One operation's drop-target state. `freeDrag` creates it in `activate`, its
+ * operation holds it, and that operation's release ends it.
+ */
+export interface DropTargetTracker {
+  /** The target under the pointer at the last resolution, or `null`. */
+  readonly current: HTMLElement | null;
+  /** A sample: resolution is coalesced to the next frame, which runs as a step. */
+  move(): void;
+  /** Resolves at the release position now, in the drop step. */
+  drop(): void;
+  /** Staleness only: the rects are re-read at the next resolution. */
+  invalidate(): void;
+}
+
+/**
+ * The value a `freeDrag` config holds. It holds no state, so one config serves
+ * every controller. `freeDrag` calls it once per operation, from `activate`. A
+ * third party may supply its own.
+ */
+export type DropTargets = (
+  drag: DropTargetDrag,
+  window: Window,
+) => DropTargetTracker;
+
+const OVER = 'data-drop-over';
+
+export function dropTargets(options: DropTargetsOptions): DropTargets {
+  return (drag, window) => {
+    const rects: DOMRectReadOnly[] = [];
+    let elements: readonly HTMLElement[] = [];
+    let stale = true;
+    let current: HTMLElement | null = null;
+
+    const mark = (next: HTMLElement | null): void => {
+      if (next !== current) {
+        current?.removeAttribute(OVER);
+        next?.setAttribute(OVER, '');
+        current = next;
+      }
+    };
+
+    /** Reads layout only when stale, and allocates nothing otherwise. */
+    const resolve = (): void => {
+      if (stale) {
+        stale = false;
+
+        for (let i = 0; i < elements.length; i += 1) {
+          rects[i] = elements[i]!.getBoundingClientRect();
+        }
+      }
+
+      const { x, y } = drag;
+      let hit: HTMLElement | null = null;
+
+      for (let i = 0; i < elements.length && !hit; i += 1) {
+        const rect = rects[i]!;
+
+        if (
+          x >= rect.left &&
+          x < rect.right &&
+          y >= rect.top &&
+          y < rect.bottom
+        ) {
+          hit = elements[i]!;
+        }
+      }
+
+      mark(hit);
+    };
+
+    const frame = createFrameTask(window, () => {
+      // A frame is foreign code, so its write goes through a step (DRI-8).
+      drag.requestStep(resolve);
+    });
+
+    // Registered before the consumer's `targets()` runs, so the release covers a
+    // throw from it as well as a terminal, a cancel or destruction.
+    drag.use(() => {
+      frame.cancel();
+      mark(null);
+    });
+    elements = options.targets();
+
+    return {
+      get current() {
+        return current;
+      },
+      move() {
+        frame.schedule();
+      },
+      drop() {
+        frame.cancel();
+        resolve();
+      },
+      invalidate() {
+        stale = true;
+      },
+    };
+  };
+}
