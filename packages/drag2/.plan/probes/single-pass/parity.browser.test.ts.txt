/**
 * A differential run: one script, driven through Playwright's real mouse where a
 * pointer is involved, against the production controller and the prototype.
 *
 * A scenario produces a trace of everything a consumer can observe: callbacks in
 * order, their discriminating fields, whether a click reached the page, and the
 * DOM left behind. Equal traces are parity for that script and nothing more.
 * A scenario listed in `EXPECTED_DIFFERENCE` must differ, so a claimed
 * divergence cannot silently stop being one.
 */

const ROW = 40;
const LEFT = 100;
const TOP = 50;
const WIDTH = 200;

const nextFrames = async (count = 2): Promise<void> => {
  for (let i = 0; i < count; i += 1) {
    // oxlint-disable-next-line no-await-in-loop
    await new Promise((resolve) => {
      requestAnimationFrame(resolve);
    });
  }
};

const tick = async (): Promise<void> =>
  new Promise((resolve) => {
    setTimeout(resolve, 0);
  });

const rowY = (index: number): number => TOP + ROW * index + ROW / 2;

class Harness {
  readonly trace: string[] = [];
  readonly list = document.createElement('ul');
  readonly rows: HTMLElement[] = [];
  source: readonly HTMLElement[] = [];
  controller!: SortableController;
  lastPointerId = 1;
  lastSignal: AbortSignal | null = null;
  readonly #abort = new AbortController();

  constructor() {
    const { list } = this;

    list.style.cssText = `position:absolute;left:${LEFT}px;top:${TOP}px;width:${WIDTH}px;margin:0;padding:0;list-style:none`;

    for (let i = 0; i < 8; i += 1) {
      const row = document.createElement('li');

      row.textContent = String(i + 1);
      row.tabIndex = 0;
      row.style.cssText = `height:${ROW}px;box-sizing:border-box;background:#ccc;touch-action:none;user-select:none`;
      list.append(row);
      this.rows.push(row);
    }

    document.body.append(list);
    this.source = [...this.rows];

    const { signal } = this.#abort;

    list.addEventListener(
      'click',
      () => {
        this.trace.push('click reached the page');
      },
      { signal },
    );
    window.addEventListener(
      'pointerdown',
      (event) => {
        this.lastPointerId = event.pointerId;
      },
      { capture: true, signal },
    );
  }

  log = (entry: string): void => {
    this.trace.push(entry);
  };

  describeError = (error: unknown): string => {
    if (error instanceof DraggableError) {
      return `onError error stage=${String(error.stage)} ${error.message}`;
    }

    return `onError warning ${(error as Error).message}`;
  };

  /** The consumer every sortable scenario starts from: logs, applies, accepts. */
  config(overrides: Partial<SortableConfig> = {}): SortableConfig {
    return {
      items: () => this.source,
      axis: y(),
      onStart: () => {
        this.log('onStart');
      },
      onReorder: (request, { signal }) => {
        this.lastSignal = signal;
        this.log(`onReorder ${request.from}->${request.to}`);
        this.apply(request.item, request.after);
        return ReorderResolution.accept();
      },
      onEnd: (result) => {
        this.logEnd(result);
      },
      onError: (error) => {
        this.log(this.describeError(error));
      },
      ...overrides,
    };
  }

  apply(item: HTMLElement, after: HTMLElement | null): void {
    this.list.insertBefore(item, after);
    this.source = [...this.list.children].filter(
      (node) => !node.hasAttribute('data-drag-placeholder'),
    ) as HTMLElement[];
  }

  logEnd(result: {
    type: string;
    reason?: unknown;
    origin?: unknown;
    stage?: unknown;
  }): void {
    this.log(
      `onEnd ${result.type} reason=${String(
        result.reason instanceof Error ? result.reason.message : result.reason,
      )} origin=${String(result.origin)} stage=${String(result.stage)}`,
    );
  }

  /** What the drag left behind, including anything it failed to restore. */
  state(label: string): void {
    const order = [...this.list.children]
      .map((node) =>
        node.hasAttribute('data-drag-placeholder') ? 'P' : node.textContent,
      )
      .join(',');
    const styled = this.rows.filter(
      (row) =>
        row.style.position !== '' ||
        row.style.transform !== '' ||
        row.hasAttribute('popover'),
    ).length;

    this.log(`${label}: order=${order} lifted=${styled}`);
  }

  async press(index: number): Promise<void> {
    await commands.pointerPress(LEFT + 50, rowY(index));
  }

  async moveToRow(index: number, steps = 6): Promise<void> {
    await commands.pointerSweep(LEFT + 50, rowY(index), steps);
    await nextFrames();
  }

  async release(): Promise<void> {
    await commands.pointerRelease();
    await tick();
  }

  async drag(from: number, to: number): Promise<void> {
    await this.press(from);
    await this.moveToRow(to);
    await this.release();
  }

  /**
   * Whether a further drag still works, and what the old controller does with
   * one. Run after every scenario, so a leaked session shows as a refusal.
   */
  async probeAlive(): Promise<void> {
    const before = this.trace.length;

    await this.press(0);
    await this.moveToRow(0, 2);
    await commands.pointerSweep(LEFT + 50, rowY(0) + 12, 3);
    await nextFrames();
    await this.release();
    await nextFrames();
    this.log(
      this.trace.slice(before).some((entry) => entry.startsWith('onStart'))
        ? 'probe: alive'
        : 'probe: inert',
    );
  }

  dispose(): void {
    this.#abort.abort();
    this.list.remove();
  }
}

type Scenario = (h: Harness, impl: Implementation) => Promise<void>;

const key = (target: EventTarget, value: string): void => {
  target.dispatchEvent(
    new KeyboardEvent('keydown', {
      key: value,
      bubbles: true,
      composed: true,
      cancelable: true,
    }),
  );
};

const deferred = <T,>(): {
  promise: Promise<T>;
  resolve: (value: T) => void;
  reject: (error: unknown) => void;
} => {
  let resolve!: (value: T) => void;
  let reject!: (error: unknown) => void;
  const promise = new Promise<T>((res, rej) => {
    resolve = res;
    reject = rej;
  });

  return { promise, resolve, reject };
};

/**
 * Where the prototype is expected to differ, with the reason. Each entry is a
 * statement in the architecture report and is pinned here by requiring the two
 * traces to differ.
 */
type Pin = Readonly<{ why: string; holds(prototype: string[]): boolean }>;

const count = (trace: string[], prefix: string): number =>
  trace.filter((entry) => entry.startsWith(prefix)).length;

/** Asserted on the first drag's portion of a trace, up to the probe. */
const firstDrag = (trace: string[]): string[] =>
  trace.slice(
    0,
    trace.findIndex((entry) => entry.startsWith('after:')),
  );

const EXPECTED_DIFFERENCE: Record<string, Pin> = {
  'divergence: a throw from the crossing step is a fault, not a panic': {
    why: 'one catch per pass: only a throw escaping the loop itself panics, so the controller survives',
    holds: (trace) => trace.at(-1) === 'probe: alive',
  },
  'divergence: onEnd requests destruction, then throws': {
    why: 'the fault in a terminal is reported as stage 14 whatever else the callback requested',
    holds: (trace) => count(firstDrag(trace), 'onError error stage=14') === 1,
  },
  'divergence: invalidate removing the item, then a throw, from onStart': {
    why: 'a superseded warning always follows the terminal; production has both orders',
    holds: (trace) => {
      const drag = firstDrag(trace);

      return (
        drag.findIndex((entry) => entry.startsWith('onEnd canceled')) <
        drag.indexOf('onError warning drag: failure/superseded-by-cancel')
      );
    },
  },
  'divergence: two invalidates in one callback pull once': {
    why: 'requests coalesce into one slot, so items() is pulled once per boundary',
    holds: (trace) => count(firstDrag(trace), 'items pull') === 2,
  },
  'divergence: start failure with a cancel already registered': {

      count(firstDrag(trace), 'onEnd') === 0,
  },
  'divergence: a pointerup dispatched from onMove': {
    why: 'a native sample raised inside a pass is refused, not queued',
    holds: (trace) =>
      trace.indexOf('real release next') < trace.indexOf('onDrop'),
  },
  'divergence: destroy from inside onError keeps the terminal': {
    why: 'the report and the terminal are one step, so a destruction requested in the report applies after the terminal',
    holds: (trace) => count(firstDrag(trace), 'onEnd canceled') === 1,
  },
};

const results: Record<string, unknown> = {};

describe('single-pass prototype against production', () => {
  afterAll(async () => {
    await commands.writeFile(
      'parity-results.json',
      JSON.stringify(results, null, 2),
    );
  });

  const active: Harness[] = [];

  afterEach(async () => {
    for (const harness of active.splice(0)) {
      harness.dispose();
    }

    await commands.pointerRelease().catch(() => {});
  });

  for (const [name, scenario] of Object.entries(SCENARIOS)) {
    it(name, async () => {
      const traces: Record<string, string[]> = {};

      for (const impl of IMPLEMENTATIONS) {
        const harness = new Harness();

        active.push(harness);
        // oxlint-disable-next-line no-await-in-loop
        await scenario(harness, impl);
        // oxlint-disable-next-line no-await-in-loop
        await harness.probeAlive();
        traces[impl.name] = harness.trace;
        // oxlint-disable-next-line no-await-in-loop
        await harness.controller.requestDestroy();
        harness.dispose();
        // oxlint-disable-next-line no-await-in-loop
        await commands.pointerSweep(5, 5, 1);
      }

      results[name] = {
        equal:
          JSON.stringify(traces['production']) ===
          JSON.stringify(traces['prototype']),
        ...traces,
      };

      const pin = EXPECTED_DIFFERENCE[name];

      if (pin) {
        expect(traces['prototype']).not.toEqual(traces['production']);
        expect(pin.holds(traces['prototype']!), pin.why).toBe(true);
      } else {
        expect(traces['prototype']).toEqual(traces['production']);
      }
    });
  }
});

describe('stack depth at the recorded write sites', () => {
  it('counts library frames', async () => {
    const depth: Record<string, number> = {};
    const frames = (): number =>
      (new Error().stack ?? '')
        .split('\n')
        .filter(
          (line) =>
            /\/(src|prototype)\/[^:]*\.ts/.test(line) &&
            !line.includes('/tests/'),
        ).length;
    const { before } = Element.prototype;
    const limit = Error.stackTraceLimit;

    Error.stackTraceLimit = 200;

    try {
      for (const impl of IMPLEMENTATIONS) {
        const harness = new Harness();
        const row = harness.rows[1]!;
        const real = row.style;
        const writes: number[] = [];
        let move = 0;

        Object.defineProperty(row, 'style', {
          configurable: true,
          get: () =>
            new Proxy(real, {
              get(target, property) {
                const value = Reflect.get(target, property, target) as unknown;

                return typeof value === 'function' ? value.bind(target) : value;
              },
              set(target, property, value) {
                if (property === 'transform') {
                  writes.push(frames());
                }

                return Reflect.set(target, property, value, target);
              },
            }),
        });
        Element.prototype.before = function patched(
          this: Element,
          ...nodes: Array<Node | string>
        ): void {
          if ((nodes[0] as Element).hasAttribute?.('data-drag-placeholder')) {
            move = Math.max(move, frames());
          }

          before.apply(this, nodes);
        };

        harness.controller = impl.sortable(harness.list, harness.config());
        // oxlint-disable-next-line no-await-in-loop
        await harness.drag(1, 5);
        writes.sort((a, b) => a - b);
        depth[`${impl.name} transform write, median sample`] =
          writes[writes.length >> 1]!;
        depth[`${impl.name} transform write, deepest`] = writes.at(-1)!;
        depth[`${impl.name} placeholder move`] = move;
        Element.prototype.before = before;
        // oxlint-disable-next-line no-await-in-loop
        await harness.controller.requestDestroy();
        harness.dispose();
      }
    } finally {
      Error.stackTraceLimit = limit;
      Element.prototype.before = before;
    }

    await commands.writeFile('stack-depth.json', JSON.stringify(depth, null, 2));
    expect(depth['prototype placeholder move']).toBeGreaterThan(0);
  });
});

describe('one active pointer sample', () => {
  it('times synthetic samples, alternating implementations', async () => {
    const SAMPLES = 20_000;
    const ROUNDS = 7;
    const times: Record<string, number[]> = { production: [], prototype: [] };

    for (let round = 0; round < ROUNDS; round += 1) {
      for (const impl of IMPLEMENTATIONS) {
        const harness = new Harness();

        harness.controller = impl.sortable(harness.list, harness.config());
        // oxlint-disable-next-line no-await-in-loop
        await harness.press(1);
        // oxlint-disable-next-line no-await-in-loop
        await harness.moveToRow(3);

        const event = new PointerEvent('pointermove', {
          pointerId: harness.lastPointerId,
          bubbles: true,
          clientX: LEFT + 50,
          clientY: rowY(3),
        });
        const started = performance.now();

        for (let i = 0; i < SAMPLES; i += 1) {
          document.dispatchEvent(event);
        }

        times[impl.name]!.push(
          ((performance.now() - started) / SAMPLES) * 1000,
        );
        // oxlint-disable-next-line no-await-in-loop
        await harness.release();
        // oxlint-disable-next-line no-await-in-loop
        await harness.controller.requestDestroy();
        harness.dispose();
      }
    }

    const median = (values: number[]): number =>
      [...values].sort((a, b) => a - b)[values.length >> 1]!;

    await commands.writeFile(
      'sample-time.json',
      JSON.stringify(
        {
          unit: 'microseconds per dispatched sample, one identical event object',
          production: { median: median(times['production']!), runs: times['production'] },
          prototype: { median: median(times['prototype']!), runs: times['prototype'] },
        },
        null,
        2,
      ),
    );
    expect(times['prototype']!.length).toBe(ROUNDS);
  });
});
