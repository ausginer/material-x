// Minimal DOM stubs: listener registries that honour AbortSignal.
export type Registry = { live(type: string): number; fire(type: string, event: unknown): void; add: any; remove: any };
export function registry(): Registry {
  const entries: { type: string; fn: (e: unknown) => void; signal?: AbortSignal }[] = [];
  return {
    add(type: string, fn: any, options?: { signal?: AbortSignal }) {
      const signal = options?.signal;
      if (signal?.aborted) return;
      const entry = { type, fn, signal };
      entries.push(entry);
      signal?.addEventListener('abort', () => { const i = entries.indexOf(entry); if (i >= 0) entries.splice(i, 1); });
    },
    remove() {},
    live(type) { return entries.filter((e) => e.type === type).length; },
    fire(type, event) { for (const e of entries.filter((x) => x.type === type)) e.fn(event); },
  };
}
export function createEnv() {
  const doc = registry();
  const rootReg = registry();
  class FakeElement {}
  const view = { HTMLElement: FakeElement, getSelection: () => null };
  const document = { defaultView: view, addEventListener: doc.add, removeEventListener: doc.remove };
  const root = { ownerDocument: document, addEventListener: rootReg.add, removeEventListener: rootReg.remove, isConnected: true } as unknown as HTMLElement;
  const element = () => Object.assign(new FakeElement(), { ownerDocument: document }) as unknown as HTMLElement;
  const press = (pointerId: number) => ({ type: 'pointerdown', button: 0, isPrimary: true, pointerId, clientX: 0, clientY: 0, preventDefault() {} });
  return { doc, rootReg, root, element, press };
}
