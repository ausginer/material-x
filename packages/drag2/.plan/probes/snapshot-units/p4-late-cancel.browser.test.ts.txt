// P4: a late cancel() raised inside anchorTarget. Run by run.sh, which copies
// this file under tests/ for the package browser project and removes it.
import { afterEach, expect, it } from 'vitest';
import { LIFT_FLAT } from '../../src/kernel/presentation.ts';
import type { BehaviorContext, BehaviorSpec } from '../../src/kernel/spec.ts';
import { draggable } from '../../src/kernel.ts';

type Part = { item: HTMLElement | null };
const ID = 7;
const cleanup: Array<() => void> = [];
afterEach(() => { for (const f of cleanup.splice(0)) f(); });

function setup(anchorHook: (k: BehaviorContext) => void) {
  const root = document.createElement('div');
  const item = document.createElement('div');
  Object.assign(root.style, { position: 'absolute', inset: '0' });
  Object.assign(item.style, { position: 'absolute', left: '10px', top: '20px', width: '60px', height: '30px' });
  root.append(item);
  document.body.append(root);
  root.setPointerCapture = () => {};
  root.releasePointerCapture = () => {};
  const calls: string[] = [];
  let kernel!: BehaviorContext;
  const controller = draggable<BehaviorContext, Part, true>(root, (k) => {
    kernel = k;
    const spec: BehaviorSpec<Part, true> = {
      createFramePart: () => ({ item: null }),
      resetFramePart: (p) => { p.item = null; },
      config: { threshold: 8, liftMode: LIFT_FLAT, actionTags: 1 },
      admit: (_e, draft) => { draft.item = item; return item; },
      activation: {
        prepare: () => true,
        effect: (_c, _p, scope) => { scope.presentation.use(() => calls.push('presentation.released')); calls.push('onStart'); },
      },
      release: { prepare: () => ({ invoke: null }), effect: () => {} },
      settlement: { prepare: () => true, effect: () => calls.push('settlement.effect') },
      action: { prepare: () => true, effect: () => {} },
      moved: () => {},
      anchorTarget: () => { calls.push('anchorTarget'); anchorHook(kernel); return { x: 0, y: 0 }; },
      finalized: (c) => calls.push(`finalized phase=${c.phase}`),
      reportError: (e) => calls.push(`report ${String(e)}`),
      retire: () => calls.push('retire'),
    };
    return { spec, controller: k };
  });
  cleanup.push(() => { void controller.destroy(); root.remove(); });
  const ev = (type: string, x: number, y: number, target: EventTarget = document) =>
    target.dispatchEvent(new PointerEvent(type, { bubbles: true, cancelable: true, pointerId: ID, isPrimary: true, button: 0, buttons: 1, clientX: x, clientY: y }));
  return { item, calls, ev };
}

const flush = () => new Promise((r) => setTimeout(r, 0));

it('P4: a late cancel() raised inside anchorTarget', async () => {
  const h = setup((k) => k.cancel('late'));
  h.ev('pointerdown', 10, 10, h.item);
  h.ev('pointermove', 40, 10);
  h.ev('pointerup', 40, 10);
  await flush();
  const first = [...h.calls];
  h.calls.length = 0;
  // A second, independent drag.
  h.ev('pointerdown', 10, 10, h.item);
  h.ev('pointermove', 40, 10);
  console.log('P4 first:', JSON.stringify(first));
  console.log('P4 second press:', JSON.stringify(h.calls));
  expect(true).toBe(true);
});

it('P4-control: no cancel inside anchorTarget', async () => {
  const h = setup(() => {});
  h.ev('pointerdown', 10, 10, h.item);
  h.ev('pointermove', 40, 10);
  h.ev('pointerup', 40, 10);
  await flush();
  console.log('P4-control:', JSON.stringify(h.calls));
});
